% Dinámica real + energética + ZypyZape 5 unidades
% Víctor Manzanares Alberola — Mayo 2026
% Asistencia: Claude

clear all; close all; clc;

%% ============================================================
%% PARÁMETROS
%% ============================================================

% Geometría
R_rot = 5.0;          % radio de rotación [m]
r_min = 3.0;          % radio mínimo [m]
r_max = 7.0;          % radio máximo [m]
L_carril = 2.0;       % longitud carril [m]
m_bola = 500;         % masa por bola [kg]
N_bolas = 3;          % número de bolas (desfase 120°)

% Rotación
omega_rot = 1.0;      % velocidad angular [rad/s]
T_ciclo = 2*pi/omega_rot; % período [s]

% Fuerzas
g = 9.81;             % gravedad [m/s²]
rho_fluido = 1000;    % densidad fluido (agua) [kg/m³]
mu_k = 0.05;          % coef. fricción
c_amort = 100;        % coef. amortiguamiento [N·s/m]
A_slide = 0.5;        % amplitud deslizamiento [m/s]

% Simulación
T_sim = 86400;        % tiempo total: 1 día [s]
dt = 0.01;            % paso integración [s]
N_steps = T_sim / dt;

% ZypyZape
N_km = 5;             % número de Kilómetros en enjambre

fprintf('=== SIMULACIÓN KILÓMETRO v1 ===\n');
fprintf('Período ciclo: %.2f s (%.2f ciclos/hora)\n', T_ciclo, 3600/T_ciclo);
fprintf('Masa total: %.0f kg × %d = %.0f kg\n', m_bola, N_bolas, m_bola*N_bolas);
fprintf('Tiempo simulado: %.0f s (%.1f horas, %.1f días)\n', T_sim, T_sim/3600, T_sim/86400);
fprintf('Pasos integración: %.0f\n\n', N_steps);

%% ============================================================
%% INICIALIZACIÓN ARRAYS
%% ============================================================

t = linspace(0, T_sim, N_steps)';
dt_array = diff(t); dt_array = [dt; dt_array];

% Una bola (i=1); generalizamos después
r = zeros(N_steps, 1);
v_r = zeros(N_steps, 1);
a_r = zeros(N_steps, 1);

% Posición inicial
r(1) = (r_min + r_max) / 2;  % radio medio
v_r(1) = 0;

% Energías
E_pot = zeros(N_steps, 1);
E_cin_rot = zeros(N_steps, 1);
E_cin_radial = zeros(N_steps, 1);
P_gen = zeros(N_steps, 1);
P_cons = zeros(N_steps, 1);
P_friccion = zeros(N_steps, 1);
eta_inst = zeros(N_steps, 1);

% Fuerzas
F_centrifuga = zeros(N_steps, 1);
F_friccion_f = zeros(N_steps, 1);
F_neta = zeros(N_steps, 1);

% Control
v_slide_cmd = zeros(N_steps, 1);

%% ============================================================
%% LOOP SIMULACIÓN
%% ============================================================

for k = 2:N_steps
    
    % Tiempo actual
    t_k = t(k);
    
    % === CONTROL: Velocidad comandada (sinusoidal) ===
    % Ciclo: bajada (v>0, masa expande) / subida (v<0, masa retrae)
    phi_k = mod(omega_rot * t_k, 2*pi);
    v_slide_cmd(k) = A_slide * sin(phi_k);
    
    % === FUERZAS ===
    
    % 1. Fuerza centrífuga (tira hacia fuera)
    F_centrifuga(k) = m_bola * omega_rot^2 * r(k-1);
    
    % 2. Fricción (resiste deslizamiento)
    % F_friction = mu_k * N_normal, donde N = peso efectivo en referencial rotatorio
    N_normal = m_bola * g;  % normal al carril (simplificado)
    F_friccion_f(k) = mu_k * N_normal;
    if v_r(k-1) > 0
        F_friccion_f(k) = -F_friccion_f(k);  % resiste expansión
    else
        F_friccion_f(k) = +F_friccion_f(k);  % resiste contracción
    end
    
    % 3. Amortiguamiento (viscoso)
    F_amort = -c_amort * v_r(k-1);
    
    % 4. Fuerza de control (servomotor que intenta mantener v_slide_cmd)
    % En realidad: servo intenta dar esa velocidad, pero hay resistencias
    % Simplificamos: si hay error, servo añade fuerza proporcional
    K_servo = 10000;  % ganancia servo [N/(m/s)]
    F_servo = K_servo * (v_slide_cmd(k) - v_r(k-1));
    
    % Fuerza neta
    F_neta(k) = F_centrifuga(k) + F_friccion_f(k) + F_amort + F_servo;
    
    % === ECUACIÓN DE MOVIMIENTO ===
    a_r(k) = F_neta(k) / m_bola;
    
    % Integración (Euler)
    v_r(k) = v_r(k-1) + a_r(k) * dt;

Com s’utilitza el “ninot de neu” per explicar girs i òrbites no sincròniques?
Metàfora: Un “ninot de neu” (composat per diverses esferes unides verticalment) té la seva pròpia inèrcia per la manera com les masses estan distribuïdes. Si rep una empenta lateral, la bola grossa (més pes) tendeix a frenar o accelerar el gir, o pot fins i tot fer que l’eix s’inclini o canviï de pla de rotació.
No sincronia: Quan el “ninot” gira però rep una força de la gravetat no alineada amb el seu eix (o hi ha una asimetria), acaba precessionant o entrant en ressonància, i la seva òrbita real no coincideix amb la prevista (es “canvia d’òrbita”). Això s’assembla a la precessió d’un giroscopi o d’una baldufa.
Aplicat a turbines: Pots tenir masses dins de pales que es mouen en un patró determinat, però la gravetat i les forces dinàmiques poden fer variar el centre de màsses efectiu al llarg del cicle. Això fa que, encara que el motor doni un cert rang angular, la resposta del sistema pot “canviar d’òrbita” segons l’estat dinàmic.
