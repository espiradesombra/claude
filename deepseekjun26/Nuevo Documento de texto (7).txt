import math

def pi_real(x):
    """Recompte exacte de prims ≤ x (criba d'Eratòstenes)"""
    if x < 2:
        return 0
    criba = [True] * (x + 1)
    criba[0] = criba[1] = False
    for i in range(2, int(x**0.5) + 1):
        if criba[i]:
            for j in range(i*i, x+1, i):
                criba[j] = False
    return sum(criba)

def goldbach_demo(n):
    """Verifica que existeix almenys un parell pp per a 2n"""
    if n <= 2:
        return True
    for p in range(2, n+1):
        if pi_real(p) - pi_real(p-1) == 1:  # és primer
            q = 2*n - p
            if q > 1 and pi_real(q) - pi_real(q-1) == 1:
                return True
    return False

# Exemple
for n in range(2, 101):
    assert goldbach_demo(n), f"Goldbach falla per a n={n}"
print("✅ Goldbach verificat per a n ≤ 100")
