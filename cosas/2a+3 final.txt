decimal.getcontext().prec = 80  # Precisión para detectar la curvatura de la onda

def obtener_fase(m, K):
    """Calcula la parte decimal de v (la onda de fase)."""
    # K = (2m+3)(2v+3) -> v = (K/(2m+3) - 3) / 2
    m_dec = Decimal(m)
    K_dec = Decimal(K)
    v = (K_dec / (Decimal(2) * m_dec + Decimal(3)) - Decimal(3)) / Decimal(2)
    return v - v.to_integral_value(rounding=decimal.ROUND_FLOOR)

def es_candidato_valido(n):
    """Filtro de red: n no mod 3 y no mod 5 (Estructura aY +- b)."""
    return n % 3 != 0 and n % 5 != 0

def factorizacion_por_fusion(K):
    print(f"Iniciando análisis físico de K = {K}")

    m = Decimal(1)
    paso_base = 1
    
    while (2 * m + 3)**2 <= K:
        if not es_candidato_valido(int(m)):
            m += paso_base
            continue

        # --- BLOQUE DE 8 MEDICIONES (4 por dirección en modo pinza) ---
        mediciones = []
        for i in range(8):
            curr_m = m + i
            fase = obtener_fase(curr_m, K)
            mediciones.append(fase)

        # --- ANÁLISIS CINEMÁTICO ---
        # v1: Velocidad (Primera diferencia)
        v1 = mediciones[1] - mediciones[0]
        # v2: Aceleración (Segunda diferencia)
        v2 = (mediciones[2] - mediciones[1]) - v1
        # v3: Variación Acelerativa (Jerk/Tirón)
        v3 = ((mediciones[3] - mediciones[2]) - (mediciones[2] - mediciones[1])) - v2

        # --- DETECCIÓN DE FACTOR (Fase toca 0 o 1) ---
        for i, f in enumerate(mediciones):
            if f < Decimal('1e-70') or f > Decimal('0.999999999999999999999999999999'):
                factor = int(2 * (m + i) + 3)
                if K % factor == 0:
                    return factor, K // factor

        # --- ESTRATEGIA DE SALTO (La Fusión) ---

        if abs(v3) < Decimal('1e-60'):

def preparar_rueda(nivel):
    """Precomputa las constantes de la rueda y el salto del primorial"""
    primos = [3, 5, 7, 11, 13, 17, 19, 23]
    primorial = math.prod(primos[:nivel])

    constante_fase = Decimal(0)
    for i in range(2, nivel + 2):
        suma_cuadrados = sum(k**2 for k in range(1, i + 1))
        constante_fase += Decimal(suma_cuadrados).sqrt()
        
    return primorial, constante_fase

    La función de onda: v = (K/(2m+3) - 3)/2
    Retornamos la parte decimal (el residuo de la fase)
    """
    v = (Decimal(K) / Decimal(p) - 3) / 2
    return v % 1

def ejecutar_pasos_constantes(K, v_n, primorial, constante):

    Implementación de la pinza de 8 pasos (simplificada)

    f1 = medir_fase(K, p0 + delta)

    velocidad = (f1 - f0) / delta
    return p0, f0, velocidad

from decimal import Decimal, getcontext

# Precisión de 80 dígitos para detectar la "onda"
getcontext().prec = 80

class MotorResonancia:
    def __init__(self, K, nivel_rueda=4):
        self.K = Decimal(K)
        self.primorial, self.constante_fase = self._precomputar_rueda(nivel_rueda)
        
    def _precomputar_rueda(self, nivel):
        import math
        primos = [3, 5, 7, 11, 13, 17]
        p_res = math.prod(primos[:nivel])
        # Diagonal del hipercubo para sintonizar fase
        c_fase = sum(Decimal(sum(k**2 for k in range(1, i+1))).sqrt() for i in range(2, nivel+2))
        return p_res, c_fase

    def obtener_fase(self, m_val):
        """Calcula la posición en el diente de sierra"""
        p = (Decimal(self.primorial) * Decimal(m_val) + self.constante_fase) / Decimal(2).sqrt()
        v = (self.K / p - 3) / 2
        return v % 1, p

    def analizar_vector_salto(self, m_actual):
        """
        ANALIZADOR DE VECTORES (La Pinza)
        Mide 4 puntos para calcular velocidad y aceleración
        """
        puntos = []
        paso_micro = Decimal('0.0000000001') # Discretización fina para la onda
        
        for i in range(4):
            fase, p_real = self.obtener_fase(m_actual + i * paso_micro)
            puntos.append(fase)
            
        # Cálculo de Vectores (Derivadas de la onda)
        v1 = puntos[1] - puntos[0] # Velocidad inicial
        v2 = puntos[2] - puntos[1] # Velocidad siguiente
        aceleracion = v2 - v1      # Curvatura de la parábola
        
        return puntos[0], v1, aceleracion, p_real

    def ejecutar_busqueda(self):
        m = Decimal(1)
        print(f"[*] Iniciando navegación vectorial sobre K={self.K}")
        
        while True:
            fase, vel, acel, p_actual = self.analizar_vector_salto(m)
            
            # LÓGICA DE SALTO (El Espejo)
            # Si la aceleración es estable, calculamos la distancia al cero:
            # d = -v / a (aproximación lineal de la convergencia)
            if acel != 0:
                distancia_al_cero = abs(fase / vel) if vel != 0 else 1
                
                # Si estamos muy cerca del cero (Fusión)
                if fase < Decimal('1e-20') or fase > Decimal('0.99999999999999999999'):
                    factor = int(p_actual.to_integral_value())
                    if self.K % factor == 0:
                        return factor
                
                # SALTO DE ARMÓNICO: 
                # En lugar de m = m + 1, saltamos según la velocidad de la onda
                salto = max(1, int(distancia_al_cero * 0.5)) # Salto conservador al 50%
                m += salto
            else:
                m += 1

// Calcula la fase decimal en un punto p dado por la rueda
float128 obtener_fase(uint128_t K, uint128_t v_n, Rueda R) {
    // p = (Primorial * v_n + Constante) / sqrt(2)
    float128 p = (R.primorial * v_n + R.constante_fase) / sqrt(2.0Q);
    
    // v = (K/p - 3) / 2 -> Extraemos solo la parte decimal (la onda)
    float128 v = (K / p - 3.0Q) / 2.0Q;
    return v - floorq(v); // Retorna la fase entre 0 y 1
}

#include <iostream>
#include <gmp.h>
#include <gmpxx.h>
#include <cmath>

// Configuración de precisión (mínimo 256 bits para ~80 dígitos)
const int PRECISION_BITS = 512;

struct VectorDirector {
    mpf_class velocidad;
    mpf_class aceleracion;
    mpf_class fase_centro;
};

// 1. EL SENSOR: Obtiene la fase decimal de la función v(m)
mpf_class obtener_fase(const mpz_class& K, const mpz_class& m) {
    mpf_class k_f(K, PRECISION_BITS);
    mpf_class divisor = mpf_class(2 * m + 3, PRECISION_BITS);
    
    mpf_class division = k_f / divisor;
    mpf_class entero = floor(division);
    return division - entero; // Retorna solo la parte decimal [0, 1)
}

// 2. LA PINZA: Mide 8 puntos para extraer la cinemática de la onda
VectorDirector disparar_pinza(const mpz_class& K, const mpz_class& m_central) {
    VectorDirector vd;
    mpz_class h = 1; // Paso infinitesimal de la pinza
    
    // Medimos puntos clave para derivadas
    mpf_class f_atras = obtener_fase(K, m_central - h);
    mpf_class f_centro = obtener_fase(K, m_central);
    mpf_class f_adelante = obtener_fase(K, m_central + h);
    
    // Cálculo de Vectores (Diferencias finitas)
    vd.velocidad = (f_adelante - f_atras) / 2.0;
    vd.aceleracion = (f_adelante - 2.0 * f_centro + f_atras);
    vd.fase_centro = f_centro;
    
    return vd;
}

// 3. NAVEGACIÓN: El choque de vectores que reduce el coste a log(n)
void ejecutar_fusion(mpz_class K, mpz_class m_inicio) {
    mpz_class m_actual = m_inicio;
    int iteracion = 0;
    
    std::cout << "Iniciando Navegación de Fase para K = " << K << std::endl;
    
    while (true) {
        iteracion++;
        VectorDirector pinza = disparar_pinza(K, m_actual);
        
        // Verificación de éxito (Fase colapsa a cero)
        if (pinza.fase_centro < 1e-50 || pinza.fase_centro > (1.0 - 1e-50)) {
            std::cout << "¡FUSIÓN COMPLETADA en iteración " << iteracion << "!" << std::endl;
            std::cout << "Factor hallado (2m+3): " << (2 * m_actual + 3) << std::endl;
            break;
        }

        // PREDICCIÓN DE ATERRIZAJE: El "salto" que rompe la raíz de n
        // Usamos la física de la parábola: m_salto = -v / a
        mpf_class salto_f = -pinza.velocidad / pinza.aceleracion;
        mpz_class salto_z(salto_f);
        
        std::cout << "Iteración " << iteracion << " | m: " << m_actual 
                  << " | Salto Vectorial: " << salto_z << std::endl;

        // Aplicamos el salto (con freno de seguridad)
        m_actual += salto_z;
        
        if (iteracion > 20) { // Salvaguarda
            std::cout << "Ajuste de precisión requerido o factor no hallado." << std::endl;
            break;
        }
    }
}
