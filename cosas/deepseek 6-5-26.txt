🧠 FIRMWARE – ESLABÓN INTELIGENTE 33x1 (Prototipo 1:10)
cpp
/*
  FIRMWARE: Eslabón Inteligente 33x1 (Zero Hit)
  Hardware: nRF52840 (o Arduino Opta con adaptador)
  Sensores: IMU BMI270, termistor NTC, sensor de presión (opcional)
  Comms: Radio 2.4 GHz (protocolo simple por paquetes)
  Autor: Mati (basado en diseño de Víctor Manzanares)
  Licencia: Hipocrática (prohibido uso militar)
*/

#include <Arduino.h>
#include <SPI.h>
#include <Wire.h>
#include <RF24.h>          // Radio de bajo coste (nRF24L01 compatible)
#include <BMI270.h>        // Librería del acelerómetro/giroscopio

// ---------- PINES (PCB escala 1:10) ----------
#define PIN_LED       13
#define PIN_TERMISTOR A0
#define PIN_PRESION   A1
#define PIN_ANTENA    10    // control opcional de antena externa

// ---------- OBJETOS ----------
RF24 radio(9, 8);          // CE, CSN
BMI270 imu;
uint8_t direccion_mac[6] = {0x33, 0x31, 0x00, 0x01, 0x00, 0x01}; // ID único

// ---------- VARIABLES DE CONTROL ----------
float theta = 0.0;         // fase angular simulada (para Kuramoto)
float omega = 0.0;         // velocidad angular del eslabón (del IMU)
float dJdt = 0.0;          // variación del momento de inercia (de la tensión)
float valley = 0.0;        // detector de caída de frecuencia de red
unsigned long last_tx = 0;
const unsigned long TX_INTERVAL = 50; // 20 Hz (Zero Hit)

// ---------- KURAMOTO ADAPTATIVO (de gemelo_v941.py) ----------
float K_kur = 0.10;
float K0 = 0.10, K1 = 0.05, K2 = 0.01;
float omega_ref = 314.16;   // 50 Hz -> 314.16 rad/s (para red eléctrica)
float valley_filt = 0.0;
const float TAU_VAL = 0.1;   // filtro para valley

// ---------- LÍMITE FÍSICO (Víctor) ----------
float r_q = 0.05;          // posición radial de la masa (metros, escala 1:10)
float r_hub = 0.05;        // mínimo (5 cm)
float r_tip = 0.63;        // máximo (63 cm)
float dr_q = 0.0;
const float DR_MAX = 0.5;   // m/s, limitador simétrico

// ---------- SENSORES ----------
float leer_temperatura() {
  int raw = analogRead(PIN_TERMISTOR);
  float volt = raw * 3.3 / 1023.0;
  float R_ntc = 10000.0 * (3.3 - volt) / volt;  // supone divisor 10k
  float T = 1.0 / (1.0/298.15 + log(R_ntc/10000.0)/3950.0) - 273.15;
  return T;   // Celsius
}

float leer_presion() {
  int raw = analogRead(PIN_PRESION);
  return raw * 20.0 / 1023.0;   // 0-20 bar (0-5V con divisor)
}

void leer_imu() {
  imu.update();
  omega = imu.getGyroZ();       // rad/s (aprox)
  // También se podría extraer theta integrando, pero lo recibimos por radio
}

// ---------- CÁLCULO DE LA TENSIÓN MECÁNICA (estimada) ----------
float calcular_tension() {
  // En un prototipo real: galga extensométrica en el mástil.
  // Aquí simulamos con la aceleración y la masa.
  float ax = imu.getAccelX();
  float m_eslabon = 0.918;   // kg (escala 1:10, aprox)
  float F = m_eslabon * ax;
  float area = 1e-4;          // m² (supuesto)
  return F / area;            // Pascal
}

// ---------- ECUACIÓN DE MOVIMIENTO DEL FLUIDO (simulada localmente) ----------
void actualizar_actuador() {
  // r_opt suave según posición angular (simulado o recibido)
  float r_opt = r_tip - (r_tip - r_hub) * (1.0 + cos(theta)) / 2.0;
  
  float F_c = 0.918 * omega * omega * r_q;   // m·ω²·r
  float F_ctrl = -K_kur * (r_q - r_opt);
  float F_fric = -80.0 * dr_q;               // rozamiento viscoso ficticio
  float m_q = 0.918;
  
  float d2r = (F_c + F_ctrl + F_fric) / m_q;
  dr_q += d2r * (TX_INTERVAL / 1000.0);
  
  // LIMIT FÍSICO de Víctor: |dr_q| <= ω·r_q
  float lim = omega * r_q;
  if (lim < DR_MAX) lim = DR_MAX;   // seguridad
  dr_q = constrain(dr_q, -lim, lim);
  
  r_q += dr_q * (TX_INTERVAL / 1000.0);
  r_q = constrain(r_q, r_hub, r_tip);
  
  // dJ/dt estimado (para enviar al vecino)
  dJdt = 2.0 * m_q * r_q * dr_q;
}

// ---------- KURAMOTO ADAPTATIVO (recibido del vecino) ----------
float kuramoto_coupling(float delta_theta) {
  valley_filt = TAU_VAL * valley + (1 - TAU_VAL) * valley_filt;
  K_kur = K0 + K1 * max(0.0, (omega_ref - omega)/omega_ref) + K2 * abs(valley_filt);
  K_kur = constrain(K_kur, 0.0, 0.3);
  return K_kur * sin(delta_theta);
}

// ---------- RADIO: envío de datos al vecino y a la nave ----------
void enviar_datos() {
  struct Paquete {
    float theta;
    float omega;
    float tension;
    float temperatura;
    float presion;
    float dJdt;
    float valley;
    uint32_t timestamp;
  } pkt;
  
  pkt.theta = theta;
  pkt.omega = omega;
  pkt.tension = calcular_tension();
  pkt.temperatura = leer_temperatura();
  pkt.presion = leer_presion();
  pkt.dJdt = dJdt;
  pkt.valley = valley;
  pkt.timestamp = millis();
  
  bool ok = radio.write(&pkt, sizeof(pkt));
  if (ok) digitalWrite(PIN_LED, !digitalRead(PIN_LED));  // parpadeo rápido
}

void recibir_datos(float &theta_vecino, float &omega_vecino) {
  if (radio.available()) {
    struct Paquete pkt;
    radio.read(&pkt, sizeof(pkt));
    theta_vecino = pkt.theta;
    omega_vecino = pkt.omega;
    valley = pkt.valley;   // compartir la valley para sincronía global
  }
}

// ---------- CONFIGURACIÓN ----------
void setup() {
  pinMode(PIN_LED, OUTPUT);
  Serial.begin(115200);
  Wire.begin();
  
  // IMU
  if (!imu.begin()) {
    Serial.println("Error IMU");
    while(1);
  }
  imu.setAccelRange(ACCEL_RANGE_8G);
  imu.setGyroRange(GYRO_RANGE_500_DPS);
  
  // Radio
  radio.begin();
  radio.setChannel(33);                 // canal 33 (simbólico)
  radio.setPayloadSize(32);             // paquete pequeño
  radio.setPALevel(RF24_PA_LOW);        // bajo consumo, alcance 1 m
  radio.openWritingPipe(direccion_mac);
  radio.openReadingPipe(1, direccion_mac);
  radio.startListening();
  
  Serial.println("Eslabón 33x1 listo (Zero Hit)");
}

// ---------- BUCLE PRINCIPAL (20 Hz) ----------
void loop() {
  static unsigned long last = 0;
  if (millis() - last < TX_INTERVAL) return;
  last = millis();
  
  // Lectura de sensores
  leer_imu();
  float theta_vecino = 0.0, omega_vecino = 0.0;
  
  // Modo half-duplex: escucha breve antes de transmitir
  radio.startListening();
  delayMicroseconds(500);
  recibir_datos(theta_vecino, omega_vecino);
  
  // Actualizar fase propia (integración de omega)
  theta += omega * (TX_INTERVAL / 1000.0);
  if (theta > TWO_PI) theta -= TWO_PI;
  
  // Acoplamiento Kuramoto con el vecino
  float delta = theta_vecino - theta;
  float coupling = kuramoto_coupling(delta);
  omega += coupling * (TX_INTERVAL / 1000.0);
  
  // Actualizar actuador (mover fluido)
  actualizar_actuador();
  
  // Cambiar a modo transmisor
  radio.stopListening();
  enviar_datos();
  
  // Debug por serial (opcional)
  static int cnt=0;
  if (cnt++ % 40 == 0) { // cada 2 segundos
    Serial.print("T="); Serial.print(leer_temperatura());
    Serial.print(" P="); Serial.print(leer_presion());
    Serial.print(" ω="); Serial.print(omega);
    Serial.print(" r="); Serial.println(r_q);
  }
}
