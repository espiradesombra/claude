#   - Kp, Ki: Ganancias del controlador PI (ajustables)

# 1. Detección de SSR (3-6 Hz):
def detect_SSR(i_q, omega):
    # Filtrar la señal de i_q para detectar oscilaciones en 3-6 Hz
    # (Usar un filtro pasa-banda o FFT en tiempo real)
    ssr_detected = check_frequency(i_q, low=3, high=6)  # Devuelve True si hay SSR

# 2. Controlador PI para ajustar m_i:
def adaptive_inertia_control(i_q, omega, m_i, Kp=1.0, Ki=0.1):
    error = i_q - i_q_ref  # i_q_ref = 0 (objetivo: minimizar oscilaciones)
    integral = 0

    while ssr_detected:
        proportional = Kp * error
        integral += Ki * error * dt  # dt = tiempo de muestreo
        delta_m = proportional + integral

        # Ajustar m_i (límites: 0.1x a 4x la inercia nominal)
        m_i_new = m_i * (1 + delta_m)
        m_i_new = clamp(m_i_new, 0.1 * m_i_nominal, 4 * m_i_nominal)

        # Aplicar el cambio al molino i
        set_inertia(molino_i, m_i_new)
        return m_i_new

# 3. Integración con el sistema dq0:
def dq0_control_loop():
    for molino in molinos:
        i_d, i_q = get_dq_current(molino)  # Transformada dq0
        omega = get_grid_frequency()
        m_i = get_current_inertia(molino)

        if detect_SSR(i_q, omega):
            m_i = adaptive_inertia_control(i_q, omega, m_i)
            log(f"Molino {molino.id}: Ajustada inercia a {m_i} para absorber SSR")
