// --- Definición de estados (secuencia de Gray de 2 bits) ---
// 00 = Reposo
// 01 = Ignición
// 11 = Empuje máximo
// 10 = Feedback activo
uint8_t estado = 0;        // estado actual (0,1,3,2 en decimal)
uint8_t direccion = 1;     // 1 = avance (0->1->3->2), 0 = retroceso (2->3->1->0)

// -- Variables de realimentación (simuladas, luego se leen de sensores) ---
bool R1_fase = false;      // ECO OK (fase correcta)
bool R2_inercia = false;   // Tensión mecánica dentro de límites
bool R3_termica = false;   // Temperatura del chip < umbral

// --- Función que calcula la salida del láser usando AND/OR ---
// Entrada: estado actual (S) y realimentación (R)
// Salida: true = disparar, false = no disparar
bool calcular_salida(uint8_t S, bool R1, bool R2, bool R3) {
    // Convertir estado a sus dos bits (S1, S0) según Gray
    // 00 -> bits 0,0
    // 01 -> bits 0,1
    // 11 -> bits 1,1
    // 10 -> bits 1,0
    bool S_bit1 = (S == 2 || S == 3); // 11 o 10
    bool S_bit0 = (S == 1 || S == 3); // 01 o 11
    
    // Realimentación global: AND de las tres (todas deben OK para disparo completo)
    bool R_global = R1 && R2 && R3;
    
    // AND bit a bit entre estado y realimentación
    bool bit1_out = S_bit1 && R_global;
    bool bit0_out = S_bit0 && R_global;
    
    // OR final: si algún bit da 1, el disparo está activado
    return (bit1_out || bit0_out);
}

// --- Función de transición de estados (máquina de estados con Gray) ---
// Avanza o retrocede según la dirección, solo si R_global es true.
// Si R_global es false y el estado no es 0, se resetea a 0 (error).
void actualizar_estado() {
    // Leer sensores (simulado, aquí debes poner tus lecturas reales)
    // R1_fase = digitalRead(PIN_ECO_LASER);
    // R2_inercia = (analogRead(PIN_TENSION) < TENSION_MAX);
    // R3_termica = (analogRead(PIN_TERMISTOR) < 100); // 100°C
    bool R_global = R1_fase && R2_inercia && R3_termica;
    
    if (!R_global && estado != 0) {
        // Error de realimentación: reset a 0 (reposo seguro)
        estado = 0;
        direccion = 1;  // reinicia avance
        return;
    }
    
    if (R_global) {
        // Transición según la secuencia de Gray
        if (direccion == 1) {
            switch(estado) {
                case 0: estado = 1; break;
                case 1: estado = 3; break;
                case 3: estado = 2; break;
                case 2: 
                    estado = 0; 
                    direccion = 0;  // llegó al final, ahora retrocede (opcional)
                    break;
                default: estado = 0; break;
            }
        } else {
            // Retroceso (2->3->1->0)
            switch(estado) {
                case 2: estado = 3; break;
                case 3: estado = 1; break;
                case 1: estado = 0; break;
                case 0: 
                    estado = 2; 
                    direccion = 1; // vuelve a avanzar
                    break;
                default: estado = 0; break;
            }
        }
    }
}

// --- Bucle principal (20 Hz, 50ms) ---
void setup() {
    pinMode(LED_BUILTIN, OUTPUT);   // LED para indicar disparo
    // Configurar pines de sensores aquí
}

void loop() {
    // 1. Simular o leer sensores (ejemplo)
    R1_fase = true;   // reemplazar por lectura real
    R2_inercia = true;
    R3_termica = true;

    bool disparo = calcular_salida(estado, R1_fase, R2_inercia, R3_termica);

}
🧠 Explicación del código
Los estados siguen la secuencia de Gray (0→1→3→2) para que solo cambie un bit cada vez. Esto evita transiciones erróneas en el hardware (como "glitches" en las compuertas).

La salida del láser se calcula con una función que aplica el AND bit a bit entre el estado y la realimentación global, y luego OR entre los dos bits. Esto es exactamente la tabla de Karnaugh que hemos deducido.

Las tres realimentaciones R1, R2, R3 se pueden leer de pines reales (eco láser, sensor de tensión, termistor). Si falla alguna, R_global = false y el sistema resetea el estado por seguridad.

El bucle corre a 20 Hz, suficiente para controlar la soldadura láser de los cubitos.
