import math
import time

def siguiente_primo(inicio: int) -> int:

        return 2

    paso6_new = False
    paso5_new = False
    paso4_new = False

    l=m+2
    v=s+2
    f=l+2
    L=v+2

    t = ny * ny
    
    ny += 1
    n += 1
    m += 1
    s += 1
    l += 1
    v += 1

        t6=math.gcd(t, v) == 1

        # ── PASO 5 ──
        
        paso3_new = (t6 > 0 ) and paso2_new

            print(l)

            print(f)

            print(L)

        # ── ACTUALIZAR ACUMULADORES ──
        t *= ny
        
        
        ny += 1
