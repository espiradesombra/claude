<?php
header('Content-Type: application/json');
header("Access-Control-Allow-Origin: *");
header("Access-Control-Allow-Methods: POST");

$data = json_decode(file_get_contents('php://input'), true);
$n_str = isset($data['n']) ? trim($data['n']) : '';

if (empty($n_str) || !ctype_digit($n_str)) {
    echo json_encode(['status'=>'error','message'=>'Número inválido']);
    exit;
}

$n = (int)$n_str;
if ((string)$n !== $n_str) {
    echo json_encode(['status'=>'error','message'=>'Overflow 64-bit']);
    exit;
}

$start = microtime(true);

/* ================= CONFIG ================= */
$PRIMOS = [2,3,5,7,11,13,17,19,23];
$PRIMORIAL = array_product($PRIMOS);
$LIM_CRIBA = 2000000;
$MAX_BISEC = 6000;
$TALL_ASIM = 4;
$MICRO_SWEEP = 5000; // 👈 clave

/* ================= UTIL ================= */

function gcd($a,$b){while($b){$t=$b;$b=$a%$b;$a=$t;}return $a;}

function pasa_rueda($m,$primorial){
    return ($m>=1 && gcd(2*$m+3,$primorial)===1);
}

function proper_valid($m,$pas,$primorial){
    for($i=0;$i<200;$i++){
        if($m<1) return null;
        if(pasa_rueda($m,$primorial)) return $m;
        $m+=$pas;
    }
    return null;
}

/* ================= ΔΦ ================= */

function sgn_desfase($m,$N){
    $B=2*$m+3;
    if($B<=1||$B>=$N) return 0;

    $D_B=2*$B;
    $R_B=$N % $D_B;

    $A=intdiv($N,$B);
    if($A<3) return 0;

    $m_e=intdiv($A-3,2);
    if($m_e<1) return 0;

    $D_A=2*(2*$m_e+3);
    $R_A=$N % $D_A;

    $d=$R_B*$D_A - $R_A*$D_B;

    return $d>0?1:($d<0?-1:0);
}

function check_factor($m,$N){
    $D=2*$m+3;
    return ($D>1 && $D<$N && $N%$D===0);
}

/* ================= MICRO SWEEP ================= */

function micro_sweep($N,$m_center,$range){
    $start = max(1,$m_center-$range);
    $end   = $m_center+$range;

    for($m=$start;$m<=$end;$m++){
        if(check_factor($m,$N)) return 2*$m+3;
    }
    return null;
}

/* ================= BISECCIÓN ================= */

function biseccion_fase($N,$m_ini,$m_fi,$primorial){
    global $MAX_BISEC,$TALL_ASIM,$MICRO_SWEEP;

    for($it=0;$it<$MAX_BISEC;$it++){

        $rang=$m_fi-$m_ini;

        if($rang<100){
            return micro_sweep($N,intdiv($m_ini+$m_fi,2),200);
        }

        $m_25=proper_valid($m_ini+intdiv($rang,$TALL_ASIM),1,$primorial);
        $m_e =proper_valid($m_ini,1,$primorial);
        $m_d =proper_valid($m_fi,-1,$primorial);

        if(!$m_25||!$m_e||!$m_d) return null;

        foreach([$m_e,$m_25,$m_d] as $mp){
            if(check_factor($mp,$N)) return 2*$mp+3;
        }

        $s_e  = sgn_desfase($m_e,$N);
        $s_25 = sgn_desfase($m_25,$N);
        $s_d  = sgn_desfase($m_d,$N);

        if($s_e && $s_25 && $s_e!=$s_25){
            $m_fi=$m_25;
        }
        elseif($s_25 && $s_d && $s_25!=$s_d){
            $m_ini=$m_25;
        }
        else{
            // 👇 aquí está la mejora real
            $m_mid=intdiv($m_ini+$m_fi,2);
            return micro_sweep($N,$m_mid,$MICRO_SWEEP);
        }

        if((microtime(true)-$_SERVER['REQUEST_TIME_FLOAT'])>4.8){
            return "timeout";
        }
    }

    return null;
}

/* ================= CORE ================= */

function factor_mdc_hibrido($N){
    global $PRIMOS,$PRIMORIAL,$LIM_CRIBA;

    foreach($PRIMOS as $p){
        if($N%$p===0 && $p<$N) return $p;
    }

    $r=(int)sqrt($N);
    if($r*$r===$N) return $r;

    $m_max=intdiv($r-3,2);
    $lim=min($m_max,$LIM_CRIBA);

    for($m=1;$m<=$lim;$m++){
        if(pasa_rueda($m,$PRIMORIAL)){
            $D=2*$m+3;
            if($N%$D===0) return $D;
        }
    }

    if($lim>=$m_max) return null;

    return biseccion_fase($N,$lim+1,$m_max,$PRIMORIAL);
}

/* ================= RUN ================= */

$res=factor_mdc_hibrido($n);

if($res==="timeout"){
    echo json_encode(['status'=>'limit']);
}
elseif($res){
    echo json_encode([
        'status'=>'success',
        'p'=>(string)$res,
        'q'=>(string)intdiv($n,$res),
        'method'=>'MDC_FASE+MICRO'
    ]);
}
else{
    echo json_encode([
        'status'=>'fail',
        'message'=>'No detectado (ni fase ni micro-sweep)'
    ]);
}
?>
