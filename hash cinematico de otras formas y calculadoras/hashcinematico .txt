i += x (salta los x ceros).


Procesa B[j] (similar, pero en sentido inverso).
Choque: Si i >= j, registra (i, j, tiempo) en choques.
Incrementa tiempo += 1.

import hashlib

def hash_cinematico(input_bytes):
    # Convierte el input a bits
    bits = ''.join(f'{byte:08b}' for byte in input_bytes)
    n = len(bits)

    # Parámetros
    v = 1.0
    i, j = 0, n - 1
    tiempo = 0
    choques = []

    while i <= j:
        # Procesa bit en i
        if bits[i] == '1':
            v += 1
            i += 1
        else:
            # Cuenta ceros consecutivos
            x = 0
            while i + x < n and bits[i + x] == '0':
                x += 1
            # Aplica regla: v = v * (2/3)^x * 2^x
            v *= (2/3)**x * (2**x)
            i += x

        # Procesa bit en j (similar)
        if j >= i and bits[j] == '1':
            v += 1
            j -= 1
        elif j >= i:
            x = 0
            while j - x >= 0 and bits[j - x] == '0':
                x += 1
            v *= (2/3)**x * (2**x)
            j -= x

        tiempo += 1

        # Choque
        if i > j:
            choques.append((i, j, tiempo))
            break

    # Combina todo en un string
    estado = f"{v}_{i}_{j}_{tiempo}_{choques}"
    # Usa SHA-256 para asegurar uniformidad
    return hashlib.sha256(estado.encode()).hexdigest()

# Ejemplo de uso
print(hash_cinematico(b"hello"))  # Output: ej: "a1b2c3..."

import hashlib
from typing import List, Tuple, Dict, Any

def hash_cinematico_parametrizable(
    input_bits: str,
    pasadas: int = 1,
    factor_1: float = 2.0,
    factor_0: float = 2/3,
    modo_0_consecutivos: str = "primer_0",  # "primer_0", "todos", "ninguno"
    inicial_v: float = 1.0
) -> Dict[str, Any]:
    """
    Hash cinemático parametrizable con X pasadas y dos punteros.

    Args:
        input_bits: Cadena de bits (ej: "10011")
        pasadas: Número de pasadas
        factor_1: Multiplicador para bit '1'
        factor_0: Multiplicador para bit '0'
        modo_0_consecutivos: Cómo manejar ceros consecutivos
        inicial_v: Velocidad inicial

    Returns:
        Diccionario con todos los datos de la simulación
    """
    resultados = []

    for pasada in range(pasadas):
        v = inicial_v
        i, j = 0, len(input_bits) - 1
        t = 0
        choques = []
        tiempos_A = []
        tiempos_B = []
        ultimo_bit_A = None
        ultimo_bit_B = None

        while i <= j:
            t += 1

            # Turno A
            if i <= j:
                bit = input_bits[i]
                # Aplicar regla para bit
                if bit == '1':
                    v *= factor_1
                else:  # bit == '0'
                    if modo_0_consecutivos == "primer_0":
                        if ultimo_bit_A != '0':
                            v *= factor_0
                    elif modo_0_consecutivos == "todos":
                        v *= factor_0
                    # "ninguno" no hace nada
                tiempos_A.append(t)
                i += 1
                ultimo_bit_A = bit

            # Turno B
            if i <= j:
                bit = input_bits[j]
                # Aplicar regla para bit
                if bit == '1':
                    v *= factor_1
                else:  # bit == '0'
                    if modo_0_consecutivos == "primer_0":
                        if ultimo_bit_B != '0':
                            v *= factor_0
                    elif modo_0_consecutivos == "todos":
                        v *= factor_0
                tiempos_B.append(t)
                j -= 1
                ultimo_bit_B = bit

            # Check choque
            if i > j:
                choques.append((t, i-1, j+1))

        resultados.append({
            "v_final": v,
            "t_final": t,
            "choques": choques,
            "tiempos_A": tiempos_A,
            "tiempos_B": tiempos_B,
            "bits_procesados_A": input_bits[:len(tiempos_A)],
            "bits_procesados_B": input_bits[len(input_bits)-len(tiempos_B):][::-1]
        })

        # Para la siguiente pasada, reiniciamos v a inicial_v
        # (Opcional: podríamos usar v como inicial_v para la siguiente pasada)

    # Generar hash final
    hash_data = {
        "input": input_bits,
        "pasadas": pasadas,
        "params": {
            "factor_1": factor_1,
            "factor_0": factor_0,
            "modo_0_consecutivos": modo_0_consecutivos,
            "inicial_v": inicial_v
        },
        "resultados": resultados
    }

    # Convertir a string y hashear
    hash_str = str(hash_data)
    hash_final = hashlib.sha256(hash_str.encode()).hexdigest()

    return {
        "hash": hash_final,
        "datos": hash_data
    }

from fractions import Fraction
import json
import hashlib

def hash_cinematico_final(
    input_bits: str,
    pasadas: int = 1,
    factor_1: str = "2/1",
    factor_0: str = "2/3",
    modo_0_consecutivos: str = "primer_0",
    inicial_v: str = "1/1"
):
    # Parsear factores
    def parse_frac(s):
        if isinstance(s, str) and '/' in s:
            n, d = s.split('/')
            return Fraction(int(n), int(d))
        return Fraction(s)

    factor_1 = parse_frac(factor_1)
    factor_0 = parse_frac(factor_0)
    v_inicial = parse_frac(inicial_v)

    resultados = []

    for _ in range(pasadas):
        v = v_inicial
        i, j = 0, len(input_bits) - 1
        t = 0
        choques = []
        tiempos_A = []
        tiempos_B = []
        ultimo_A = None
        ultimo_B = None

        while i <= j:
            t += 1

            # Turno A
            if i <= j:
                bit = input_bits[i]
                if bit == '1':
                    v *= factor_1
                else:
                    if modo_0_consecutivos == "primer_0" and ultimo_A != '0':
                        v *= factor_0
                    elif modo_0_consecutivos == "todos":
                        v *= factor_0
                tiempos_A.append(t)
                i += 1
                ultimo_A = bit

            # Turno B
            if i <= j:
                bit = input_bits[j]
                if bit == '1':
                    v *= factor_1
                else:
                    if modo_0_consecutivos == "primer_0" and ultimo_B != '0':
                        v *= factor_0
                    elif modo_0_consecutivos == "todos":
                        v *= factor_0
                tiempos_B.append(t)
                j -= 1
                ultimo_B = bit

            if i > j:
                choques.append((t, i-1, j+1))

        resultados.append({
            "v_final": str(v),
            "t_final": t,
            "choques": [(int(t), int(i), int(j)) for (t, i, j) in choques],
            "tiempos_A": tiempos_A,
            "tiempos_B": tiempos_B
        })

    # Crear objeto para hashear
    hash_obj = {
        "input": input_bits,
        "pasadas": pasadas,
        "params": {
            "factor_1": str(factor_1),
            "factor_0": str(factor_0),
            "modo_0_consecutivos": modo_0_consecutivos,
            "inicial_v": str(v_inicial)
        },
        "resultados": resultados
    }

    # Serializar a JSON (determinista con sort_keys=True)
    hash_str = json.dumps(hash_obj, sort_keys=True, separators=(',', ':'))
    hash_final = hashlib.sha256(hash_str.encode()).hexdigest()

    return {
        "hash": hash_final,
        "datos": hash_obj
    }

1️⃣ Factores de Aceleración/Deceleración

1: v *= factor_1 (ej: factor_1 = 2 → v *= 2).
0: v *= factor_0 (ej: factor_0 = 2/3 → v *= 2/3).

0_acelera_1_decelera:
0: v *= factor_0 (acelera si factor_0 > 1).
1: v *= factor_1 (decelera si factor_1 < 1).

ambos_aceleran:
1: v *= factor_1.
0: v *= factor_0 (ambos >1).

ambos_deceleran:
1: v *= factor_1 (ambos <1).
0: v *= factor_0.

2️⃣ Manejo de 0s Consecutivos

  
    
      Modo
      Comportamiento
      Ejemplo (input="000", factor_0=2/3)
    
  
  
    
      "primer_0"
      Solo el primer 0 en una secuencia aplica factor_0. Los siguientes no modifican v.
      v *= 2/3 (solo una vez).
    
    
      "todos"
      Todos los 0s aplican factor_0.
      v *= (2/3)^3.
    
    
      "ninguno"
      Los 0s no afectan a v.
      v no cambia.
    
    
      "acumular"
      factor_0 se multiplica por el número de 0s consecutivos.
      v *= (2/3) * 3 = 2.
    
  




3️⃣ Punteros y Colisiones
Puntero A: Avanza de izquierda a derecha (índice i).
Puntero B: Avanza de derecha a izquierda (índice j).
Turnos: Alternados (A en pasos impares, B en pares).
Colisión: Cuando i > j. Se registra:Tiempo de colisión (t).

import json
import hashlib
from fractions import Fraction
from typing import Dict, List, Tuple, Any

def hash_cinematico(
    input_bits: str,
    pasadas: int = 1,
    factor_1: str = "2",
    factor_0: str = "2/3",
    modo_0: str = "primer_0",
    modo_operacion: str = "1_acelera_0_decelera",
    inicial_v: str = "1"
) -> Dict[str, Any]:
    # --- Parsear parámetros a Fraction (para precisión exacta) ---
    def parse_factor(s: str) -> Fraction:
        if isinstance(s, str) and '/' in s:
            num, den = map(int, s.split('/'))
            return Fraction(num, den)
        return Fraction(int(float(s)), 1) if '.' in s else Fraction(int(s), 1)

    factor_1 = parse_factor(factor_1)
    factor_0 = parse_factor(factor_0)
    v_inicial = parse_factor(inicial_v)

    # --- Simulación ---
    resultados = []
    for _ in range(pasadas):
        v = v_inicial
        i, j = 0, len(input_bits) - 1
        t = 0
        choques = []
        tiempos_A = []
        tiempos_B = []
        bits_A = []
        bits_B = []
        ultimo_A = None
        ultimo_B = None

        while i <= j:
            t += 1
            # Turno A (izquierda a derecha)
            if i <= j:
                bit = input_bits[i]
                bits_A.append(bit)
                # Aplicar regla según modo_operacion
                if modo_operacion == "1_acelera_0_decelera":
                    if bit == '1':
                        v *= factor_1
                    else:
                        if modo_0 == "primer_0" and ultimo_A != '0':
                            v *= factor_0
                        elif modo_0 == "todos":
                            v *= factor_0
                        # "ninguno" o "acumular" no hacen nada aquí
                elif modo_operacion == "0_acelera_1_decelera":
                    if bit == '0':
                        v *= factor_0
                    else:
                        if modo_0 == "primer_0" and ultimo_A != '1':
                            v *= factor_1
                        elif modo_0 == "todos":
                            v *= factor_1
                elif modo_operacion == "ambos_aceleran":
                    v *= factor_1 if bit == '1' else factor_0
                elif modo_operacion == "ambos_deceleran":
                    v *= factor_1 if bit == '1' else factor_0
                tiempos_A.append(t)
                i += 1
                ultimo_A = bit

            # Turno B (derecha a izquierda)
            if i <= j:
                bit = input_bits[j]
                bits_B.append(bit)
                if modo_operacion == "1_acelera_0_decelera":
                    if bit == '1':
                        v *= factor_1
                    else:
                        if modo_0 == "primer_0" and ultimo_B != '0':
                            v *= factor_0
                        elif modo_0 == "todos":
                            v *= factor_0
                elif modo_operacion == "0_acelera_1_decelera":
                    if bit == '0':
                        v *= factor_0
                    else:
                        if modo_0 == "primer_0" and ultimo_B != '1':
                            v *= factor_1
                        elif modo_0 == "todos":
                            v *= factor_1
                elif modo_operacion == "ambos_aceleran":
                    v *= factor_1 if bit == '1' else factor_0
                elif modo_operacion == "ambos_deceleran":
                    v *= factor_1 if bit == '1' else factor_0
                tiempos_B.append(t)
                j -= 1
                ultimo_B = bit

            # Registrar choque
            if i > j:
                choques.append((t, i - 1, j + 1))

        resultados.append({
            "v_final": str(v),
            "t_final": t,
            "choques": [[int(x) for x in choque] for choque in choques],
            "tiempos_A": tiempos_A,
            "tiempos_B": tiempos_B,
            "bits_A": ''.join(bits_A),
            "bits_B": ''.join(bits_B[::-1])  # Invertir para mantener orden original
        })

    # --- Generar hash determinista ---
    hash_data = {
        "input": input_bits,
        "pasadas": pasadas,
        "params": {
            "factor_0": str(factor_0),
            "factor_1": str(factor_1),
            "modo_0": modo_0,
            "modo_operacion": modo_operacion,
            "inicial_v": str(v_inicial)
        },
        "resultados": resultados
    }
    # Serializar a JSON (determinista: orden de claves fijo)
    hash_str = json.dumps(hash_data, sort_keys=True, separators=(',', ':'))
    hash_final = hashlib.sha256(hash_str.encode()).hexdigest()

    return {
        "hash": hash_final,
        "datos": hash_data
    }

# --- Ejemplo de uso ---
result = hash_cinematico(
    input_bits="10011",
    pasadas=2,
    factor_1="2",
    factor_0="2/3",
    modo_0="primer_0",
    modo_operacion="1_acelera_0_decelera"
)
print("Hash:", result["hash"])
print("Datos:", json.dumps(result["datos"], indent=2))

    }

    // 2. Multiplicar por posición
    if (params.use_position) {
        const pos = (bit === '1') ? i : j;
        factor *= (1 + pos * 0.01);
    }

    // 3. Aplicar XOR
    if (params.use_xor) {
        const vInt = Math.floor(v * 1e6);
        const posInt = i * 1000 + j;
        v = (vInt ^ posInt) / 1e6;
    }

    // 4. Aplicar factor
    v *= factor;

    // 5. Aplicar salt
    if (params.use_salt) {
        const salt = inputBits.split('').reduce((sum, b, idx) => 
            sum + (b === '1' ? idx : 0), 0);
        v *= (1 + salt * 0.001);
    }

    return v;
}

// --- Simulación (SIN mod intermedio) ---
function simular(input, params) {
  let v = params.inicial_v; // Dejar crecer libremente (ej: 1e+100)
  let i = 0, j = input.length - 1;
  let t = 0.0;
  const choques = [];
  const tiempos_A = [];
  const tiempos_B = [];

  while (i <= j) {
    t += 1.0;
    // Procesar A y B (sin mod, solo multiplicar)
    if (i <= j) {
      const bit = input[i];
      v *= (bit === '1') ? params.factor_1 : params.factor_0;
      tiempos_A.push(t);
      i++;
    }
    if (i <= j) {
      const bit = input[j];
      v *= (bit === '1') ? params.factor_1 : params.factor_0;
      tiempos_B.push(t);
      j--;
    }
    if (i > j) choques.push([t, i-1, j+1]);
  }

  return { v, t, choques, tiempos_A, tiempos_B };
}

// --- Generar Hash Final (con mod) ---
function generarHash(resultados, params, bits = 64) {
  // 1. Convertir TODO a formato científico (máxima precisión)
  const dataStr = resultados.map(res => {
    return [
      res.v.toExponential(),       // "1.23e+100"

      res.choques.map(c => c.map(x => x.toExponential()).join(',')).join(';'),
      res.tiempos_A.map(t => t.toExponential()).join(','),
      res.tiempos_B.map(t => t.toExponential()).join(',')
    ].join('|');
  }).join('||');

  const fullStr = `${params.input}|${params.factor_1}|${params.factor_0}|${dataStr}`;

  // 3. Convertir a BigInt (para manejar números grandes)
  const hashInt = stringToBigInt(fullStr); // Función aux para convertir string a BigInt

  // 4. Aplicar mod 2^bits
  const mod = 2n ** BigInt(bits);

  return hashFinal.toString(16).padStart(bits / 4, '0');

// --- Función auxiliar: string a BigInt ---
function stringToBigInt(str) {
  let result = 0n;
  for (let i = 0; i < str.length; i++) {
    result = (result * 256n) + BigInt(str.charCodeAt(i));
  }
  return result;
}
