def es_primo_con_restos(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False

    # Calculamos restos hasta (n-1)//2
    restos = []
    for i in range(2, (n - 1) // 2 + 1):
        restos.append(n % i)

    # Primera secuencia: debería ser creciente (pendiente ~1)
    primera_secuencia = restos[:len(restos)//2]
    es_creciente = all(primera_secuencia[i] >= primera_secuencia[i-1] for i in range(1, len(primera_secuencia)))

    # Segunda secuencia: debería ser decreciente (pendiente ~-1)
    segunda_secuencia = restos[len(restos)//2:]
    es_decreciente = all(segunda_secuencia[i] <= segunda_secuencia[i-1] for i in range(1, len(segunda_secuencia)))

    # Si cumple ambas secuencias, es primo
    return es_creciente and es_decreciente
