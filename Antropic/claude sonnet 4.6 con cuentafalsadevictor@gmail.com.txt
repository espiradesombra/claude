Ablació
Procés pel qual un material s'elimina per escalfament intens. En el Projecte 33×1, el làser escalfa el gel de l'asteroide fins que aquest sublima, creant un xorro que empeny l'asteroide.

Recurs clau del projecte. L'aigua es pot usar per a sublimació o com a combustible (H2/O2) per a coets futurs.

Bennu

Innovació clau. 3–5 braços que giren en sentit contrari a la rotació de l'asteroide, permetent una sublimació contínua en direcció de contraober.

C
C-type (asteroide carbonaci)
Tipus d'asteroide ric en carboni i gel/argiles. L'objectiu principal del projecte.

Contraober
Concepte central. Disparar el làser sobre el cap que apunta cap endavant. El gas surt cap enrere, frenant l'asteroide (empenta retrògrada).

Δ (Delta)
ΔV (Delta-V)

E

Espiral orbital

Blocs de gel portats des de la Terra (100 tones) per compensar la falta de gel natural suficient en asteroides petits per mantenir la frenada anys.

"""
GEMELO 1: DIRECT AIR CAPTURE (DAC) FROM MOTOR EXHAUST

Simula:
1. Flujo escape motor (gasoil 3×O2)
2. Composición: CO2 + H2O + NOx + residuos
3. Sistema zeolita + calor residual (120°C)
4. Captura eficiencia: 85-92%
5. Presurización: 150 bar
6. Depósito secundario: dinámico
"""

import numpy as np
import matplotlib.pyplot as plt
from dataclasses import dataclass
from typing import Tuple

@dataclass
class MotorEscape:
    """Escape motor gasoil 3×O2 por ciclo"""
    gasoil_litros: float = 1.0
    tiempo_ciclo_s: float = 0.5
    
    # Combustión estequiométrica
    # C12H23 + 18 O2 → 12 CO2 + 11.5 H2O
    co2_kg_por_litro: float = 3.15
    h2o_kg_por_litro: float = 1.20
    nox_ppm: float = 200  # partes por millón
    co_ppm: float = 50
    hc_ppm: float = 30
    
    def flujo_horario(self) -> dict:
        """Extrapolación a consumo horario"""
        ciclos_por_hora = 3600 / self.tiempo_ciclo_s
        litros_hora = self.gasoil_litros * ciclos_por_hora
        
        return {
            'litros_h': litros_hora,
            'co2_kg_h': self.co2_kg_por_litro * litros_hora,
            'h2o_kg_h': self.h2o_kg_por_litro * litros_hora,
            'nox_kg_h': self.nox_ppm * litros_hora * 1e-6 * 46 / 1000,
            'co_kg_h': self.co_ppm * litros_hora * 1e-6 * 28 / 1000,
            'hc_kg_h': self.hc_ppm * litros_hora * 1e-6 * 14 / 1000,
        }

@dataclass
class SistemaDAC:
    """Sistema Direct Air Capture con zeolita"""
    material: str = "zeolita_13X"  # MOF alternativa
    masa_kg: float = 5.0  # Masa de adsorbente
    capacidad_absorcion: float = 0.25  # kg CO2 / kg zeolita
    temp_activacion_c: float = 120  # Usa calor residual motor
    presion_bar: float = 150  # Compresión
    eficiencia_captura: float = 0.88  # 88% del CO2
    
    # Energética
    calor_residual_disponible_kw: float = 4.61  # Del motor
    energia_dac_kw: float = 0.35  # Para el proceso DAC
    
    def capacidad_total_co2(self) -> float:
        """Máxima capacidad de captura por ciclo"""
        return self.masa_kg * self.capacidad_absorcion
    
    def co2_capturado(self, co2_entrada_kg: float) -> float:
        """CO2 realmente capturado"""
        return co2_entrada_kg * self.eficiencia_captura
    
    def energia_neta(self) -> float:
        """Energía neta disponible tras DAC"""
        return self.calor_residual_disponible_kw - self.energia_dac_kw

class DepositoSecundarioCO2:
    """Depósito de CO2 presurizado (150 bar)"""
    def __init__(self, volumen_litros: float = 10.0):
        self.volumen_l = volumen_litros
        self.presion_bar = 150
        
        # Densidad CO2 a 150 bar, 20°C ≈ 500 kg/m³
        self.densidad_co2_kg_m3 = 500
        self.masa_max_co2 = (volumen_litros / 1000) * self.densidad_co2_kg_m3
        self.masa_actual_kg = 0
    
    def llenar(self, co2_kg: float) -> Tuple[float, float]:
        """Intenta llenar depósito"""
        espacio_disponible = self.masa_max_co2 - self.masa_actual_kg
        co2_almacenado = min(co2_kg, espacio_disponible)
        self.masa_actual_kg += co2_almacenado
        co2_exceso = co2_kg - co2_almacenado
        
        return co2_almacenado, co2_exceso
    
    def porcentaje_lleno(self) -> float:
        return (self.masa_actual_kg / self.masa_max_co2) * 100

# SIMULACIÓN
def simular_dac_ciclo_completo(duracion_horas: float = 8.0):
    """Simula 8 horas de conducción con captura DAC"""
    
    motor = MotorEscape(gasoil_litros=1.0, tiempo_ciclo_s=0.5)
    dac = SistemaDAC()
    deposito = DepositoSecundarioCO2(volumen_litros=15.0)
    
    flujo_h = motor.flujo_horario()
    
    # Arrays para gráficas
    horas_simuladas = np.linspace(0, duracion_horas, int(duracion_horas * 60))
    co2_capturado_acumulado = []
    co2_emitido_acumulado = []
    deposito_porcentaje = []
    energia_disponible_acumulada = []
    
    co2_cap_total = 0
    co2_emit_total = 0
    energia_dac_total = 0
    
    for hora in horas_simuladas:
        # Por cada hora
        co2_entrada = flujo_h['co2_kg_h']
        co2_capturado = dac.co2_capturado(co2_entrada)
        co2_emitido = co2_entrada - co2_capturado
        
        # Llenar depósito
        co2_guardado, co2_overflow = deposito.llenar(co2_capturado)
        
        # Energía
        energia_dac = dac.energia_dac_kw * 1.0  # 1 hora
        
        # Acumulativos
        co2_cap_total += co2_guardado
        co2_emit_total += co2_emitido + co2_overflow
        energia_dac_total += energia_dac
        
        # Registro
        co2_capturado_acumulado.append(co2_cap_total)
        co2_emitido_acumulado.append(co2_emit_total)
        deposito_porcentaje.append(deposito.porcentaje_lleno())
        energia_disponible_acumulada.append(dac.calor_residual_disponible_kw * hora)
    
    # RESULTADOS
    print("\n" + "="*70)
    print("GEMELO 1: DIRECT AIR CAPTURE (DAC) - RESULTADOS 8 HORAS")
    print("="*70)
    print(f"CO2 CAPTURADO:          {co2_cap_total:.2f} kg")
    print(f"CO2 EMITIDO:            {co2_emit_total:.2f} kg")
    print(f"REDUCCIÓN:              {(co2_cap_total/(co2_cap_total+co2_emit_total)*100):.1f}%")
    print(f"DEPÓSITO UTILIZADO:     {deposito_porcentaje[-1]:.1f}%")
    print(f"ENERGÍA DAC CONSUMIDA:  {energia_dac_total:.2f} kWh")
    print(f"ENERGÍA RESIDUAL TOTAL: {dac.calor_residual_disponible_kw * duracion_horas:.2f} kWh")
    print("="*70)
    
    # Gráficas
    plt.figure(figsize=(14, 10))
    
    plt.subplot(2, 2, 1)
    plt.plot(horas_simuladas, co2_capturado_acumulado, 'g-', linewidth=2, label='CO2 Capturado')
    plt.plot(horas_simuladas, co2_emitido_acumulado, 'r-', linewidth=2, label='CO2 Emitido')
    plt.xlabel('Tiempo (horas)')
    plt.ylabel('CO2 acumulado (kg)')
    plt.title('Captura vs Emisión de CO2')
    plt.legend()
    plt.grid(True)
    
    plt.subplot(2, 2, 2)
    plt.fill_between(horas_simuladas, deposito_porcentaje, alpha=0.5, color='blue')
    plt.axhline(y=100, color='r', linestyle='--', label='Capacidad máxima')
    plt.xlabel('Tiempo (horas)')
    plt.ylabel('Depósito CO2 (%)')
    plt.title(f'Llenado Depósito Secundario (Capacidad: {deposito.masa_max_co2:.1f} kg)')
    plt.legend()
    plt.grid(True)
    
    plt.subplot(2, 2, 3)
    plt.plot(horas_simuladas, energia_disponible_acumulada, 'orange', linewidth=2)
    plt.xlabel('Tiempo (horas)')
    plt.ylabel('Energía acumulada (kWh)')
    plt.title('Energía Calor Residual Disponible')
    plt.grid(True)
    
    plt.subplot(2, 2, 4)
    eficiencia_captura = [(cap/(cap+emit)*100) if (cap+emit) > 0 else 0 
                          for cap, emit in zip(co2_capturado_acumulado, co2_emitido_acumulado)]
    plt.plot(horas_simuladas, eficiencia_captura, 'purple', linewidth=2)
    plt.xlabel('Tiempo (horas)')
    plt.ylabel('Eficiencia captura (%)')
    plt.ylim([0, 100])
    plt.title('Eficiencia de Captura a lo largo del tiempo')
    plt.grid(True)
    
    plt.tight_layout()
    plt.savefig('gemelo_1_dac_resultados.png', dpi=300)
    plt.show()
    
    return {
        'co2_capturado_kg': co2_cap_total,
        'co2_emitido_kg': co2_emit_total,
        'eficiencia_captura_pct': (co2_cap_total/(co2_cap_total+co2_emit_total)*100),
        'deposito_utilizado_pct': deposito_porcentaje[-1],
        'energia_consumida_dac_kwh': energia_dac_total,
        'energia_residual_total_kwh': dac.calor_residual_disponible_kw * duracion_horas,
    }

if __name__ == "__main__":
    resultados = simular_dac_ciclo_completo(duracion_horas=8.0)

"""
GEMELO 2: REACTOR SABATIER - SÍNTESIS CO2 → GASOIL SINTÉTICO

Simula:
1. Entrada: CO2 capturado (depósito secundario)
2. Electrolizador: H2O → H2 + O2 (requiere energía)
3. Reactor Sabatier: CO2 + 4 H2 → CH4 + 2 H2O (200-400°C)
4. Síntesis Fischer-Tropsch: CH4 → gasoil (C12H23)
5. Energía requerida: 46 MJ/mol CO2 = 12.8 kWh/mol
6. Fuente energía: Kilómetro (1-2 kWh disponible)
"""

import numpy as np
from dataclasses import dataclass
import json

@dataclass
class ReactorSabatier:
    """Reactor catalítico Sabatier (CO2 + H2 → CH4)"""
    temp_c: float = 300  # Temperatura de operación
    presion_bar: float = 30
    catalizador: str = "Ni-Ru/Al2O3"
    eficiencia_conversion: float = 0.92  # 92% de conversión
    
    # Energética
    energia_requerida_mj_mol_co2: float = 46.0  # Teórica
    energia_requerida_kwh_mol_co2: float = 12.8  # En kWh
    
    def convertir_co2_a_ch4(self, co2_kg: float) -> Tuple[float, float]:
        """Convierte CO2 a CH4 (metano)"""
        # Masa molar: CO2 = 44 kg/kmol, CH4 = 16 kg/kmol
        co2_kmol = co2_kg / 44
        ch4_kmol = co2_kmol * self.eficiencia_conversion  # 1 CO2 → 1 CH4
        ch4_kg = ch4_kmol * 16
        
        energia_requerida_kwh = co2_kmol * self.energia_requerida_kwh_mol_co2
        
        return ch4_kg, energia_requerida_kwh

class ElectrolizadorPEM:
    """Electrolizador PEM para producir H2 a partir de H2O"""
    def __init__(self):
        self.eficiencia: float = 0.70  # 70% eficiencia (round-trip)
        self.energia_kwh_kg_h2: float = 52.5  # kWh/kg H2
    
    def producir_h2(self, energia_disponible_kwh: float) -> float:
        """Produce H2 disponible con energía dada"""
        h2_kg = (energia_disponible_kwh * self.eficiencia) / self.energia_kwh_kg_h2
        return h2_kg

@dataclass
class SintesisFischerTropsch:
    """Convierte CH4 → Gasoil (C12H23) de cadena larga"""
    presion_bar: float = 25
    temp_c: float = 250
    catalizador: str = "Fe-Co/SiO2"
    rendimiento_molar: float = 0.80  # 80% rendimiento
    
    def ch4_a_gasoil(self, ch4_kg: float) -> float:
        """Convierte CH4 a gasoil sintético"""
        # Aproximación: 4 CH4 → 1 C12H23 (gasoil)
        # Masa molar: CH4 = 16, C12H23 ≈ 170
        
        ch4_kmol = ch4_kg / 16
        gasoil_kmol = ch4_kmol * 0.25 * self.rendimiento_molar  # 4 CH4 → 1 gasoil
        gasoil_kg = gasoil_kmol * 170
        
        return gasoil_kg

class KilometroEnergia:
    """Modelado de energía extra generada por Kilómetro"""
    def __init__(self):
        self.energia_extra_kwh_por_ciclo: float = 1.5  # 1-2 kWh
        self.ciclos_por_hora: float = 120  # Aproximado
    
    def energia_disponible_hora(self) -> float:
        return self.energia_extra_kwh_por_ciclo * self.ciclos_por_hora

def simular_sintesis_completa(co2_capturado_kg: float, energia_kilometro_kwh: float):
    """Simula ciclo completo: CO2 → CH4 → Gasoil"""
    
    sabatier = ReactorSabatier()
    electrolizador = ElectrolizadorPEM()
    fischer = SintesisFischerTropsch()
    
    print("\n" + "="*70)
    print("GEMELO 2: SÍNTESIS CO2 → GASOIL SINTÉTICO")
    print("="*70)
    print(f"ENTRADA CO2 CAPTURADO:  {co2_capturado_kg:.2f} kg")
    print(f"ENERGÍA DISPONIBLE:     {energia_kilometro_kwh:.2f} kWh")
    print()
    
    # PASO 1: Electrolizador
    h2_producible_kg = electrolizador.producir_h2(energia_kilometro_kwh)
    print(f"PASO 1: ELECTRÓLISIS")
    print(f"  H2 producible:        {h2_producible_kg:.3f} kg")
    
    # Limitante: CO2 o H2
    # Reacción: CO2 + 4 H2 → CH4 + 2 H2O
    # Necesidad de H2 por CO2
    h2_requerido_kg = (co2_capturado_kg / 44) * 4 * 2  # 44=MMolar CO2, 2=M Hidrógeno
    h2_limitante = min(h2_producible_kg, h2_requerido_kg)
    co2_a_convertir = (h2_limitante / 8) * 44  # H2 limitante
    
    print(f"  H2 requerido:         {h2_requerido_kg:.3f} kg")
    print(f"  H2 LIMITANTE:         {h2_limitante:.3f} kg ← Cuello de botella")
    print()
    
    # PASO 2: Reactor Sabatier
    print(f"PASO 2: REACTOR SABATIER (CO2 + 4 H2 → CH4)")
    ch4_kg, energia_sabatier = sabatier.convertir_co2_a_ch4(co2_a_convertir)
    print(f"  CO2 convertido:       {co2_a_convertir:.2f} kg")
    print(f"  CH4 producido:        {ch4_kg:.2f} kg")
    print(f"  Energía requerida:    {energia_sabatier:.2f} kWh")
    print()
    
    # PASO 3: Fischer-Tropsch
    print(f"PASO 3: FISCHER-TROPSCH (CH4 → GASOIL)")
    gasoil_kg = fischer.ch4_a_gasoil(ch4_kg)
    gasoil_litros = gasoil_kg / 0.8  # Densidad gasoil ≈ 0.8 kg/L
    print(f"  CH4 entrada:          {ch4_kg:.2f} kg")
    print(f"  Gasoil producido:     {gasoil_kg:.2f} kg = {gasoil_litros:.2f} L")
    print()
    
    # ANÁLISIS ENERGÉTICO
    print(f"ANÁLISIS ENERGÉTICO:")
    print(f"  Energía entrada (Kilómetro):  {energia_kilometro_kwh:.2f} kWh")
    print(f"  Energía Sabatier:             {energia_sabatier:.2f} kWh")
    print(f"  Relación:                     {(energia_sabatier/energia_kilometro_kwh)*100:.1f}%")
    
    if energia_sabatier > energia_kilometro_kwh:
        deficit = energia_sabatier - energia_kilometro_kwh
        print(f"  ⚠️  DÉFICIT:                   {deficit:.2f} kWh ← Requiere energía externa")
    else:
        exceso = energia_kilometro_kwh - energia_sabatier
        print(f"  ✅ EXCESO:                    {exceso:.2f} kWh ← Disponible para otros usos")
    
    print()
    
    # RESULTADO NETO
    print(f"RESULTADO NETO POR CICLO:")
    print(f"  CO2 capturado:        {co2_capturado_kg:.2f} kg")
    print(f"  Gasoil regenerado:    {gasoil_litros:.3f} L = {gasoil_kg:.2f} kg")
    print(f"  Ratio regeneración:   {(gasoil_litros/6.67)*100:.1f}% del consumo")
    print("="*70)
    
    return {
        'co2_entrada_kg': co2_capturado_kg,
        'h2_producido_kg': h2_limitante,
        'ch4_producido_kg': ch4_kg,
        'gasoil_regenerado_litros': gasoil_litros,
        'gasoil_regenerado_kg': gasoil_kg,
        'energia_requerida_kwh': energia_sabatier,
        'energia_disponible_kwh': energia_kilometro_kwh,
        'deficit_exceso_kwh': energia_kilometro_kwh - energia_sabatier,
        'ratio_regeneracion_pct': (gasoil_litros/6.67)*100,
    }

if __name__ == "__main__":

"""
GEMELO 3: CICLO COMPLETO INTEGRADO

Simula durante 1 SEMANA DE CONDUCCIÓN:
- Motor gasoil consume 1L/15km
- DAC captura 85% del CO2
- Kilómetro genera 1-2 kWh extra/ciclo
- Síntesis reconvierte CO2 → gasoil sintético
- Balance neto de gasoil + emisiones

RESULTADO: Evaluación de viabilidad real
"""

import numpy as np
import matplotlib.pyplot as plt
from dataclasses import dataclass
import json

@dataclass
class SemanadeConductor:
    """Perfil de conducción semanal típico"""
    km_lunes_viernes: float = 60  # km/día
    km_sabado: float = 100  # km
    km_domingo: float = 0  # Descanso
    dias_semana: int = 7
    
    def total_km(self) -> float:
        return 5 * self.km_lunes_viernes + self.km_sabado + self.km_domingo

class CicloCompletoSemanal:
    def __init__(self):
        self.consumo_l_por_km = 1/15  # 15 km/L
        self.co2_kg_por_litro = 3.15
        self.eficiencia_dac = 0.88
        self.energia_kilometro_kwh_por_dia = 1.5 * 120  # 120 ciclos/día
        self.energia_sabatier_kwh_por_kg_co2 = 12.8 / 44  # 0.29 kWh/kg
    
    def simular_semana(self, km_totales: float) -> dict:
        """Simula una semana completa"""
        
        # Consumo
        gasoil_consumido_l = km_totales * self.consumo_l_por_km
        co2_total_kg = gasoil_consumido_l * self.co2_kg_por_litro
        
        # Captura
        co2_capturado_kg = co2_total_kg * self.eficiencia_dac
        co2_emitido_kg = co2_total_kg - co2_capturado_kg
        
        # Energía disponible
        energia_disponible_semana = self.energia_kilometro_kwh_por_dia * 7
        
        # Síntesis: Cuánto CO2 puede regenerar con energía disponible
        energia_para_sintesis = energia_disponible_semana
        co2_sintetizable_kg = energia_para_sintesis / self.energia_sabatier_kwh_por_kg_co2
        
        # Limitante: CO2 capturado o CO2 sintetizable
        co2_a_sintetizar = min(co2_capturado_kg, co2_sintetizable_kg)
        
        # Gasoil regenerado
        # 1 kg CO2 → 0.27 L gasoil (aproximado por estequiometría)
        gasoil_regenerado_l = co2_a_sintetizar * 0.27
        
        # Balance neto
        gasoil_neto_l = gasoil_consumido_l - gasoil_regenerado_l
        co2_emitido_final_kg = co2_emitido_kg + (co2_capturado_kg - co2_a_sintetizar)
        
        return {
            'km_totales': km_totales,
            'gasoil_consumido_l': gasoil_consumido_l,
            'gasoil_regenerado_l': gasoil_regenerado_l,
            'gasoil_neto_l': gasoil_neto_l,
            'co2_total_emitido_kg': co2_total_kg,
            'co2_capturado_kg': co2_capturado_kg,
            'co2_sintetizado_kg': co2_a_sintetizar,
            'co2_emitido_final_kg': co2_emitido_final_kg,
            'porcentaje_gasoil_ahorrado': (gasoil_regenerado_l / gasoil_consumido_l) * 100,
            'porcentaje_co2_capturado': (co2_capturado_kg / co2_total_kg) * 100,
            'energia_disponible_kwh': energia_disponible_semana,
            'energia_requerida_kwh': co2_a_sintetizar * self.energia_sabatier_kwh_por_kg_co2,
        }

if __name__ == "__main__":
    semana = SemanadeConductor(km_lunes_viernes=60, km_sabado=100)
    ciclo = CicloCompletoSemanal()
    
    km_total = semana.total_km()
    resultados = ciclo.simular_semana(km_total)
    
    print("\n" + "="*70)
    print("GEMELO 3: CICLO COMPLETO SEMANAL")
    print("="*70)
    print(f"\nCONDUCCIÓN:")
    print(f"  Km totales/semana:    {km_total:.0f} km")
    print(f"  Gasoil consumido:     {resultados['gasoil_consumido_l']:.2f} L")
    print(f"\nCAPITURA Y SÍNTESIS:")
    print(f"  CO2 capturado:        {resultados['co2_capturado_kg']:.2f} kg ({resultados['porcentaje_co2_capturado']:.1f}%)")
    print(f"  CO2 sintetizado:      {resultados['co2_sintetizado_kg']:.2f} kg")
    print(f"  Gasoil regenerado:    {resultados['gasoil_regenerado_l']:.2f} L ({resultados['porcentaje_gasoil_ahorrado']:.1f}%)")
    print(f"\nBALANCE NETO:")
    print(f"  Gasoil neto:          {resultados['gasoil_neto_l']:.2f} L")
    print(f"  CO2 final emitido:    {resultados['co2_emitido_final_kg']:.2f} kg")
    print(f"  REDUCCIÓN CO2:        {((resultados['co2_total_emitido_kg']-resultados['co2_emitido_final_kg'])/resultados['co2_total_emitido_kg']*100):.1f}%")
    print("="*70)
    
    with open('gemelo_3_ciclo_completo_metricas.json', 'w') as f:
        json.dump(resultados, f, indent=2)

            ingresos_venta_camiones = volumen_camiones * margen_camion
            ingresos_recurrentes_camiones = volumen_camiones * INGRESOS_RECURRENTES_ANUAL_CAMION['subtotal']
            
            año_dict['ingresos_venta'] = ingresos_venta_coches + ingresos_venta_camiones
            año_dict['ingresos_recurrentes'] = ingresos_recurrentes_coches + ingresos_recurrentes_camiones
            año_dict['ingresos_total'] = año_dict['ingresos_venta'] + año_dict['ingresos_recurrentes']
            
            # FLUJO NETO
            año_dict['flujo_neto'] = año_dict['ingresos_total'] - año_dict['costos_rdsm']
            
            # FLUJO DESCONTADO
            años_transcurridos = año - AÑO_INICIO
            año_dict['factor_descuento'] = 1 / ((1 + TASA_DESCUENTO) ** años_transcurridos)
            año_dict['flujo_descontado'] = año_dict['flujo_neto'] * año_dict['factor_descuento']
            
            flows.append(año_dict)
        
        self.df_flows = pd.DataFrame(flows)
        return self.df_flows
    
    def calcular_metricas(self):
        """Calcula VPN, TIR, Payback"""
        
        # VPN (Valor Presente Neto)
        self.vpn_total = self.df_flows['flujo_descontado'].sum()
        
        # Payback Period
        flujo_acumulado = self.df_flows['flujo_neto'].cumsum()
        idx_payback = (flujo_acumulado >= 0).idxmax()
        self.payback = self.df_flows.loc[idx_payback, 'año'] - AÑO_INICIO if flujo_acumulado[idx_payback] >= 0 else None
        
        return {
            'vpn_millones': self.vpn_total / 1_000_000,
            'payback_anos': self.payback,
        }
    
    def print_resumen(self):
        """Imprime resumen ejecutivo"""
        
        print("\n" + "="*80)
        print("ANÁLISIS FINANCIERO: REPSOL GASOIL VERDE 33×1 (2027-2037)")
        print("="*80)
        
        print("\n📊 FLUJOS DE CAJA (€ millones):")
        print(self.df_flows[['año', 'costos_rdsm', 'ingresos_total', 'flujo_neto']].to_string(index=False))
        
        metricas = self.calcular_metricas()
        print(f"\n💰 MÉTRICAS CLAVE:")
        print(f"  VPN (10 años):           €{metricas['vpn_millones']:.0f}M")
        print(f"  Payback period:          {metricas['payback_anos']:.1f} años")
        print(f"  ROI total:               {((self.df_flows['flujo_neto'].sum() / (self.df_flows['costos_rdsm'].sum())) * 100):.0f}%")
        
        print("\n🚗 RENTABILIDAD POR VEHÍCULO (PRIMER AÑO):")
