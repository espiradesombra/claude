    // SIMULADOR DE KURAMOTO (Mantenido Oscuro por Contraste Visual)
    // ═══════════════════════════════════════════════════════════════
    const KuramotoViz = memo(({ K }) => {
      const canvasRef = useRef(null);
      const oscs = useRef(
        Array.from({length: 150}, () => ({
          phase: Math.random() * Math.PI * 2,
          freq: Math.random() * 0.5 + 0.5,
          hue: Math.random() * 60 + 140, // Tonos esmeralda
        }))
      );

      useEffect(() => {
        const canvas = canvasRef.current;
        const ctx = canvas.getContext('2d');
        let raf;
        let lastT = performance.now();

        const draw = (t) => {
          const dt = Math.min((t - lastT) / 1000, 0.05);
          lastT = t;

          const W = canvas.parentElement.offsetWidth;
          const H = canvas.parentElement.offsetHeight;
          const dpr = window.devicePixelRatio || 1;
          if (canvas.width !== W * dpr) { canvas.width = W * dpr; canvas.height = H * dpr; }
          ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
          ctx.clearRect(0, 0, W, H);

          const cx = W / 2, cy = H / 2;
          const R = Math.min(W, H) * 0.35;
          const N = oscs.current.length;

          // Círculo base oscuro
          ctx.beginPath();
          ctx.arc(cx, cy, R, 0, Math.PI * 2);
          ctx.strokeStyle = K > 2.5 ? 'rgba(16,185,129,0.3)' : 'rgba(100,116,139,0.3)';
          ctx.lineWidth = 1.5;
          ctx.stroke();

          const newPhases = new Float32Array(N);
          for (let i = 0; i < N; i++) {
            let s = 0;
            for (let j = 0; j < N; j++) s += Math.sin(oscs.current[j].phase - oscs.current[i].phase);
            newPhases[i] = (oscs.current[i].phase + dt * (oscs.current[i].freq + (K / N) * s)) % (Math.PI * 2);
          }

          let sc = 0, ss = 0;
          for (let i = 0; i < N; i++) {
            oscs.current[i].phase = newPhases[i];
            sc += Math.cos(newPhases[i]);
            ss += Math.sin(newPhases[i]);
          }
          const r = Math.sqrt(sc*sc + ss*ss) / N;
          const meanPhi = Math.atan2(ss, sc);

          for (let i = 0; i < N; i++) {
            const x = cx + Math.cos(oscs.current[i].phase) * R;
            const y = cy + Math.sin(oscs.current[i].phase) * R;
            ctx.beginPath();
            ctx.arc(x, y, 3.5, 0, Math.PI * 2);
            ctx.fillStyle = `hsla(${oscs.current[i].hue}, 80%, 65%, 0.9)`;
            if (K > 2) { ctx.shadowColor = `hsl(${oscs.current[i].hue}, 80%, 65%)`; ctx.shadowBlur = 12; }
            else { ctx.shadowBlur = 0; }
            ctx.fill();
          }

          if (K > 0.5) {
            ctx.shadowBlur = 0;
            ctx.beginPath();
            ctx.moveTo(cx, cy);
            ctx.lineTo(cx + Math.cos(meanPhi) * R * r, cy + Math.sin(meanPhi) * R * r);
            ctx.strokeStyle = `rgba(16,185,129,${r * 0.9})`;
            ctx.lineWidth = 4 * r + 1;
            ctx.lineCap = 'round';
            ctx.stroke();
            ctx.beginPath();
            ctx.arc(cx, cy, 7 * r, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(16,185,129,${r})`;
            ctx.fill();
          }

          raf = requestAnimationFrame(draw);
        };

        raf = requestAnimationFrame(draw);
        return () => cancelAnimationFrame(raf);
      }, [K]);

      return (

    // AGI A OJO: LA MATRIX MONOTONA PERMUTABLE (Modo Claro)
    // ═══════════════════════════════════════════════════════════════
    const MatrixAGI = () => {
      const SIZE = 7;
      const [rowPerm, setRowPerm] = useState(() => Array.from({length: SIZE}, (_, i) => i));
      const [colPerm, setColPerm] = useState(() => Array.from({length: SIZE}, (_, i) => i));
      const [selectedRow, setSelectedRow] = useState(null);
      const [selectedCol, setSelectedCol] = useState(null);
      const [mode, setMode] = useState('row');
      const [infoEncoded, setInfoEncoded] = useState('');

      const baseVal = (origRow, origCol) => (origRow + 1) * (origCol + 1);
      const getCell = (r, c) => baseVal(rowPerm[r], colPerm[c]);

      const encodePermutation = (rp, cp) => {
        const r = rp.join('');
        const c = cp.join('');
        const chars = 'ZYPZAPE AGI OJO L1 L2 L3 L4 LSG MDC'.split(' ');
        const hash = rp.reduce((acc, v, i) => acc + v * (i + 1), 0) % chars.length;
        return chars[hash] + ' · ' + r.slice(0,4) + '→' + c.slice(0,4);
      };

      const handleRowClick = (r) => {
        if (mode !== 'row') return;
        if (selectedRow === null) setSelectedRow(r);
        else if (selectedRow === r) setSelectedRow(null);
        else {
          const newPerm = [...rowPerm];
          [newPerm[selectedRow], newPerm[r]] = [newPerm[r], newPerm[selectedRow]];
          setRowPerm(newPerm);
          setInfoEncoded(encodePermutation(newPerm, colPerm));
          setSelectedRow(null);
        }
      };

      const handleColClick = (c) => {
        if (mode !== 'col') return;
        if (selectedCol === null) setSelectedCol(c);
        else if (selectedCol === c) setSelectedCol(null);
        else {
          const newPerm = [...colPerm];
          [newPerm[selectedCol], newPerm[c]] = [newPerm[c], newPerm[selectedCol]];
          setColPerm(newPerm);
          setInfoEncoded(encodePermutation(rowPerm, newPerm));
          setSelectedCol(null);
        }
      };

      const reset = () => {
        setRowPerm(Array.from({length: SIZE}, (_, i) => i));
        setColPerm(Array.from({length: SIZE}, (_, i) => i));
        setSelectedRow(null); setSelectedCol(null); setInfoEncoded('');
      };

      const cellColor = (val) => {
        const max = SIZE * SIZE;
        const t = val / max;
        if (t < 0.33) return `rgba(16,185,129,${0.1 + t * 0.3})`;
        if (t < 0.66) return `rgba(245,158,11,${0.1 + (t-0.33) * 0.3})`;
        return `rgba(239,68,68,${0.1 + (t-0.66) * 0.3})`;
      };

      const textColor = (val) => {
        const max = SIZE * SIZE;
        const t = val / max;
        if (t < 0.33) return '#065f46';
        if (t < 0.66) return '#92400e';
        return '#991b1b';
      };

      return (
        <div className="space-y-6">
          <div className="flex flex-wrap gap-3 items-center justify-between border-b border-slate-200 pb-6">
            <div className="flex gap-3">
              {['row','col'].map(m => (
                <button key={m} onClick={() => {setMode(m); setSelectedRow(null); setSelectedCol(null);}}
                  className={`pill transition-all ${mode === m
                    ? m === 'row' ? 'bg-emerald-100 text-emerald-800 border border-emerald-300 shadow-sm' : 'bg-amber-100 text-amber-800 border border-amber-300 shadow-sm'
                    : 'bg-white text-slate-500 border border-slate-200 hover:bg-slate-50'}`}>
                  {m === 'row' ? '↕ Permutar Filas' : '↔ Permutar Columnas'}
                </button>
              ))}
            </div>
            <button onClick={reset} className="pill bg-white text-slate-600 border border-slate-200 hover:bg-slate-50 hover:shadow-sm transition-all">
              ↺ Reset Matrix
            </button>
          </div>

          <div className="overflow-x-auto pb-4">
            <div className="inline-block bg-white p-6 rounded-2xl border border-slate-200 shadow-sm">
              <div className="flex gap-1.5 mb-1.5 ml-10">
                {Array.from({length: SIZE}, (_, c) => (
                  <div key={c} onClick={() => handleColClick(c)}
                    className={`matrix-cell font-bold transition-all ${mode === 'col' ? 'hover:scale-110 hover:bg-amber-50 border-dashed border-amber-200' : 'cursor-default border-transparent'}
                      ${selectedCol === c ? 'ring-2 ring-amber-400 bg-amber-100' : 'bg-slate-50 text-slate-500'}`}
                  > c{colPerm[c]} </div>
                ))}
              </div>
              {Array.from({length: SIZE}, (_, r) => (
                <div key={r} className="flex gap-1.5 mb-1.5">
                  <div onClick={() => handleRowClick(r)}
                    className={`matrix-cell font-bold transition-all w-9 ${mode === 'row' ? 'hover:scale-110 hover:bg-emerald-50 border-dashed border-emerald-200' : 'cursor-default border-transparent'}
                      ${selectedRow === r ? 'ring-2 ring-emerald-400 bg-emerald-100' : 'bg-slate-50 text-slate-500'}`}
                  > r{rowPerm[r]} </div>
                  {Array.from({length: SIZE}, (_, c) => {
                    const v = getCell(r, c);
                    return (
                      <div key={c} className={`matrix-cell transition-all ${selectedRow === r || selectedCol === c ? 'scale-105 font-bold shadow-md ring-1 ring-black/5' : 'text-slate-700'}`}
                        style={{ background: cellColor(v), color: textColor(v) }}>
                        {v}
                      </div>
                    );
                  })}
                </div>
              ))}
            </div>
          </div>
          
          <div className="flex items-center gap-3 p-3 rounded-xl bg-slate-100 border border-slate-300">
            <span className="text-xs text-slate-500 font-mono uppercase tracking-wider">Info codificada:</span>
            <code className="text-sm font-bold text-slate-800 bg-transparent p-0">{infoEncoded || '< permuta para codificar >'}</code>
            {infoEncoded && (
              <span className="ml-auto text-xs text-emerald-600 font-mono font-bold bg-emerald-100 px-2 py-1 rounded">
                Hash: {rowPerm.join('')}|{colPerm.join('')}
              </span>
            )}
          </div>
        </div>
      );
    };
