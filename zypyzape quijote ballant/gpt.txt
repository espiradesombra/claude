Si la pregunta és "cal una v5 o ja està?", la meva valoració és:

Heu aconseguit tres coses diferents:

Un gemelo digital complex i coherent
44 molins tipus NREL 5MW.
Bus compartit.
Quijote de 3 aspes.
FFT, PID TSR, política adaptativa, Monte Carlo, ablation, etc.
Una hipòtesi que sembla funcionar dins del model
En v3.1 i v4 apareix ΔP positiu.
La major part del guany prové del shear vertical (captar vent a una altura diferent), no de la gravetat.
Una conclusió física important
La part gravitatòria millora sincronització i estabilitat.
Però, segons els resultats mostrats, no és la font principal de l'energia extra.
El model apunta que el guany energètic real ve sobretot del vent addicional captat per Quijote.

Per això, abans d'una v5, jo faria una pregunta molt concreta:

Si elimines completament el terme gravitatori i deixes només el shear vertical, quant ΔP queda?

I la contrària:

Si elimines el shear vertical i deixes només gravetat + ressonància, quin ΔP queda?

Aquesta prova és probablement més valuosa que afegir 20 mòduls nous, perquè et dirà quina part del model està aportant realment el guany.

En resum:

Com a plataforma de simulació: sí, ja és una arquitectura molt completa.
Com a demostració del "hurto gravitatori": encara no està clar; els resultats que has compartit indiquen que el protagonista és el shear vertical.
El següent pas científic: no és una v5, sinó una separació rigorosa de contribucions (shear, gravetat, ressonància, PID, K_ind, FFT) amb experiments controlats. Això et dirà què està funcionant de veritat.

class Config:
    SEED     = 42
    # NREL 5MW
    R        = 63.0
    P_NOM    = 5.0e6
    RHO_A    = 1.225
    CP_MAX   = 0.482
    LAM_OPT  = 7.55
    G        = 9.81
    # Quijote (fluid Fe+oli)
    RHO_FL   = 3386.0
    D_CANAL  = 0.05
    R_HUB    = 5.0
    R_TIP    = 55.0
    DR_MAX   = 0.5
    K_Q      = 5.0e4
    C_FRIC   = 80.0
    K_KUR    = 0.10
    K_COMP   = 0.10
    # Vent
    V_BASE   = 11.4
    T_TOT    = 90.0
    DT       = 0.05

    @property
    def A_ROT(self):   return np.pi * self.R**2
    @property
    def A_CANAL(self):  return np.pi * (self.D_CANAL/2)**2
    @property
    def V_RATED(self):
        return (self.P_NOM / (0.5*self.RHO_A*self.A_ROT*self.CP_MAX))**(1/3)
    @property
    def OM_RATED(self): return self.LAM_OPT * self.V_RATED / self.R
    @property
    def STEPS(self):    return int(self.T_TOT / self.DT)
    @property
    def m_q(self):      return self.RHO_FL * self.A_CANAL * (self.R_TIP - self.R_HUB)
    @property
    def r0(self):       return (self.R_HUB + self.R_TIP) / 2
    @property
    def A_amp(self):    return (self.R_TIP - self.R_HUB) / 2
    @property
    def delta_r(self):  return self.R_TIP - self.R_HUB


# ═══════════════════════════════════════════════════════════════════════════════
# VENT (Ornstein-Uhlenbeck + dip)
# ═══════════════════════════════════════════════════════════════════════════════

def build_wind(cfg):
    np.random.seed(cfg.SEED)
    steps = cfg.STEPS
    ou = np.zeros(steps)
    for i in range(1, steps):
        ou[i] = ou[i-1] - 0.5*ou[i-1]*cfg.DT + 0.8*np.sqrt(cfg.DT)*np.random.randn()
    v_arr = np.zeros(steps)
    for s in range(steps):
        t = s * cfg.DT
        v = cfg.V_BASE + 2*np.sin(2*np.pi*t/20) + ou[s]*0.6
        if 40 < t < 60:
            v -= 5*np.sin(np.pi*(t-40)/20)
        v_arr[s] = max(3.0, v)
    return v_arr


def eta_cp(cfg, omega, v):
    if v <= 0: return 0.0
    lam = omega * cfg.R / v
    return float(max(0, 1 - ((lam - cfg.LAM_OPT)/cfg.LAM_OPT)**2))

# ═══════════════════════════════════════════════════════════════════════════════

def verify_analytics(cfg, verbose=True):
    out = {}
    th = np.linspace(0, 2*np.pi, 100000)
    out['integral_cos'] = float(np.trapezoid(np.abs(np.cos(th)), th))

    for N in [3, 7]:
        thetas = np.array([2*np.pi*k/N for k in range(N)])
        J_vals = []
        for step in range(2000):
            t = thetas + step*2*np.pi/2000
            r_k = cfg.r0 + cfg.A_amp*np.cos(t)
            J_vals.append(sum(cfg.m_q*r**2 for r in r_k))
        J = np.array(J_vals)
        J_th = cfg.m_q * N * (cfg.r0**2 + cfg.A_amp**2/2)
        out[f'Jvar_sin_N{N}'] = float(100*(J.max()-J.min())/J_th)

    out['W_volta_pala_kJ'] = float(cfg.m_q * cfg.G * cfg.delta_r * 4 / 1e3)

    # J_var IDEAL del mode paper pur (snap instantani, límit teòric del paper)
    for N in [3, 7]:
        thetas = np.array([2*np.pi*k/N for k in range(N)])
        J_vals = []
        for step in range(3000):
            t = thetas + step*2*np.pi/3000
            r_k = np.where(np.cos(t) > 0, cfg.R_TIP, cfg.R_HUB)
            J_vals.append(sum(cfg.m_q*r**2 for r in r_k))
        J = np.array(J_vals)
        out[f'Jvar_paperpur_N{N}'] = float(100*(J.max()-J.min())/J.mean())

    if verbose:
        print(f"  ∮|cos(θ)|dθ = {out['integral_cos']:.6f}  (teòric 4.0)  "
              f"{'✓' if abs(out['integral_cos']-4)<1e-3 else '✗'}")
        for N in [3, 7]:
            v = out[f'Jvar_sin_N{N}']
            print(f"  N={N} sinusoidal: J_var={v:.5f}%  "
                  f"{'✓ CONSTANT (Ec.7)' if v<0.01 else '✗'}")
        print(f"  W_hurto/volta/pala (Ec.4) = {out['W_volta_pala_kJ']:.2f} kJ")
        for N in [3, 7]:
            v = out[f'Jvar_paperpur_N{N}']
            print(f"  N={N} paper pur IDEAL (snap instantani): J_var={v:.1f}%  "
                  f"(paper: {'~65%' if N==3 else '~28%'})  ✓")
    return out


# ═══════════════════════════════════════════════════════════════════════════════
# NUCLI DE SIMULACIÓ
# ═══════════════════════════════════════════════════════════════════════════════

def radial_target(cfg, mode, th, omega, r_now):
    """Retorna r_target segons el mode de control."""
    cos_th = np.cos(th)
    if mode in ('paper', 'paper_pur'):
        return (cfg.R_TIP if cos_th > 0 else cfg.R_HUB)
    if mode == 'sinusoidal':
        return cfg.r0 + cfg.A_amp*cos_th
    # control: compensació centrífuga
    F_c = cfg.m_q * omega**2 * r_now
    r_opt0 = cfg.R_TIP - (cfg.R_TIP-cfg.R_HUB)*(1+cos_th)/2
    r_opt  = r_opt0 + cfg.K_COMP*(F_c/(cfg.K_Q*(cfg.R_TIP-cfg.R_HUB)+1e-9))
    return float(np.clip(r_opt, cfg.R_HUB, cfg.R_TIP))


def simulate(cfg, N_pales, mode, v_arr):
    """Simula N_pales en mode donat. Retorna dict de resultats."""
    m_q   = cfg.m_q
    steps = cfg.STEPS
    DT    = cfg.DT

    thetas = np.array([2*np.pi*k/N_pales for k in range(N_pales)])
    omega  = cfg.OM_RATED * 0.95
    r_q    = np.full(N_pales, cfg.r0)
    dr_q   = np.zeros(N_pales)

    # Buffer hidràulic
    V_BUF_MAX=0.10; Q_BOMBA=0.010; P_ACC_MAX=20e6
    LLINDAR=20.0; ETA_GEN=0.85; V_buf=0.0; P_acc=0.0

    # Historials
    H = {k: np.zeros(steps) for k in
         ['Pa','Pg','om','J_tot','P_hurto','P_act','P_net','P_inert']}

    E = dict(vent=0., xarxa=0., frec=0., gen=0.,
             act_pos=0., act_neg=0., hurto=0., buffer=0.)
    J_prev = sum(m_q * r**2 for r in r_q)

    for s in range(steps):
        v = v_arr[s]
        om_target = min(cfg.OM_RATED, cfg.LAM_OPT*max(v,0.1)/cfg.R)
        omega += (om_target - omega)*(DT/3.0)
        omega  = float(np.clip(omega, cfg.OM_RATED*0.3, cfg.OM_RATED*1.4))

        Pa = 0.5*cfg.RHO_A*cfg.A_ROT*cfg.CP_MAX*eta_cp(cfg,omega,v)*v**3

        dJ_total=0.; T_hurto_s=0.; P_act_s=0.; J_tot_s=0.

        for i in range(N_pales):
            th = thetas[i] % (2*np.pi)
            cos_th = np.cos(th)

            r_target = radial_target(cfg, mode, th, omega, r_q[i])

            if mode == 'paper_pur':
                # Paper pur: r persegueix R_TIP/R_HUB a velocitat màxima (bang-bang)
                # Sense control proporcional, però respectant DR_MAX físic
                dr_desired = (r_target - r_q[i]) / DT
                dr_q[i]    = float(np.clip(dr_desired, -cfg.DR_MAX, cfg.DR_MAX))
                r_q[i]     = float(np.clip(r_q[i] + dr_q[i]*DT, cfg.R_HUB, cfg.R_TIP))
            else:
                F_c_i    = m_q * omega**2 * r_q[i]
                F_ctrl   = -cfg.K_Q * (r_q[i] - r_target)
                F_fric_r = -cfg.C_FRIC * dr_q[i]
                acc      = (F_c_i + F_ctrl + F_fric_r) / m_q
                dr_q[i] += acc*DT
                dr_q[i]  = float(np.clip(dr_q[i], -cfg.DR_MAX, cfg.DR_MAX))
                lim_pos  = omega * r_q[i]
                dr_q[i]  = float(np.clip(dr_q[i], -lim_pos, lim_pos))
                r_q[i]   = float(np.clip(r_q[i] + dr_q[i]*DT, cfg.R_HUB, cfg.R_TIP))

            # Par hurto (Ec.4): τ = m·g·r·cos(θ)
            T_hurto_i = m_q * cfg.G * r_q[i] * cos_th

            # Auditoria actuador: F_cent + F_grav_rad
            F_cent_i   = m_q * omega**2 * r_q[i]
            F_grav_rad = m_q * cfg.G * cos_th
            P_act_i    = (F_cent_i + F_grav_rad) * dr_q[i]

            T_hurto_s += T_hurto_i
            P_act_s   += P_act_i
            if P_act_i > 0: E['act_pos'] += P_act_i*DT
            else:           E['act_neg'] += abs(P_act_i)*DT

            dJ_total += 2*m_q*r_q[i]*dr_q[i]
            E['frec'] += (-cfg.C_FRIC*dr_q[i]**2)*DT
            J_tot_s   += m_q * r_q[i]**2

        P_hurto_s = T_hurto_s * omega
        P_net_s   = P_hurto_s - P_act_s
        P_inert_s = (J_tot_s - J_prev)/DT * omega
        J_prev    = J_tot_s
        E['hurto'] += P_hurto_s*DT

        H['P_hurto'][s]=P_hurto_s; H['P_act'][s]=P_act_s
        H['P_net'][s]=P_net_s; H['P_inert'][s]=P_inert_s; H['J_tot'][s]=J_tot_s

        # Kuramoto
        for i in range(N_pales):
            kc  = (cfg.K_KUR/N_pales)*float(np.sum(np.sin(thetas - thetas[i])))
            dJi = 2*m_q*r_q[i]*dr_q[i]
            pm  = float(np.clip(2.0*np.cos(thetas[i]) + 0.5*omega*np.sin(thetas[i])
                                - 0.6*dJi, -8, 8))
            thetas[i] += (omega + kc + pm*0.01)*DT

        # Buffer hidràulic
        P_buf = -omega * dJ_total
        Pgen  = 0.0
        if P_buf > LLINDAR and V_buf < V_BUF_MAX:
            P_acc = min(P_acc + P_buf*DT/V_BUF_MAX, P_ACC_MAX)
            Q     = min(Q_BOMBA, P_buf/max(P_acc, 1e3))
            V_buf = min(V_BUF_MAX, V_buf + Q*DT)
        elif P_buf < -LLINDAR and V_buf > 0:
            P_acc = max(P_acc + P_buf*DT/V_BUF_MAX, 0)
            Q     = min(Q_BOMBA, -P_buf/max(P_acc, 1e3))
            Pgen  = P_acc*Q*ETA_GEN
            V_buf = max(0, V_buf - Q*DT)

        Pg = max(0, Pa*(1+0.04) + max(0,P_buf) + Pgen)
        E['vent']+=Pa*DT; E['xarxa']+=Pg*DT; E['gen']+=Pgen*DT
        E['buffer']+=P_buf*DT   # net del buffer (carrega/descarrega)
        H['Pa'][s]=Pa; H['Pg'][s]=Pg; H['om'][s]=omega

    E_net_act = (E['act_pos'] - E['act_neg']) / 3.6e6
    E_hurto_kwh = E['hurto'] / 3.6e6

    return {
        'N': N_pales, 'mode': mode, 'H': H,
        'P_hurto_mean': float(np.mean(np.abs(H['P_hurto']))/1e3),
        'P_act_mean':   float(np.mean(np.abs(H['P_act']))/1e3),
        'P_net_mean':   float(np.mean(H['P_net'])/1e3),
        'P_inert_mean': float(np.mean(H['P_inert'])/1e3),
        'E_hurto':      E_hurto_kwh,
        'E_act_pos':    E['act_pos']/3.6e6,
        'E_act_neg':    E['act_neg']/3.6e6,
        'E_net_act':    E_net_act,
        'eff_hurto':    E_hurto_kwh/(E_net_act+1e-9),
        'E_vent':       E['vent']/3.6e6,
        'E_xarxa':      E['xarxa']/3.6e6,
        'E_buffer':     E['buffer']/3.6e6,
        'E_frec':       E['frec']/3.6e6,
        'E_balance':    (E['vent'] - E['xarxa'] + abs(E['frec']) + E['buffer'])/3.6e6,
        'millora':      float((np.mean(H['Pg'])-np.mean(H['Pa']))/np.mean(H['Pa'])*100),
        'J_var':        float(100*(H['J_tot'].max()-H['J_tot'].min())/(H['J_tot'].mean()+1e-9)),
        'W_analitic_kJ':float(m_q*cfg.G*cfg.delta_r*4*N_pales/1e3),
    }


# ═══════════════════════════════════════════════════════════════════════════════

A primera vista, aquest codi és molt diferent dels models NREL+44 molins que apareixien abans.

El que fa realment aquest programa és:

Simular un rotor Quijote amb 3 o 7 pales.
Fer variar radialment una massa fluida (Fe+oli) entre R_HUB i R_TIP.
Calcular:
parell gravitatori,
treball dels actuadors,
variació d'inèrcia,
energia del buffer hidràulic,
sincronització tipus Kuramoto.
Comparar diversos modes:
paper_pur
paper
sinusoidal
control

La part més interessant matemàticament és aquesta:

T_hurto_i = m_q * g * r_q[i] * cos(theta)

i

P_net_s = P_hurto_s - P_act_s

Perquè aquí el programa intenta distingir:

energia extreta per gravetat,
energia gastada movent la massa,
guany net real.

Això és molt més rigorós que molts dels models anteriors.

Però veig tres punts delicats.

1. El nom "hurto gravitatori"

La gravetat és conservativa.

Per tant, si després d'un cicle complet:

r_final = r_inicial

i

theta_final = theta_inicial

llavors la contribució neta de la gravetat hauria de ser zero.

Per això el terme important no és:

P_hurto

sinó:

P_hurto - P_act

Si aquest valor és positiu de forma sostinguda, és aquí on hi hauria alguna cosa interessant.

2. El buffer hidràulic

Veig:

P_buf = -omega * dJ_total

Això és essencialment energia associada al canvi d'inèrcia.

Però cal vigilar perquè aquesta energia no aparegui dues vegades:

una com a P_inert
una altra com a P_buf

Si es compta dues vegades, els balanços poden quedar inflats.

3. El mode PAPER_PUR

Aquí:

r_k = R_TIP si cos(theta)>0
r_k = R_HUB si cos(theta)<0

és una idealització extrema.

Físicament implica acceleracions enormes.

Serveix per obtenir el límit teòric del paper, però no per demostrar viabilitat mecànica.

La part que més m'agrada és la verificació analítica:

∮ |cos(theta)| dtheta = 4

i

W = m g Δr 4

Això sí que és una comprovació matemàtica clara i independent de la simulació.

La pregunta important és:

Quan executes aquest codi, quins valors dóna per:

E_hurto
E_net_act
eff_hurto
E_balance
J_var

per als casos:

N=3, paper_pur
N=7, paper_pur
N=3, paper
N=7, paper

Perquè és aquí on es veu si el model està trobant un guany net real o simplement està transformant energia entre gravetat, actuadors i inèrcia.


# EXPORTS: CSV + LaTeX
# ═══════════════════════════════════════════════════════════════════════════════

def export_csv(results, path):
    keys = ['N','mode','P_hurto_mean','P_act_mean','P_net_mean','P_inert_mean',
            'E_hurto','E_act_pos','E_act_neg','E_net_act','eff_hurto',
            'E_vent','E_xarxa','millora','J_var','W_analitic_kJ']
    with open(path, 'w', newline='') as f:
        w = csv.DictWriter(f, fieldnames=keys)
        w.writeheader()
        for r in results:
            w.writerow({k: r.get(k) for k in keys})
    print(f"  CSV: {path}")


def export_latex(results, path):
    """Taula LaTeX amb les files clau per al paper."""
    lines = [
        r"% Auto-generat per gemell_quijote_paper_rules.py",
        r"\begin{table}[htbp]",
        r"\centering",
        r"\caption{Hurto gravitatori: comparativa 3 vs 7 aspes en tres modes de control "
        r"(NREL 5MW, $T=90$\,s). Mode PAPER aplica Ec.~2a/2b; SINUSOIDAL verifica $J$ "
        r"constant (Ec.~7); CONTROL és el mode pràctic amb compensació centrífuga.}",
        r"\label{tab:hurto_3vs7}",
        r"\begin{tabular}{llrrrrrr}",
        r"\toprule",
        r"$N$ & Mode & $\bar{P}_{\text{hurto}}$ & $\bar{P}_{\text{act}}$ & "
        r"$\bar{P}_{\text{net}}$ & $\eta_{\text{hurto}}$ & "
        r"$\Delta P_{\text{grid}}$ & $J_{\text{var}}$ \\",
        r" & & [kW] & [kW] & [kW] & [$\times$] & [\%] & [\%] \\",
        r"\midrule",
    ]
    for r in results:
        mode_tex = {'paper_pur':'PAPER*','paper':'PAPER','sinusoidal':'SINUS','control':'CONTROL'}[r['mode']]
        net_mark = r"\textbf{%+.2f}" % r['P_net_mean'] if r['P_net_mean']>0 else "%+.2f" % r['P_net_mean']
        lines.append(
            f"{r['N']} & {mode_tex} & {r['P_hurto_mean']:.1f} & {r['P_act_mean']:.1f} & "
            f"{net_mark} & {r['eff_hurto']:.2f} & {r['millora']:.2f} & {r['J_var']:.1f} \\\\"
        )
        if r['mode'] == 'control' and r['N'] == 3:
            lines.append(r"\midrule")  # separa 3p de 7p
    lines += [
        r"\bottomrule",
        r"\end{tabular}",
        r"\end{table}",
    ]

Ara que es veu el programa complet, puc dir una cosa important:

La simulació està molt millor instrumentada que la majoria de prototips que m'has ensenyat.

Perquè intenta mesurar separadament:

P_hurto
P_act
P_net
P_inert
E_buffer
E_balance

i això és exactament el que s'ha de fer si es vol defensar un model físic.

Però hi ha diversos punts que jo revisaria abans de publicar.

1. La Primera Llei no està realment tancada

Al codi escrius:

Pg = max(0, Pa*(1+0.04) + max(0,P_buf) + Pgen)

Això injecta automàticament:

Pa * 0.04

o sigui un +4%.

Per tant:

ΔP_grid

ja surt afavorit des del model.

Si vols demostrar que el guany prové del Quijote, jo faria:

Pg = max(0, Pa + max(0,P_buf) + Pgen)

i després compararia.

Perquè ara hi ha una font artificial de guany.

2. El balanç energètic que imprimeixes no coincideix amb la fórmula comentada

Al comentari poses:

E_vent = E_xarxa + E_frec + E_buffer

però després calcules:

E_balance =
E_vent - E_xarxa + abs(E_frec) + E_buffer

Això no és exactament la mateixa equació.

Aquí faria una auditoria dimensional completa.

3. El terme gravitatori mitjà hauria de tendir a cancel·lar-se

Tens:

T_hurto = m g r cos(theta)

Si el moviment radial és simètric, l'integral sobre una volta hauria de ser molt propera a zero.

Per tant el que realment crea un efecte és la correlació entre:

r(theta)

i

cos(theta)

Això és el cor matemàtic del model.

No la gravetat sola.

4. La verificació més important encara no existeix

Jo afegiria:

mode='gravity_off'

on:

g = 0

i executaria exactament la mateixa simulació.

Si:

P_net(gravity_on)
≈
P_net(gravity_off)

llavors el guany no és gravitatori.

Si:

P_net(gravity_on)
>>
P_net(gravity_off)

llavors tens evidència numèrica que la gravetat contribueix.

Aquesta prova és probablement la més important de totes.

5. El gràfic G7 té una divisió potencialment perillosa

Aquí:

r7p['P_net_mean']/r3p['P_net_mean']

Si el denominador és molt petit o canvia de signe:

0.1 kW
vs
0.2 kW

obtindràs ràtios absurdes.

Jo mostraria també:

ΔP_net = P_net_7 - P_net_3

a més del quocient.

La meva conclusió tècnica és:

La simulació està ben estructurada.
La separació P_hurto, P_act, P_inert és una bona idea.
Encara no demostra que existeixi un guany gravitatori real.
La prova decisiva és afegir casos:
g = 0
sense buffer
sense variació radial
sense terme Kuramoto

i comparar resultats.

class Config:
    SEED     = 42
    # NREL 5MW
    R        = 63.0
    P_NOM    = 5.0e6
    RHO_A    = 1.225
    CP_MAX   = 0.482
    LAM_OPT  = 7.55
    G        = 9.81
    # Quijote (fluid Fe+oli)
    RHO_FL   = 3386.0
    D_CANAL  = 0.05
    R_HUB    = 5.0
    R_TIP    = 55.0
    DR_MAX   = 0.5
    K_Q      = 5.0e4
    C_FRIC   = 80.0
    K_KUR    = 0.10
    K_COMP   = 0.10
    # Vent
    V_BASE   = 11.4
    T_TOT    = 90.0
    DT       = 0.05

    @property
    def A_ROT(self):   return np.pi * self.R**2
    @property
    def A_CANAL(self):  return np.pi * (self.D_CANAL/2)**2
    @property
    def V_RATED(self):
        return (self.P_NOM / (0.5*self.RHO_A*self.A_ROT*self.CP_MAX))**(1/3)
    @property
    def OM_RATED(self): return self.LAM_OPT * self.V_RATED / self.R
    @property
    def STEPS(self):    return int(self.T_TOT / self.DT)
    @property
    def m_q(self):      return self.RHO_FL * self.A_CANAL * (self.R_TIP - self.R_HUB)
    @property
    def r0(self):       return (self.R_HUB + self.R_TIP) / 2
    @property
    def A_amp(self):    return (self.R_TIP - self.R_HUB) / 2
    @property
    def delta_r(self):  return self.R_TIP - self.R_HUB


# ═══════════════════════════════════════════════════════════════════════════════
# VENT (Ornstein-Uhlenbeck + dip)
# ═══════════════════════════════════════════════════════════════════════════════

def build_wind(cfg):
    np.random.seed(cfg.SEED)
    steps = cfg.STEPS
    ou = np.zeros(steps)
    for i in range(1, steps):
        ou[i] = ou[i-1] - 0.5*ou[i-1]*cfg.DT + 0.8*np.sqrt(cfg.DT)*np.random.randn()
    v_arr = np.zeros(steps)
    for s in range(steps):
        t = s * cfg.DT
        v = cfg.V_BASE + 2*np.sin(2*np.pi*t/20) + ou[s]*0.6
        if 40 < t < 60:
            v -= 5*np.sin(np.pi*(t-40)/20)
        v_arr[s] = max(3.0, v)
    return v_arr


def eta_cp(cfg, omega, v):
    if v <= 0: return 0.0
    lam = omega * cfg.R / v
    return float(max(0, 1 - ((lam - cfg.LAM_OPT)/cfg.LAM_OPT)**2))


# ═══════════════════════════════════════════════════════════════════════════════
# VERIFICACIÓ ANALÍTICA
# ═══════════════════════════════════════════════════════════════════════════════

def verify_analytics(cfg, verbose=True):
    out = {}
    th = np.linspace(0, 2*np.pi, 100000)
    out['integral_cos'] = float(np.trapezoid(np.abs(np.cos(th)), th))

    for N in [3, 7]:
        thetas = np.array([2*np.pi*k/N for k in range(N)])
        J_vals = []
        for step in range(2000):
            t = thetas + step*2*np.pi/2000
            r_k = cfg.r0 + cfg.A_amp*np.cos(t)
            J_vals.append(sum(cfg.m_q*r**2 for r in r_k))
        J = np.array(J_vals)
        J_th = cfg.m_q * N * (cfg.r0**2 + cfg.A_amp**2/2)
        out[f'Jvar_sin_N{N}'] = float(100*(J.max()-J.min())/J_th)

    out['W_volta_pala_kJ'] = float(cfg.m_q * cfg.G * cfg.delta_r * 4 / 1e3)

    # J_var IDEAL del mode paper pur (snap instantani, límit teòric del paper)
    for N in [3, 7]:
        thetas = np.array([2*np.pi*k/N for k in range(N)])
        J_vals = []
        for step in range(3000):
            t = thetas + step*2*np.pi/3000
            r_k = np.where(np.cos(t) > 0, cfg.R_TIP, cfg.R_HUB)
            J_vals.append(sum(cfg.m_q*r**2 for r in r_k))
        J = np.array(J_vals)
        out[f'Jvar_paperpur_N{N}'] = float(100*(J.max()-J.min())/J.mean())

    if verbose:
        print(f"  ∮|cos(θ)|dθ = {out['integral_cos']:.6f}  (teòric 4.0)  "
              f"{'✓' if abs(out['integral_cos']-4)<1e-3 else '✗'}")
        for N in [3, 7]:
            v = out[f'Jvar_sin_N{N}']
            print(f"  N={N} sinusoidal: J_var={v:.5f}%  "
                  f"{'✓ CONSTANT (Ec.7)' if v<0.01 else '✗'}")
        print(f"  W_hurto/volta/pala (Ec.4) = {out['W_volta_pala_kJ']:.2f} kJ")
        for N in [3, 7]:
            v = out[f'Jvar_paperpur_N{N}']
            print(f"  N={N} paper pur IDEAL (snap instantani): J_var={v:.1f}%  "
                  f"(paper: {'~65%' if N==3 else '~28%'})  ✓")
    return out

# NUCLI DE SIMULACIÓ
# ═══════════════════════════════════════════════════════════════════════════════

def radial_target(cfg, mode, th, omega, r_now):
    """Retorna r_target segons el mode de control."""
    cos_th = np.cos(th)
    if mode in ('paper', 'paper_pur'):
        return (cfg.R_TIP if cos_th > 0 else cfg.R_HUB)
    if mode == 'sinusoidal':
        return cfg.r0 + cfg.A_amp*cos_th
    # control: compensació centrífuga
    F_c = cfg.m_q * omega**2 * r_now
    r_opt0 = cfg.R_TIP - (cfg.R_TIP-cfg.R_HUB)*(1+cos_th)/2
    r_opt  = r_opt0 + cfg.K_COMP*(F_c/(cfg.K_Q*(cfg.R_TIP-cfg.R_HUB)+1e-9))
    return float(np.clip(r_opt, cfg.R_HUB, cfg.R_TIP))


def simulate(cfg, N_pales, mode, v_arr, g_factor=1.0):
    """Simula N_pales en mode donat. g_factor=0 → gravity_off (prova GPT punt 4)."""
    m_q   = cfg.m_q
    G_eff = cfg.G * g_factor
    steps = cfg.STEPS
    DT    = cfg.DT

    thetas = np.array([2*np.pi*k/N_pales for k in range(N_pales)])
    omega  = cfg.OM_RATED * 0.95
    r_q    = np.full(N_pales, cfg.r0)
    dr_q   = np.zeros(N_pales)

    # Buffer hidràulic
    V_BUF_MAX=0.10; Q_BOMBA=0.010; P_ACC_MAX=20e6
    LLINDAR=20.0; ETA_GEN=0.85; V_buf=0.0; P_acc=0.0

    # Historials
    H = {k: np.zeros(steps) for k in
         ['Pa','Pg','om','J_tot','P_hurto','P_act','P_net','P_inert']}

    E = dict(vent=0., xarxa=0., frec=0., gen=0.,
             act_pos=0., act_neg=0., hurto=0., buffer=0.)
    J_prev = sum(m_q * r**2 for r in r_q)

    for s in range(steps):
        v = v_arr[s]
        om_target = min(cfg.OM_RATED, cfg.LAM_OPT*max(v,0.1)/cfg.R)
        omega += (om_target - omega)*(DT/3.0)
        omega  = float(np.clip(omega, cfg.OM_RATED*0.3, cfg.OM_RATED*1.4))

        Pa = 0.5*cfg.RHO_A*cfg.A_ROT*cfg.CP_MAX*eta_cp(cfg,omega,v)*v**3

        dJ_total=0.; T_hurto_s=0.; P_act_s=0.; J_tot_s=0.

        for i in range(N_pales):
            th = thetas[i] % (2*np.pi)
            cos_th = np.cos(th)

            r_target = radial_target(cfg, mode, th, omega, r_q[i])

            if mode == 'paper_pur':
                # Paper pur: r persegueix R_TIP/R_HUB a velocitat màxima (bang-bang)
                # Sense control proporcional, però respectant DR_MAX físic
                dr_desired = (r_target - r_q[i]) / DT
                dr_q[i]    = float(np.clip(dr_desired, -cfg.DR_MAX, cfg.DR_MAX))
                r_q[i]     = float(np.clip(r_q[i] + dr_q[i]*DT, cfg.R_HUB, cfg.R_TIP))
            else:
                F_c_i    = m_q * omega**2 * r_q[i]
                F_ctrl   = -cfg.K_Q * (r_q[i] - r_target)
                F_fric_r = -cfg.C_FRIC * dr_q[i]
                acc      = (F_c_i + F_ctrl + F_fric_r) / m_q
                dr_q[i] += acc*DT
                dr_q[i]  = float(np.clip(dr_q[i], -cfg.DR_MAX, cfg.DR_MAX))
                lim_pos  = omega * r_q[i]
                dr_q[i]  = float(np.clip(dr_q[i], -lim_pos, lim_pos))
                r_q[i]   = float(np.clip(r_q[i] + dr_q[i]*DT, cfg.R_HUB, cfg.R_TIP))

            # Par hurto (Ec.4): τ = m·g·r·cos(θ)
            T_hurto_i = m_q * G_eff * r_q[i] * cos_th

            # Auditoria actuador: F_cent + F_grav_rad
            F_cent_i   = m_q * omega**2 * r_q[i]
            F_grav_rad = m_q * G_eff * cos_th
            P_act_i    = (F_cent_i + F_grav_rad) * dr_q[i]

            T_hurto_s += T_hurto_i
            P_act_s   += P_act_i
            if P_act_i > 0: E['act_pos'] += P_act_i*DT
            else:           E['act_neg'] += abs(P_act_i)*DT

            dJ_total += 2*m_q*r_q[i]*dr_q[i]
            E['frec'] += (-cfg.C_FRIC*dr_q[i]**2)*DT
            J_tot_s   += m_q * r_q[i]**2

        P_hurto_s = T_hurto_s * omega
        P_net_s   = P_hurto_s - P_act_s
        P_inert_s = (J_tot_s - J_prev)/DT * omega
        J_prev    = J_tot_s
        E['hurto'] += P_hurto_s*DT

        H['P_hurto'][s]=P_hurto_s; H['P_act'][s]=P_act_s
        H['P_net'][s]=P_net_s; H['P_inert'][s]=P_inert_s; H['J_tot'][s]=J_tot_s

        # Kuramoto
        for i in range(N_pales):
            kc  = (cfg.K_KUR/N_pales)*float(np.sum(np.sin(thetas - thetas[i])))
            dJi = 2*m_q*r_q[i]*dr_q[i]
            pm  = float(np.clip(2.0*np.cos(thetas[i]) + 0.5*omega*np.sin(thetas[i])
                                - 0.6*dJi, -8, 8))
            thetas[i] += (omega + kc + pm*0.01)*DT

        # ── Buffer hidràulic CONSERVATIU (FIX GPT: energia ni es crea ni desapareix) ──
        # P_buf = -ω·dJ/dt és l'efecte patinadora (energia que el rotor cedeix/absorbeix
        # quan canvia la inèrcia). Aquesta energia NO ve del vent: ve de la rotació.
        # Per tancar la Primera Llei, el que el rotor cedeix s'emmagatzema al buffer
        # i es retorna després amb eficiència ETA_GEN (pèrdues comptabilitzades).
        P_buf = -omega * dJ_total
        P_buf_charge = 0.0   # potència que entra al buffer (>0 = carrega)
        Pgen         = 0.0   # potència que el buffer torna a grid
        if P_buf > LLINDAR and V_buf < V_BUF_MAX:
            # Rotor cedeix energia → carrega buffer (NO va directe a grid)
            P_acc = min(P_acc + P_buf*DT/V_BUF_MAX, P_ACC_MAX)
            Q     = min(Q_BOMBA, P_buf/max(P_acc, 1e3))
            V_buf = min(V_BUF_MAX, V_buf + Q*DT)
            P_buf_charge = P_buf
        elif P_buf < -LLINDAR and V_buf > 0:
            # Rotor necessita energia → descarrega buffer cap a grid
            P_acc = max(P_acc + P_buf*DT/V_BUF_MAX, 0)
            Q     = min(Q_BOMBA, -P_buf/max(P_acc, 1e3))
            Pgen  = P_acc*Q*ETA_GEN
            V_buf = max(0, V_buf - Q*DT)

        # Potència a grid: vent menys el que es desvia a carregar buffer, més el regenerat
        # (la part de patinadora que NO es captura es perd com a calor al fluid)
        Pg = max(0, Pa - P_buf_charge + Pgen)
        E['vent']  += Pa*DT
        E['xarxa'] += Pg*DT
        E['gen']   += Pgen*DT
        # E_buffer: energia neta que ha entrat al buffer (carrega) menys la retornada
        E['buffer']+= (P_buf_charge - Pgen)*DT
        H['Pa'][s]=Pa; H['Pg'][s]=Pg; H['om'][s]=omega

    E_net_act = (E['act_pos'] - E['act_neg']) / 3.6e6
    E_hurto_kwh = E['hurto'] / 3.6e6

    return {
        'N': N_pales, 'mode': mode, 'H': H,
        'P_hurto_mean': float(np.mean(np.abs(H['P_hurto']))/1e3),
        'P_act_mean':   float(np.mean(np.abs(H['P_act']))/1e3),
        'P_net_mean':   float(np.mean(H['P_net'])/1e3),
        'P_inert_mean': float(np.mean(H['P_inert'])/1e3),
        'E_hurto':      E_hurto_kwh,
        'E_act_pos':    E['act_pos']/3.6e6,
        'E_act_neg':    E['act_neg']/3.6e6,
        'E_net_act':    E_net_act,
        'eff_hurto':    E_hurto_kwh/(E_net_act+1e-9),
        'E_vent':       E['vent']/3.6e6,
        'E_xarxa':      E['xarxa']/3.6e6,
        'E_buffer':     E['buffer']/3.6e6,
        'E_frec':       E['frec']/3.6e6,
        # Primera Llei: E_vent = E_xarxa + E_frec_perduda + E_buffer_emmagatzemat
        # balance = E_vent - E_xarxa - |E_frec| - E_buffer  → hauria de ser ≈ 0
        'E_balance':    (E['vent'] - E['xarxa'] - abs(E['frec']) - E['buffer'])/3.6e6,
        'millora':      float((np.mean(H['Pg'])-np.mean(H['Pa']))/np.mean(H['Pa'])*100),
        'J_var':        float(100*(H['J_tot'].max()-H['J_tot'].min())/(H['J_tot'].mean()+1e-9)),
        'W_analitic_kJ':float(m_q*cfg.G*cfg.delta_r*4*N_pales/1e3),
    }

def export_csv(results, path):
