import math

def newton_inflexion(J_actual, solucion, base, historial, tol, max_iter, iter_count=0):
    # Incrementar contador de iteraciones
    iter_count += 1
    
    # 1. Criterio de parada con J*
    J_estrella = (solucion - (base**J_actual)/solucion + 1) / solucion
    if abs(1 - J_estrella) < tol:
        print(f"✅ Solución encontrada en iteración {iter_count}: J = {J_actual:.12f}, J* = {J_estrella:.12f}")
        return J_actual, iter_count
    
    if iter_count > max_iter:
        print(f"⚠️ Límite de iteraciones alcanzado. Último J = {J_actual:.6f}, J* = {J_estrella:.6f}")
        return J_actual, iter_count
    
    # 2. Si no hay 3 medidas, dar paso de Newton
    if len(historial) < 3:
        f = base**J_actual - solucion
        df = base**J_actual * math.log(base)
        if abs(df) < 1e-15:
            raise ValueError(f"Derivada cero en J = {J_actual:.6f}")
        J_siguiente = J_actual - f/df
        return newton_inflexion(J_siguiente, solucion, base, [J_siguiente, J_actual], tol, max_iter, iter_count)
    
    # 3. Extraer las 3 J
    J = historial[0]
    J2 = historial[1]
    J3 = historial[2]
    
    paso_actual = abs(J - J2)
    paso_anterior = abs(J2 - J3)
    
    # 4. Detección de inflexión: si el paso se encoge, hemos pasado la cima
    if paso_actual < paso_anterior:
        J_nuevo = (J + J2) / 2
        print(f"🪡 Inflexión en iter {iter_count}: paso_act={paso_actual:.4e} < paso_ant={paso_anterior:.4e}. Mitadeamos: J={J:.6f}, J2={J2:.6f} → J_o={J_nuevo:.6f}")
        # Reiniciamos historial
        return newton_inflexion(J_nuevo, solucion, base, [J_nuevo], tol, max_iter, iter_count)
    else:
        # Newton normal
        f = base**J - solucion
        df = base**J * math.log(base)
        if abs(df) < 1e-15:
            raise ValueError(f"Derivada cero en J = {J:.6f}")
        J_siguiente = J - f/df
        print(f"Iter {iter_count}: Newton normal. J={J:.6f} → J_sig={J_siguiente:.6f} (paso_act={paso_actual:.4e}, paso_ant={paso_anterior:.4e})")
        return newton_inflexion(J_siguiente, solucion, base, [J_siguiente, J, J2], tol, max_iter, iter_count)

    % (solo los que existan; si hay menos de 4, se va completando)
    
    persistent iter_count;
    if isempty(iter_count)
        iter_count = 0;
    end
    iter_count = iter_count + 1;
    
    % 1. Medidor J*
    J_estrella = (solucion - (base^J_actual)/solucion + 1) / solucion;
    if abs(1 - J_estrella) < tol

        iter_count = 0;
        return;
    end
    
    if iter_count > max_iter
        warning('Máx iteraciones alcanzado. J=%.6f', J_actual);
        J_final = J_actual;
        iter_count = 0;
        return;
    end
    
    % 2. Si no hay suficientes puntos, damos pasos de Newton para llenar historial
    if length(historial) < 4

    paso_antiguo  = abs(J3 - J4);

    if paso_actual < paso_anterior && paso_anterior >= paso_antiguo

def newton_inflexion4(J_actual, solucion, base, historial, tol, max_iter, iter_count=0):
    iter_count += 1
    
    J_estrella = (solucion - (base**J_actual)/solucion + 1) / solucion
    if abs(1 - J_estrella) < tol:
        print(f"✅ Solución en iter {iter_count}: J={J_actual:.12f}, J*={J_estrella:.12f}")
        return J_actual, iter_count
    
    if iter_count > max_iter:
        print(f"⚠️ Máx iteraciones. Último J={J_actual:.6f}")
        return J_actual, iter_count
    
    # Completar historial si es necesario
    if len(historial) < 4:
        f = base**J_actual - solucion
        df = base**J_actual * math.log(base)
        if abs(df) < 1e-15:
            raise ValueError("Derivada cero")
        J_siguiente = J_actual - f/df
        nuevo_historial = [J_siguiente] + historial
        return newton_inflexion4(J_siguiente, solucion, base, nuevo_historial, tol, max_iter, iter_count)
    
    # Extraer los 4 puntos
    J, J2, J3, J4 = historial[0], historial[1], historial[2], historial[3]

    paso_anterior = abs(J2 - J3)
    paso_antiguo = abs(J3 - J4)

    if paso_actual < paso_anterior and paso_anterior >= paso_antiguo:
        J_nuevo = (J3 + J2) / 2
        print(f"🪡 Primer encogido en iter {iter_count}: paso_act={paso_actual:.4e} < paso_ant={paso_anterior:.4e} y paso_ant={paso_anterior:.4e} >= paso_antiguo={paso_antiguo:.4e}. Mitadeamos entre J3={J3:.6f}, J2={J2:.6f} → J_o={J_nuevo:.6f}")

        return newton_inflexion4(J_nuevo, solucion, base, [J_nuevo], tol, max_iter, iter_count)

        # Newton normal
        f = base**J - solucion
        df = base**J * math.log(base)
        if abs(df) < 1e-15:
            raise ValueError("Derivada cero")

function J_final = newton_inflexion4(J_actual, solucion, base, historial, tol, max_iter)
    % historial = [J_actual, J_anterior, J_anterior2, J_anterior3]
    
    persistent iter_count;
    if isempty(iter_count)
        iter_count = 0;
    end
    iter_count = iter_count + 1;
    
    % Criterio de parada
    J_estrella = (solucion - (base^J_actual)/solucion + 1) / solucion;
    if abs(1 - J_estrella) < tol
        fprintf('✅ Solución en iter %d: J = %.12f, J* = %.12f\n', iter_count, J_actual, J_estrella);
        J_final = J_actual;
        iter_count = 0;
        return;
    end
    
    if iter_count > max_iter
        warning('Máximo de iteraciones. J = %.6f', J_actual);
        J_final = J_actual;
        iter_count = 0;
        return;
    end
    
    % Completar historial
    if length(historial) < 4
        f = base^J_actual - solucion;
        df = base^J_actual * log(base);
        if abs(df) < 1e-15
            error('Derivada cero en J = %.6f', J_actual);
        end
        J_siguiente = J_actual - f/df;
        nuevo_historial = [J_siguiente, historial];
        J_final = newton_inflexion4(J_siguiente, solucion, base, nuevo_historial, tol, max_iter);
        return;
    end
    
    % Extraer 4 puntos
    J = historial(1);
    J2 = historial(2);
    J3 = historial(3);
    J4 = historial(4);
    
    paso_actual = abs(J - J2);
    paso_anterior = abs(J2 - J3);
    paso_antiguo = abs(J3 - J4);
    
    % Detectar primer encogido: p3 < p2 y p2 >= p1
    if paso_actual < paso_anterior && paso_anterior >= paso_antiguo
        % Media entre el anterior y el actual (J2 y J)
        J_nuevo = (J2 + J) / 2;
        fprintf('🪡 Primer encogido en iter %d: paso_act=%.4e < paso_ant=%.4e (y p_ant=%.4e >= p_antiguo=%.4e)\n', ...
                iter_count, paso_actual, paso_anterior, paso_anterior, paso_antiguo);
        fprintf('   Mitadeamos entre J2=%.6f y J=%.6f → J_o=%.6f\n', J2, J, J_nuevo);
        % Reiniciamos historial con J_nuevo
        J_final = newton_inflexion4(J_nuevo, solucion, base, [J_nuevo], tol, max_iter);
        return;
    else
        % Newton normal
        f = base^J - solucion;
        df = base^J * log(base);
        if abs(df) < 1e-15
            error('Derivada cero en J = %.6f', J);
        end
        J_siguiente = J - f/df;
        nuevo_historial = [J_siguiente, J, J2, J3];
        fprintf('Iter %d: Newton. J=%.6f → J_sig=%.6f\n', iter_count, J, J_siguiente);
        J_final = newton_inflexion4(J_siguiente, solucion, base, nuevo_historial, tol, max_iter);
    end
end

    %   y freno de emergencia (step bounding) para evitar overflow.
    if nargin < 7
        iter_actual = 0;
    end

    if iter_actual > max_iter
        J_final = J;
        iter = iter_actual;
        exito = false;
        fprintf('⚠️  Máximo de iteraciones alcanzado.\n');
        return;
    end

    try
        J_estrella = (sol - (base^J)/sol + 1) / sol;
    catch ME
        if strcmp(ME.identifier, 'MATLAB:ArithmeticOverflow')
            fprintf('    [!] Overflow evitado en iter %d evaluando base^J.\n', iter_actual);
            J_final = J;
            iter = iter_actual;
            exito = false;
            return;
        else
            rethrow(ME);
        end
    end

    if abs(1 - J_estrella) < tol
        J_final = J;
        iter = iter_actual;
        exito = true;
        return;
    end

    % --- Llenar el historial (hasta tener 4) ---
    if length(historial) < 4
        try
            f = base^J - sol;
            df = (base^J) * log(base);
            paso = f/df;
            % 🛡️ FRENO DE EMERGENCIA (Step Bounding)
            if abs(paso) > 50.0
                paso = 50.0 * sign(paso);

            J_sig = J - paso;

function [J_final, iter, exito] = newton_estandar(J, sol, base, tol, max_iter)

                return;

function [J_final, iter, exito] = newton_rapido(J, sol, base, historial, tol, max_iter, L, rango, iter_actual)
    % Parámetros por defecto
    if nargin < 9, iter_actual = 0; end
    if nargin < 8, rango = 100; end
    if nargin < 7, L = 50; end

    % Clipping y redondeo inicial
    J = clip(J, rango);
    J = round(J, 12);

    % Límite de iteraciones
    if iter_actual > max_iter
        J_final = J;
        iter = iter_actual;
        exito = false;
        fprintf('⚠️  Máximo de iteraciones alcanzado.\n');
        return;
    end

    % Criterio de parada por J*
    try
        J_estrella = (sol - (base^J)/sol + 1) / sol;
    catch ME
        if strcmp(ME.identifier, 'MATLAB:ArithmeticOverflow')
            fprintf('    [!] Overflow evitado en iter %d evaluando base^J.\n', iter_actual);
            J_final = J;
            iter = iter_actual;
            exito = false;
            return;
        else
            rethrow(ME);
        end
    end

    if abs(1 - J_estrella) < tol
        J_final = J;
        iter = iter_actual;
        exito = true;
        return;
    end

    % --- Completar historial hasta tener 4 puntos ---
    if length(historial) < 4
        try
            f = base^J - sol;
            df = (base^J) * log(base);
            paso = f / df;
            % Step Bounding
            if abs(paso) > L
                paso = L * sign(paso);
            end
            J_sig = J - paso;
            J_sig = clip(J_sig, rango);
            J_sig = round(J_sig, 12);
        catch ME
            if strcmp(ME.identifier, 'MATLAB:ArithmeticOverflow') || strcmp(ME.identifier, 'MATLAB:divideByZero')
                fprintf('    [!] Falla matemática en cálculo de Newton.\n');
                J_final = J;
                iter = iter_actual;
                exito = false;
                return;
            else
                rethrow(ME);
            end
        end
        [J_final, iter, exito] = newton_rapido(J_sig, sol, base, [J_sig, historial], tol, max_iter, L, rango, iter_actual+1);
        return;
    end

    % --- Ya tenemos 4 puntos: extraer ---
    J_curr = historial(1);
    J2 = historial(2);
    J3 = historial(3);
    J4 = historial(4);

    p3 = abs(J_curr - J2);
    p2 = abs(J2 - J3);
    p1 = abs(J3 - J4);

    % --- Detección del primer encogido ---
    if p3 < p2 && p2 >= p1
        J_nuevo = (J2 + J_curr) / 2.0;
        J_nuevo = clip(J_nuevo, rango);
        J_nuevo = round(J_nuevo, 12);
        fprintf('    🪡 Inflexión detectada. Mitadeando J2=%.4f y J=%.4f -> J_nuevo=%.4f\n', J2, J_curr, J_nuevo);
        % Reiniciar historial
        [J_final, iter, exito] = newton_rapido(J_nuevo, sol, base, [J_nuevo], tol, max_iter, L, rango, iter_actual+1);
        return;
    else
        % --- Newton normal con step bounding ---
        try
            f = base^J_curr - sol;
            df = (base^J_curr) * log(base);
            paso = f / df;
            if abs(paso) > L
                paso = L * sign(paso);
            end
            J_sig = J_curr - paso;
            J_sig = clip(J_sig, rango);
            J_sig = round(J_sig, 12);
        catch ME
            if strcmp(ME.identifier, 'MATLAB:ArithmeticOverflow') || strcmp(ME.identifier, 'MATLAB:divideByZero')
                fprintf('    [!] Overflow matemático durante el flujo normal.\n');
                J_final = J_curr;
                iter = iter_actual;
                exito = false;
                return;
            else
                rethrow(ME);
            end
        end
        [J_final, iter, exito] = newton_rapido(J_sig, sol, base, [J_sig, J_curr, J2, J3], tol, max_iter, L, rango, iter_actual+1);
        return;
    end
end

function [J_final, iter, exito] = newton_estandar(J, sol, base, tol, max_iter, L, rango)
    % Newton estándar con clipping, redondeo y step bounding
    if nargin < 4, tol = 1e-12; end
    if nargin < 5, max_iter = 200; end
    if nargin < 6, L = 50; end
    if nargin < 7, rango = 100; end

    J = clip(J, rango);
    J = round(J, 12);

    for iter = 0:max_iter
        try
            J_estrella = (sol - (base^J)/sol + 1) / sol;
            if abs(1 - J_estrella) < tol
                J_final = J;
                exito = true;
                return;
            end
            f = base^J - sol;
            df = (base^J) * log(base);
            paso = f / df;
            if abs(paso) > L
                paso = L * sign(paso);
            end
            J = J - paso;
            J = clip(J, rango);
            J = round(J, 12);
        catch ME
            if strcmp(ME.identifier, 'MATLAB:ArithmeticOverflow') || strcmp(ME.identifier, 'MATLAB:divideByZero')
                J_final = J;
                exito = false;
                return;
            else
                rethrow(ME);
            end
        end
    end
    J_final = J;
    exito = false;
end

1. Sobre la "recursividad con parámetro J cambiante"

Sí, es muy típico y elegante. En programación funcional y en métodos numéricos adaptativos, esto se llama "Tail Recursion with State Reset" (recursión de cola con reinicio de estado).

2. Analizando tu desigualdad: j3-j4 > j2-j3 > j2-j > j-j2

Primero, corrijamos un pequeño lapsus matemático: |j2 - j| y |j - j2| son exactamente lo mismo. Seguramente quisiste decir la secuencia de pasos absolutos:

Conclusión: Cuando los pasos ya están encogiendo de forma monótona (2.0 > 0.5 > 0.2), lo peor que puedes hacer es intervenir. El método de Newton ya está "en racha" y convergerá en 1 o 2 pasos. Si "doblas" (haces media) en este momento, estropeas la convergencia cuadrática y alargas el proceso.

3. Entonces, ¿cuándo se debe "doblar" realmente?

La regla de oro que has descubierto es detectar el PRIMER ENCOGIDO, que es cuando el paso actual se encoge por primera vez después de haber estado creciendo.

Es decir:

Fase 1: Pasos creciendo → |J4-J3| < |J3-J2| (Newton se está disparando).

Fase 2: Primer encogido → |J2-J| < |J3-J2| Y además |J3-J2| >= |J4-J3| (el paso anterior no era un encogido).

Justo en ese instante, el punto de inflexión está entre J2 y J3 (o entre J2 y J). Ahí es donde sí vale la pena hacer la media, y la mejor opción (como discutimos) es la media entre J2 y J (los dos más recientes) porque J ya está entrando en la zona buena.

4. Mi opinión final sobre tu lógica

La recursividad con parámetro cambiante es una joya. Úsala sin miedo.

El criterio de "pasos encogiendo" es perfecto para saber que estás cerca, pero NO para decidir doblar. Doblado solo debe ocurrir en el primer encogido (cuando el paso anterior era más grande o igual que el antiguo).

Si ves la secuencia |J4-J3| > |J3-J2| > |J2-J| (decrecimiento estricto), no toques nada, deja que Newton termine el trabajo.

def newton_rapido(J, sol, base, historial, tol, max_iter, iter_actual=0):
    # --- Parada de seguridad ---
    if iter_actual > max_iter:
        return J, iter_actual, False

    # --- Criterio de parada por J* ---
    try:
        J_estrella = (sol - (base**J)/sol + 1) / sol
    except OverflowError:
        print(f"    [!] Overflow evitado en iter {iter_actual} evaluando base^J.")
        return J, iter_actual, False

    if abs(1 - J_estrella) < tol:
        return J, iter_actual, True

    # --- Llenar el historial (hasta tener 4) ---
    if len(historial) < 4:
        try:
            f = base**J - sol
            df = (base**J) * math.log(base)
            J_sig = J - f/df
        except (OverflowError, ZeroDivisionError):
            print("    [!] Falla matemática en cálculo de Newton.")
            return J, iter_actual, False
            
        return newton_rapido(J_sig, sol, base, [J_sig] + historial, tol, max_iter, iter_actual + 1)

    # --- Extracción de historial ---
    J_curr, J2, J3, J4 = historial[0], historial[1], historial[2], historial[3]

    p2 = abs(J2 - J3)

Voy a modificar el código para incluir dos mejoras clave:

2. **Redondeo (truncamiento de decimales):** Redondear `J` a un número fijo de decimales (por ejemplo, 12) después de cada paso para evitar la acumulación de errores numéricos y que las comparaciones de pasos (`p3 < p2`) sean más estables.



Estas técnicas son comunes en implementaciones numéricas robustas.

            df = (base**J_curr) * math.log(base)

            J_sig = clip(J_sig)

            J_sig = round(J_sig, 12)

        except OverflowError:

            print("    [!] Overflow matemático durante el flujo normal.")

            return J_curr, iter_actual, False

        return newton_rapido(J_sig, sol, base, [J_sig, J_curr, J2, J3], tol, max_iter, iter_actual + 1)



def newton_estandar(J, sol, base, tol, max_iter):

    J = clip(J)

    J = round(J, 12)

            J = J - f/df

            J = clip(J)

            J = round(J, 12)

        except OverflowError:

            return None, i, False

1. El peligro del clip absoluto (El Muro Invisible)
Al usar clip(J, -100, 100), estás limitando el espacio de búsqueda. Esto evita el overflow, sí, pero crea un problema crítico: ¿qué pasa si la solución real de la ecuación es J=150?

Imagina este escenario:

Tu algoritmo va subiendo y llega a J=90.

El siguiente paso de Newton le dice: "La raíz está más arriba, salta a 130".

Tu función clip interviene: "No puedes pasar de 100. Te dejo en 100".

En la siguiente iteración, Newton evalúa en 100 y vuelve a decir: "Sigue subiendo, salta a 150".

El clip lo vuelve a bajar a 100.

El algoritmo se quedará atrapado en un bucle infinito chocando contra el "muro" de 100, y fallará por alcanzar el límite de iteraciones.

La solución de ingeniería: En lugar de limitar la posición máxima (clip absoluto), es matemáticamente superior limitar la velocidad máxima (Step Bounding, limitar el ΔJ). Es la diferencia entre poner un muro de ladrillos en el kilómetro 100, o poner un límite de velocidad de 50 km/h: con la segunda opción, el coche tardará más, pero podrá llegar al kilómetro 150 sin estrellarse.

2. El conflicto entre round y tol (Ruido Numérico)
Has añadido round(J, 12) para limpiar los decimales, pero tu tolerancia es tol = 1e-12. Hacer ambas cosas a la vez es muy peligroso en análisis numérico.

El estándar de coma flotante de 64 bits (IEEE 754) que usan Python y MATLAB ya te da unos 15-17 decimales de precisión de forma nativa. Al forzar un redondeo a 12 decimales en cada iteración intermedia:

Estás destruyendo la información microscópica que Newton necesita para calcular gradientes precisos cuando está muy cerca de la raíz.

Puedes provocar una oscilación de truncamiento: el algoritmo intenta dar un paso minúsculo de 10 
−13

La regla de oro en métodos numéricos es: nunca redondees las variables de estado intermedias. Deja que la máquina use toda su memoria de 64 bits para iterar, y aplica el round únicamente al J_final justo en el momento del return o del fprintf para mostrarlo bonito en pantalla.

import math

def newton_rapido(J, sol, base, historial, tol, max_iter, iter_actual=0):
    if iter_actual > max_iter:
        return J, iter_actual, False

    try:
        # Criterio de parada por J*
        J_estrella = (sol - (base**J)/sol + 1) / sol
    except OverflowError:
        return J, iter_actual, False

    if abs(1 - J_estrella) < tol:
        return J, iter_actual, True

    if len(historial) < 4:

            f = base**J - sol
            df = (base**J) * math.log(base)
            paso = f/df

            if abs(paso) > 5.0:
                paso = 5.0 * (1 if paso > 0 else -1)
                
            J_sig = J - paso
        except (OverflowError, ZeroDivisionError):

    p2 = abs(J2 - J3)
    p1 = abs(J3 - J4)

import math



def newton_rapido(J, sol, base, historial, tol, max_iter, iter_actual=0, modo_media='J4_J'):

    """

    modo_media puede ser:

        'J3_J2' : media entre J3 y J2 (original)

        'J2_J'  : media entre J2 y J

        'J4_J'  : media entre J4 y J (nueva propuesta)

        'J3_J'  : media entre J3 y J

    """

    if iter_actual > max_iter:

        return J, iter_actual, False



    try:

        J_estrella = (sol - (base**J)/sol + 1) / sol

    except OverflowError:

        return J, iter_actual, False



    if abs(1 - J_estrella) < tol:

        return J, iter_actual, True



    # --- Llenar el historial (hasta tener 4) ---

    if len(historial) < 4:

        f = base**J - sol

        df = (base**J) * math.log(base)

        paso = f/df

        

        # 🛡️ FRENO DE EMERGENCIA (Step Bounding)

        if abs(paso) > 50.0:

            paso = 50.0 * (1 if paso > 0 else -1)

            

        J_sig = J - paso

        return newton_rapido(J_sig, sol, base, [J_sig] + historial, tol, max_iter, iter_actual + 1, modo_media)



    # --- Extracción de historial ---

    J_curr, J2, J3, J4 = historial[0], historial[1], historial[2], historial[3]

    p3 = abs(J_curr - J2)

    p2 = abs(J2 - J3)

    p1 = abs(J3 - J4)



    # --- DECISIÓN CLAVE: Detección del primer encogido ---

    if p3 < p2 and p2 >= p1:

        # Elegir el modo de media

        if modo_media == 'J3_J2':

            J_nuevo = (J3 + J2) / 2.0

            desc = f"media entre J3={J3:.4f} y J2={J2:.4f}"

        elif modo_media == 'J2_J':

            J_nuevo = (J2 + J_curr) / 2.0

            desc = f"media entre J2={J2:.4f} y J={J_curr:.4f}"

        elif modo_media == 'J4_J':

            J_nuevo = (J4 + J_curr) / 2.0

            desc = f"media entre J4={J4:.4f} y J={J_curr:.4f}"

        elif modo_media == 'J3_J':

            J_nuevo = (J3 + J_curr) / 2.0

            desc = f"media entre J3={J3:.4f} y J={J_curr:.4f}"

        else:

            J_nuevo = (J2 + J_curr) / 2.0  # fallback

            desc = f"media por defecto entre J2={J2:.4f} y J={J_curr:.4f}"



        print(f"    🪡 Inflexión detectada. {desc} -> J_nuevo={J_nuevo:.4f}")

        # Llamada recursiva reiniciando el historial

        return newton_rapido(J_nuevo, sol, base, [J_nuevo], tol, max_iter, iter_actual + 1, modo_media)

    else:

        # --- Newton fluye sin intervenir ---

        f = base**J_curr - sol

def newton_estandar(J, sol, base, tol, max_iter):

    for i in range(max_iter):

        try:

            J_estrella = (sol - (base**J)/sol + 1) / sol

            if abs(1 - J_estrella) < tol:

                return J, i, True

                

            f = base**J - sol

            df = (base**J) * math.log(base)

            paso = f/df

            

            if abs(paso) > 50.0:

                paso = 50.0 * (1 if paso > 0 else -1)

                

            J = J - paso

        except OverflowError:

            return None, i, False

    return J, max_iter, False

Este mecanismo actúa como un filtro de doble amortiguación. Evita por completo los saltos en rampa cuando la derivada cambia de signo bruscamente, obligando al algoritmo a converger de forma monótona y sin oscilaciones parásitas hacia el valor exacto de la raíz. Es una evolución excelente para estabilizar métodos numéricos en funciones altamente no lineales.

Aquí tienes el script completo y listo para ejecutar que integra la Doble Media Aproximadora (esquema predictor-corrector en cascada).

Este diseño calcula un primer candidato amplio entre los extremos del rebote y utiliza la distancia interna como un selector dinámico para aplicar una segunda capa de promediado (fina o profunda según la magnitud del salto).

Python
import math

def newton_rapido(J, sol, base, historial, tol, max_iter, iter_actual=0, modo_media='DOBLE_APROX'):
    """
    Algoritmo de Newton optimizado con Doble Media Aproximadora (Predictor-Corrector).
    """
    if iter_actual > max_iter:
        return J, iter_actual, False

    try:
        J_estrella = (sol - (base**J)/sol + 1) / sol
    except OverflowError:
        return J, iter_actual, False

    if abs(1 - J_estrella) < tol:
        return J, iter_actual, True

    # --- Llenar el historial (hasta tener 4 puntos) ---
    if len(historial) < 4:
        f = base**J - sol
        df = (base**J) * math.log(base)
        paso = f/df
        
        # 🛡️ FRENO DE EMERGENCIA (Step Bounding)
        if abs(paso) > 50.0:
            paso = 50.0 * (1 if paso > 0 else -1)
            
        J_sig = J - paso
        return newton_rapido(J_sig, sol, base, [J_sig] + historial, tol, max_iter, iter_actual + 1, modo_media)

    # --- Extracción de historial ---
    J_curr, J2, J3, J4 = historial[0], historial[1], historial[2], historial[3]
    p3 = abs(J_curr - J2)
    p2 = abs(J2 - J3)
    p1 = abs(J3 - J4)

    # --- DECISIÓN CLAVE: Detección del primer encogido con Doble Aproximación ---
    if p3 < p2 and p2 >= p1:
        # 1. Predictor Amplio (Primera media entre el extremo antiguo J4 y el actual J_curr)
        J_primero = (J4 + J_curr) / 2.0
        
        # 2. Medidor del salto corrector (Distancia interna)
        delta_corrector = abs(J_curr - J_primero)
        
        if modo_media == 'DOBLE_APROX':
            # Si el salto es amplio, hacemos macro-corrección con J3; si es corto, micro-corrección con J_curr
            if delta_corrector > 1.0:
                J_nuevo = (J3 + J_primero) / 2.0
                desc = f"Doble media MACRO (refinado con J3 y J_primero)"
            else:
                J_nuevo = (J_curr + J_primero) / 2.0
                desc = f"Doble media MICRO (refinado con J y J_primero)"
        elif modo_media == 'J4_J':
            J_nuevo = J_primero
            desc = f"media simple entre J4={J4:.4f} y J={J_curr:.4f}"
        else:
            J_nuevo = (J3 + J2) / 2.0
            desc = f"media clásica entre J3={J3:.4f} y J2={J2:.4f}"

        print(f"    🪡 Inflexión detectada. {desc} -> J_nuevo={J_nuevo:.4f}")
        
        # Llamada recursiva con el historial reiniciado al nuevo punto optimizado
        return newton_rapido(J_nuevo, sol, base, [J_nuevo], tol, max_iter, iter_actual + 1, modo_media)
    else:
        # --- Newton fluye de manera natural sin intervenir ---
        f = base**J_curr - sol
        df = (base**J_curr) * math.log(base)
        paso = f/df
        
        # 🛡️ FRENO DE EMERGENCIA (Step Bounding)
        if abs(paso) > 50.0:
            paso = 50.0 * (1 if paso > 0 else -1)
            
        J_sig = J_curr - paso
        return newton_rapido(J_sig, sol, base, [J_sig, J_curr, J2, J3], tol, max_iter, iter_actual + 1, modo_media)

def newton_estandar(J, sol, base, tol, max_iter):
    for i in range(max_iter):
        try:
            J_estrella = (sol - (base**J)/sol + 1) / sol
            if abs(1 - J_estrella) < tol:
                return J, i, True
                
            f = base**J - sol
            df = (base**J) * math.log(base)
            paso = f/df
            
            if abs(paso) > 50.0:
                paso = 50.0 * (1 if paso > 0 else -1)
                
            J = J - paso
        except OverflowError:
            return None, i, False
    return J, max_iter, False

def ejecutar_test(base, solucion, J_inicial, tol=1e-12, max_iter=200, modo_media='DOBLE_APROX'):
    print(f"\n{'='*50}")
    print(f"CASO: {base}^J = {solucion} | J_inicial = {J_inicial} | Modo: {modo_media}")
    print(f"{'='*50}")
    
    # Newton Estándar de referencia
    print("▶ NEWTON ESTÁNDAR:")
    J_std, iter_std, exito_std = newton_estandar(J_inicial, solucion, base, tol, max_iter)
    if exito_std:
        print(f"✅ Convergió en {iter_std} iteraciones. Solución: {J_std:.10f}")
    else:
        print(f"❌ Falló.")

    # Newton Rápido con Doble Media Aproximadora
    print(f"\n▶ NEWTON RÁPIDO (modo {modo_media}):")
    J_rap, iter_rap, exito_rap = newton_rapido(J_inicial, solucion, base, [J_inicial], tol, max_iter, modo_media=modo_media)
    if exito_rap:
        print(f"✅ Convergió en {iter_rap} iteraciones. Solución: {J_rap:.10f}")
    else:
        print(f"❌ Falló.")
