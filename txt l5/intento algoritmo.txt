import math
from fractions import Fraction

def generar_primos(limite):
    """Generador simple de primos para alimentar el radar."""
    sieve = [True] * (limite + 1)
    for p in range(2, math.isqrt(limite) + 1):
        if sieve[p]:
            for i in range(p * p, limite + 1, p):
                sieve[i] = False
    return [p for p in range(2, limite + 1) if sieve[p]]

def medir_resto_exacto(N, m):
    """
    Sustituye la medición float por aritmética entera pura.
    d(m) = frac(N / 2(2m+3))
    Devuelve el resto (Numerador) y el divisor (Denominador).
    """
    divisor = 2 * (2 * m + 3)
    resto = N % divisor
    return resto, divisor

def medir_estado_exacto(N, m):
    """Evalúa la función de resto d(m) = frac(N / 2(2m+3))"""
    divisor = 2 * (2 * m + 3)
    resto = N % divisor
    # Buscamos la distancia al factor (el objetivo es que resto == divisor // 2)
    distancia_objetivo = resto - (divisor // 2)
    return resto, divisor, distancia_objetivo

def lanzar_sonda(N, m_base, nombre_sonda):
    """
    Despliega la 'Pinza' de 4 mediciones en una estación para captar
    el vector de inercia (Velocidad y Aceleración de la onda).
    """
    R1, D1, obj1 = medir_estado_exacto(N, m_base)
    R2, D2, obj2 = medir_estado_exacto(N, m_base + 1)
    R3, D3, obj3 = medir_estado_exacto(N, m_base + 2)
    R4, D4, obj4 = medir_estado_exacto(N, m_base + 3)
    
    # Derivadas discretas del numerador (Cinemática)
    V1 = R2 - R1
    V2 = R3 - R2
    Aceleracion = V2 - V1
    
    # Comprobación de corte exacto (Factor encontrado)
    if obj1 == 0: return m_base, True
    if obj2 == 0: return m_base + 1, True
    if obj3 == 0: return m_base + 2, True
    if obj4 == 0: return m_base + 3, True
    
    # Predicción del salto (A qué distancia m se cruza el cero de la parábola local)
    # Nota: Se usa Fraction para evitar pérdida de precisión en la predicción
    if Aceleracion != 0:
        salto_predicho = Fraction(-V1, Aceleracion)
    else:
        salto_predicho = Fraction(0, 1) # Línea recta, valle ancho
        
    fase_relativa = obj1 / D1 # Porcentaje de desfase dentro del diente
    
    return {
        "m_base": m_base,
        "velocidad": V1,
        "aceleracion": Aceleracion,
        "salto_local": salto_predicho,
        "fase": fase_relativa,
        "nombre": nombre_sonda

 * Usa BigInt (Nn) para aritmética entera exacta sin pérdida de precisión.
 */
class GPSDivisoresMDC {
    constructor(numeroN) {
        // Convertimos a BigInt para precisión absoluta
        this.N = BigInt(numeroN);
        
        // Espacio de búsqueda inicial
        this.m_inf = 0n;
        this.m_sup = this.N / 6n; 
        this.m_cen = this.isqrt(this.N) / 2n;
        
        this.iteracion = 0;
        this.factorEncontrado = null;
        this.colapso = false;
    }

    // Raíz cuadrada entera para BigInt (método de Newton)
    isqrt(value) {
        if (value < 0n) throw 'Raíz de número negativo';
        if (value < 2n) return value;
        let x0 = value / 2n;
        let x1 = (x0 + value / x0) / 2n;
        while (x1 < x0) {
            x0 = x1;
            x1 = (x0 + value / x0) / 2n;
        }
        return x0;
    }

    medirEstadoExacto(m) {
        let divisor = 2n * (2n * m + 3n);
        let resto = this.N % divisor;
        let objetivo = divisor / 2n; // La línea δ = 0.5
        let distancia = resto - objetivo;
        return { resto, divisor, distancia };
    }

    lanzarSonda(m_base, nombre) {
        let m0 = BigInt(m_base);
        let M1 = this.medirEstadoExacto(m0);
        let M2 = this.medirEstadoExacto(m0 + 1n);
        let M3 = this.medirEstadoExacto(m0 + 2n);
        let M4 = this.medirEstadoExacto(m0 + 3n);

        // Si la distancia al objetivo es 0, hemos interceptado el factor
        if (M1.distancia === 0n) return { exito: true, m_factor: m0 };
        if (M2.distancia === 0n) return { exito: true, m_factor: m0 + 1n };
        if (M3.distancia === 0n) return { exito: true, m_factor: m0 + 2n };
        if (M4.distancia === 0n) return { exito: true, m_factor: m0 + 3n };

        // Cinemática de la Onda (Numerador)
        let V1 = M2.resto - M1.resto;
        let V2 = M3.resto - M2.resto;
        let aceleracion = V2 - V1;

        // La fase la pasamos a Number (float) solo para el renderizado visual,
        // ya que es un porcentaje entre -0.5 y 0.5 y no altera la lógica estricta.

        let saltoLocal = 0n;
        if (aceleracion !== 0n) {
            saltoLocal = (-V1) / aceleracion; // División entera BigInt
        }

        return {
            exito: false,
            nombre: nombre,
            m_base: m0,
            velocidad: V1,
            aceleracion: aceleracion,
            faseVisual: faseVisual,
            saltoLocal: saltoLocal
        };
    }

/**
 * Motor MDC Recursiu - Divideix i Venceràs Cinemàtic
 * Utilitza BigInt per assegurar precisió infinita en l'ona de restes.
 */
class MDCRecursiu {
    constructor(numeroN) {
        this.N = BigInt(numeroN);
        
        // Optimització brutal: Només busquem fins a l'arrel quadrada.
        // Tot factor major té la seva parella per sota de l'arrel.
        this.m_inf = 0n;
        this.m_sup = this.isqrt(this.N) / 2n; 
        
        this.iteracio = 0;
        this.factorEncontrat = null;
    }

    // Arrel quadrada exacta per nombres enormes (BigInt)
    isqrt(value) {
        if (value < 0n) throw 'No es pot calcular arrel de negatius';
        if (value < 2n) return value;
        let x0 = value / 2n;
        let x1 = (x0 + value / x0) / 2n;
        while (x1 < x0) {
            x0 = x1;
            x1 = (x0 + value / x0) / 2n;
        }
        return x0;
    }

    // Avaluem la fracció com Numerador i Denominador
    mesurarEstat(m) {
        let divisor = 2n * (2n * m + 3n);
        let resta = this.N % divisor;
        let objectiu = divisor / 2n; // La línia objectiu 0.5
        return { resta, divisor, distancia: resta - objectiu };
    }

    // La "Pinza" només necessita velocitat per saber la direcció de la paràbola
    llancarPinza(m_base, nom) {
        let M1 = this.mesurarEstat(m_base);
        let M2 = this.mesurarEstat(m_base + 1n);
        let M3 = this.mesurarEstat(m_base + 2n);

        // Si tallem el 0.5 exacte, hem guanyat
        if (M1.distancia === 0n) return { exito: true, factor: m_base };
        if (M2.distancia === 0n) return { exito: true, factor: m_base + 1n };
        if (M3.distancia === 0n) return { exito: true, factor: m_base + 2n };

        let velocitat = M2.resta - M1.resta;
        
        // Retornem la fase en Float només per poder dibuixar els vectors en el teu Canvas
        let faseVisual = Number(M1.distancia) / Number(M1.divisor);

        return { exito: false, nom, m_base, velocitat, fase: faseVisual };
    }

    /**
     * Crida aquesta funció a cada "frame" o en bucle fins que acabi.
     * Avalua, talla l'espai per la meitat, i fa "zoom".
     */
    pasDeSegmentacio() {
        if (this.factorEncontrat !== null) return { estat: "FI" };

        this.iteracio++;
        let m_cen = (this.m_inf + this.m_sup) / 2n;
        let ampladaActual = this.m_sup - this.m_inf;

        // 1. RE-DESPLEGAR PINCES AL NOU SEGMENT
        let pInf = this.llancarPinza(this.m_inf, "Inici");
        if (pInf.exito) { this.factorEncontrat = pInf.factor; return { estat: "EXIT", factor: this.factorEncontrat }; }

        let pCen = this.llancarPinza(m_cen, "Centre");
        if (pCen.exito) { this.factorEncontrat = pCen.factor; return { estat: "EXIT", factor: this.factorEncontrat }; }

        let pSup = this.llancarPinza(this.m_sup, "Final");
        if (pSup.exito) { this.factorEncontrat = pSup.factor; return { estat: "EXIT", factor: this.factorEncontrat }; }

        // 2. TALLAR L'ESPAI (ANÀLISI DE DESFASE)
        let sectorAntic = { inf: this.m_inf, sup: this.m_sup, cen: m_cen };
        let direccio = "";

        if (pCen.velocitat > 0n) {
            // El centre de l'ona puja. El mínim ha quedat enrere.
            // Tallem tota la part dreta. Segmentem a la meitat esquerra.
            this.m_sup = m_cen;
            direccio = "ESQUERRA";
        } else {
            // El centre baixa. El mínim està més endavant.
            // Tallem tota la part esquerra. Segmentem a la meitat dreta.
            this.m_inf = m_cen;
            direccio = "DRETA";
        }

        // 3. ATAC FINAL SI EL SEGMENT ÉS MINÚSCUL
        if (ampladaActual < 100n) {
            for (let i = this.m_inf; i <= this.m_sup; i++) {
                if (this.N % (2n * (2n * i + 3n)) === 0n) {
                    this.factorEncontrat = i;
                    return { estat: "EXIT", factor: this.factorEncontrat };
                }
            }
            return { estat: "PRIMER" }; // Si col·lapsa a 0 i no hi ha factor, el nombre era primer!
        }

        return {
            estat: "ZOOM",
            iteracio: this.iteracio,
            direccio: direccio,
            amplada: ampladaActual,
            sectorAntic: sectorAntic,
            pinces: { inf: pInf, cen: pCen, sup: pSup }
        };
    }
}

import math
import time

class MotorMDC_Unificado:
    def __init__(self, N, primos_filtro=[2, 3, 5, 7]):
        """
        Inicializa el motor cuántico MDC-U.
        Genera la topología de islas usando Euler sobre el primorial.
        """
        self.N = N
        self.primos = primos_filtro
        self.primorial = math.prod(self.primos)
        
        # 1. CAPA TOPOLÓGICA: Generar las islas válidas en un ciclo p#
        # Solo guardamos coprimos del primorial.
        self.islas_base = [
            i for i in range(1, self.primorial + 1)
            if all(i % p != 0 for p in self.primos)
        ]
        self.num_islas_por_ciclo = len(self.islas_base)
        
        # Densidad de supervivencia (Métrica de Euler)
        self.densidad = self.num_islas_por_ciclo / self.primorial
        
    def _obtener_m_desde_indice(self, indice_isla):
        """
        Mapeo O(1): Convierte el índice de la isla en su coordenada 'm' real en la recta.
        ¡Permite navegar por números de miles de dígitos sin usar RAM!
        """
        ciclos_completos = indice_isla // self.num_islas_por_ciclo
        posicion_en_ciclo = indice_isla % self.num_islas_por_ciclo
        
        return ciclos_completos * self.primorial + self.islas_base[posicion_en_ciclo]

    def _medir_pinza(self, m_base):
        """
        CAPA DE RADAR: Dispara la pinza de 3 puntos en aritmética 100% entera.
        Evalúa f(m) = N % 2(2m+3). Objetivo: Resto == Divisor / 2.
        """
        # Distancia exacta al objetivo (0 = impacto directo)
        def medir_dist(m):
            divisor = 2 * (2 * m + 3)
            return (self.N % divisor) - (divisor // 2)

        d1 = medir_dist(m_base)
        if d1 == 0: return True, m_base, 0, 0
        
        d2 = medir_dist(m_base + 1)
        if d2 == 0: return True, m_base + 1, 0, 0
        
        d3 = medir_dist(m_base + 2)
        if d3 == 0: return True, m_base + 2, 0, 0

        # Cinemática Diferencial (Derivadas del numerador)
        velocidad = d2 - d1
        aceleracion = (d3 - d2) - velocidad
        
        return False, m_base, velocidad, aceleracion

    def ejecutar_factorizacion(self):
        """
        Ejecuta el Pipeline MDC-U uniendo Topología y Cinemática Balística.
        """
        print(f"=== Iniciando MDC-U para N = {self.N} ===")
        print(f"Filtro Primorial: p# = {self.primorial} (Primos: {self.primos})")
        print(f"Densidad de Supervivencia: {self.densidad*100:.2f}% del espacio total.\n")
        
        t_inicio = time.perf_counter()
        
        # Límite físico por simetría
        m_limite_global = (math.isqrt(self.N) - 3) // 2
        indice_isla_actual = 0
        
        saltos_parabolicos_calculados = 0
        
        while True:
            # 1. Aterrizar estrictamente en una isla permitida
            m_actual = self._obtener_m_desde_indice(indice_isla_actual)
            
            if m_actual > m_limite_global:
                break # Superada la raíz cuadrada, es primo
                
            # 2. Desplegar Sensor y leer la fase
            impacto, m_exacto, vel, acel = self._medir_pinza(m_actual)
            
            if impacto:
                factor = 2 * m_exacto + 3
                t_fin = time.perf_counter()
                self._imprimir_exito(factor, t_fin - t_inicio, saltos_parabolicos_calculados)
                return factor
                
            # 3. Disparo Cinemático (Cerca en Profundidad)
            # Solo disparamos si la onda está descendiendo (vel < 0)
            if vel < 0 and acel != 0:
                saltos_parabolicos_calculados += 1
                
                # Calculamos matemáticamente dónde la parábola corta el 0
                distancia_proyectada = -vel // acel
                m_candidato = m_actual + distancia_proyectada
                
                # Verificación de impacto sin iterar (O(1))
                if m_candidato > 0:
                    divisor_candidato = 2 * (m_candidato) + 3
                    if self.N % divisor_candidato == 0:
                        t_fin = time.perf_counter()
                        self._imprimir_exito(divisor_candidato, t_fin - t_inicio, saltos_parabolicos_calculados)
                        return divisor_candidato
            
            # Pasar a la siguiente isla
            indice_isla_actual += 1
            
        print(f"Búsqueda finalizada. El número {self.N} es PRIMO.")
        return 1

    def _imprimir_exito(self, f1, tiempo, saltos):
        f2 = self.N // f1
        print(f"🎯 ¡IMPACTO CONFIRMADO!")
        print(f"Factores exactos : {f1} × {f2}")
        print(f"Tiempo de cómputo: {tiempo*1000:.4f} ms")
        print(f"Cálculos balísticos realizados: {saltos}")

import math
import time
from fractions import Fraction

class MDC_Unificado:
    def __init__(self, N, primos_filtro=[2, 3, 5]):
        """
        Inicializa el motor. Construye la matriz topológica (Rueda Primorial)
        para asegurar que las pinzas solo aterrizan en zonas matemáticamente posibles.
        """
        self.N = N
        self.primos = primos_filtro
        self.primorial = math.prod(self.primos)
        
        # Generar las "islas" (números coprimos al primorial)
        self.islas_base = [
            i for i in range(1, self.primorial + 1)
            if all(i % p != 0 for p in self.primos)
        ]
        self.num_islas = len(self.islas_base)
        self.densidad = self.num_islas / self.primorial
        
    def _obtener_coordenada_m(self, indice):
        """
        Calcula la coordenada 'm' real en O(1). 
        Permite navegar el infinito sin consumir memoria RAM.
        """
        ciclos = indice // self.num_islas
        pos = indice % self.num_islas
        return ciclos * self.primorial + self.islas_base[pos]

    def _medir_tensor_4_puntos(self, m_base):
        """
        Despliega la pinza continua de 4 puntos.
        Aritmética de Fracciones Exactas para no deformar la parábola.
        """
        Y = []
        for i in range(4):
            m_actual = m_base + i
            divisor = 2 * (2 * m_actual + 3)
            
            # 1. Comprobación de impacto directo en la medición
            resto = self.N % divisor
            if resto == 0:
                return True, m_actual, None, None, None
            
            # 2. Medición de la onda: f(m) = (N % D)/D - 0.5
            # Usamos Fraction para precisión infinita
            fase = Fraction(resto, divisor) - Fraction(1, 2)
            Y.append(fase)

        # --- DERIVADAS DISCRETAS (El Motor Cinemático) ---
        # 1ª Derivada: 3 Velocidades
        v1 = Y[1] - Y[0]
        v2 = Y[2] - Y[1]
        v3 = Y[3] - Y[2]

        # 2ª Derivada: 2 Aceleraciones
        a1 = v2 - v1
        a2 = v3 - v2

        # 3ª Derivada: 1 Variación Acelerativa (Jerk / Invariante Parabólico)
        jerk = a2 - a1

        return False, m_base, v1, a1, jerk

    def ejecutar(self):
        print(f"[{time.strftime('%H:%M:%S')}] INICIANDO MDC-U")
        print(f"[-] Objetivo N = {self.N}")
        print(f"[-] Rueda Primorial p# = {self.primorial} | Islas por ciclo = {self.num_islas}")
        print(f"[-] Densidad de búsqueda = {self.densidad*100:.2f}% del espacio\n")

        t0 = time.perf_counter()
        
        m_limite = (math.isqrt(self.N) - 3) // 2
        indice = 0
        saltos_calculados = 0
        dientes_cruzados_evitados = 0

        while True:
            # 1. Aterrizaje Topológico Seguro
            m = self._obtener_coordenada_m(indice)
            
            if m > m_limite:
                break # Superamos la raíz, el número es primo

            # 2. Medición Cinemática de 4 Puntos
            impacto, m_exacto, v1, a1, jerk = self._medir_tensor_4_puntos(m)

            if impacto:
                factor = 2 * (2 * m_exacto + 3)
                self._reporte(factor, time.perf_counter() - t0, saltos_calculados, dientes_cruzados_evitados)
                return factor

            # 3. Validación de Seguridad (El Escudo del Jerk)
            # En una función racional, el Jerk debe ser minúsculo en el mismo diente.
            # Si el Jerk es del mismo orden de magnitud que la aceleración, 
            # significa que la pinza ha mordido el borde de un diente de sierra.
            if a1 != 0:
                variacion_relativa = abs(jerk / a1)
                
                # Si la variación es mayor al 5%, estamos cruzando un acantilado. ABORTAR SALTO.
                if variacion_relativa > Fraction(5, 100):
                    dientes_cruzados_evitados += 1
                    indice += 1
                    continue

                # 4. El Salto Parabólico (Solo si desciende y es seguro)
                if v1 < 0:
                    saltos_calculados += 1
                    
                    # Salto predictivo M: m_target = m + floor(-v / a)
                    salto_m = int(-v1 / a1) 
                    m_target = m + salto_m
                    
                    # Verificación O(1) en el punto de impacto
                    if m_target > 0:
                        divisor_target = 2 * (2 * m_target + 3)
                        if self.N % divisor_target == 0:
                            self._reporte(divisor_target // 2, time.perf_counter() - t0, saltos_calculados, dientes_cruzados_evitados)
                            return divisor_target // 2

            # Pasar a la siguiente isla
            indice += 1

        print(f"\n[!] Búsqueda finalizada. {self.N} es PRIMO.")
        return 1

    def _reporte(self, f1, t_ms, saltos, evitados):
        f2 = self.N // f1
        print(f"==================================================")
        print(f"🎯 IMPACTO CONFIRMADO: FACTORIZACIÓN EXACTA")
        print(f"==================================================")
        print(f" Factores  : {f1} × {f2}")
        print(f" Tiempo    : {t_ms * 1000:.4f} ms")
        print(f" Disparos  : {saltos} cálculos balísticos parabólicos.")
        print(f" Escudo 4P : {evitados} saltos falsos evitados por el Invariante Jerk.")
        print(f"==================================================\n")

import math

from fractions import Fraction

class MDC_Engine:
    """Motor del Método Diofántico Cinemático Unificado (MDC-U)"""
    
    def __init__(self, N, prime_pool=[2, 3, 5]):
        self.N = N
        self.prime_pool = prime_pool
        self.primorial = math.prod(prime_pool)
        
        self.valid_offsets = [
            i for i in range(1, self.primorial + 1)
            if all(i % p != 0 for p in prime_pool)
        ]
        self.cycle_length = len(self.valid_offsets)
        self.density = self.cycle_length / self.primorial

    def _get_m(self, index):
        cycles = index // self.cycle_length
        pos = index % self.cycle_length
        return cycles * self.primorial + self.valid_offsets[pos]

    def _tensor_eval(self, m_base):
        Y = []
        for i in range(4):
            m_curr = m_base + i
            divisor = 2 * (2 * m_curr + 3)
            
            rem = self.N % divisor
            if rem == 0:
                return True, m_curr, 0, 0, 0
            
            phase = Fraction(rem, divisor) - Fraction(1, 2)
            Y.append(phase)

        v1 = Y[1] - Y[0]
        v2 = Y[2] - Y[1]
        v3 = Y[3] - Y[2]

        a1 = v2 - v1
        a2 = v3 - v2
        jerk = a2 - a1

        return False, m_base, v1, a1, jerk

    def factorize(self):
        t0 = time.perf_counter()
        
        m_limit = (math.isqrt(self.N) - 3) // 2
        idx = 0
        jumps = 0
        aborts = 0

        while True:
            m = self._get_m(idx)
            if m > m_limit:
                break

            hit, m_exact, v1, a1, jerk = self._tensor_eval(m)

            if hit:
                f1 = 2 * (2 * m_exact + 3)
                return self._finalize(f1, time.perf_counter() - t0, jumps, aborts)

            if a1 != 0:
                jerk_ratio = abs(jerk / a1)
                
                if jerk_ratio > Fraction(5, 100):
                    aborts += 1
                    idx += 1
                    continue

                    jumps += 1

                        D_target = 2 * (2 * m_target + 3)
                        if self.N % D_target == 0:

ESPECIFICACIÓN TÉCNICA: MDC-U (Método Diofántico Cinemático Unificado)

El algoritmo se define como una búsqueda de raíces de una función racional paramétrica, evaluada exclusivamente sobre una topología discreta pre-calculada.

1. Mapeo Topológico (Rueda Primorial)

Para un conjunto de primos $P = \{p_1, p_2, \dots, p_k\}$, el primorial se define como $p\# = \prod p_i$.

El espacio de búsqueda $L$ se restringe a los enteros $m$ tales que el divisor candidato $D = 2(2m+3)$ es coprimo con $p\#$.

La densidad de supervivencia del espacio es exactamente la función de Euler normalizada:

$$\rho = \prod_{p \in P} \left(1 - \frac{1}{p}\right)$$

El mapeo de un índice secuencial $i$ a su coordenada real $m$ se realiza en $O(1)$ sin consumo de memoria iterativa.

2. Tensor Cinemático de 4 Puntos

Para una coordenada $m_0 \in L$, se evalúa la función de desfase fraccionario en 4 puntos consecutivos de la métrica natural (saltos de $+1$):

$$f(m) = \frac{N \pmod{2(2m+3)}}{2(2m+3)} - 0.5$$

Se extraen las derivadas discretas (diferencias finitas):

Velocidades: $v_1 = f(m_1) - f(m_0)$, $\quad v_2 = f(m_2) - f(m_1)$, $\quad v_3 = f(m_3) - f(m_2)$

Aceleraciones: $a_1 = v_2 - v_1$, $\quad a_2 = v_3 - v_2$

Variación Acelerativa (Jerk): $J = a_2 - a_1$

3. Condición de Salto (Invariante Parabólico)

El cálculo balístico solo se autoriza si se cumplen dos condiciones estrictas:

$v_1 < 0$ (La onda se aproxima a la raíz $f(m)=0$).

$|J / a_1| \le \epsilon$ (Donde $\epsilon$ es el umbral de tolerancia al error de escala, e.g., $0.05$). Si la variación supera el umbral, el tensor ha intersectado el límite de un "diente de sierra" y el salto se aborta.

Si la validación es exitosa, se proyecta el impacto asumiendo trayectoria parabólica:

$$\Delta m = \lfloor -v_1 / a_1 \rfloor$$

import math
import time
import multiprocessing as mp
from fractions import Fraction

def _worker_task(args):
    """
    Subrutina independiente para el worker. 
    Se aísla a nivel de módulo para evitar bloqueos de serialización (Pickle).
    """
    N, prime_pool, start_idx, end_idx, stop_event = args
    
    primorial = math.prod(prime_pool)
    valid_offsets = [
        i for i in range(1, primorial + 1)
        if all(i % p != 0 for p in prime_pool)
    ]
    cycle_length = len(valid_offsets)

    def get_m(idx):
        cycles = idx // cycle_length
        pos = idx % cycle_length
        return cycles * primorial + valid_offsets[pos]

    jumps = 0
    aborts = 0

    def tensor_eval(m_base):
        Y = []
        for i in range(4):
            m_curr = m_base + i
            divisor = 2 * (2 * m_curr + 3)
            
            rem = N % divisor
            if rem == 0:
                return True, m_curr, 0, 0, 0
            
            phase = Fraction(rem, divisor) - Fraction(1, 2)
            Y.append(phase)

        v1 = Y[1] - Y[0]
        v2 = Y[2] - Y[1]
        v3 = Y[3] - Y[2]

        a1 = v2 - v1
        a2 = v3 - v2
        jerk = a2 - a1

        return False, m_base, v1, a1, jerk

    idx = start_idx
    while idx <= end_idx:
        # Polling del evento de parada (cada 500 iteraciones para no añadir overhead)
        if idx % 500 == 0 and stop_event.is_set():
            return None

        m = get_m(idx)
        hit, m_exact, v1, a1, jerk = tensor_eval(m)

        if hit:
            stop_event.set()
            return (2 * (2 * m_exact + 3), jumps, aborts)

        if a1 != 0:
            jerk_ratio = abs(jerk / a1)
            
            if jerk_ratio > Fraction(5, 100):
                aborts += 1
                idx += 1
                continue

            if v1 < 0:
                jumps += 1
                m_target = m + int(-v1 / a1)
                
                if m_target > 0:
                    D_target = 2 * (2 * m_target + 3)
                    if N % D_target == 0:
                        stop_event.set()
                        return (D_target // 2, jumps, aborts)

        idx += 1

    return None

class MDC_ParallelEngine:
    """Motor MDC-U con paralelismo a nivel de bloque lógico."""
    
    def __init__(self, N, prime_pool=[2, 3, 5], cores=None):
        self.N = N
        self.prime_pool = prime_pool
        self.cores = cores or mp.cpu_count()
        self.primorial = math.prod(prime_pool)
        
        self.valid_offsets = [
            i for i in range(1, self.primorial + 1)
            if all(i % p != 0 for p in prime_pool)
        ]
        self.cycle_length = len(self.valid_offsets)

    def factorize(self):
        t0 = time.perf_counter()
        m_limit = (math.isqrt(self.N) - 3) // 2
        
        # Cálculo exacto del índice máximo (max_idx)
        cycles = m_limit // self.primorial
        rem = m_limit % self.primorial
        rem_offsets = sum(1 for x in self.valid_offsets if x <= rem)
        max_idx = cycles * self.cycle_length + rem_offsets - 1

        if max_idx < 0:
            return self._finalize(1, time.perf_counter() - t0, 0, 0, is_prime=True)

        # Sobresuscripción de chunks para balanceo de carga asimétrico
        chunk_size = max(1, (max_idx + 1) // (self.cores * 4))
        
        manager = mp.Manager()
        stop_event = manager.Event()
        
        tasks = []
        for start_idx in range(0, max_idx + 1, chunk_size):
            end_idx = min(start_idx + chunk_size - 1, max_idx)
            tasks.append((self.N, self.prime_pool, start_idx, end_idx, stop_event))

        factor = None
        total_jumps = 0
        total_aborts = 0

        with mp.Pool(processes=self.cores) as pool:
            for result in pool.imap_unordered(_worker_task, tasks):
                if result is not None:
                    f1, jumps, aborts = result
                    factor = f1
                    total_jumps = jumps
                    total_aborts = aborts
                    break

        t_elapsed = time.perf_counter() - t0

        if factor:
            return self._finalize(factor, t_elapsed, total_jumps, total_aborts)
        else:
            return self._finalize(1, t_elapsed, 0, 0, is_prime=True)

    def _finalize(self, f1, t_elapsed, jumps, aborts, is_prime=False):
        if is_prime:
            return {
                "status": "PRIME", 
                "N": self.N, 
                "time_ms": t_elapsed * 1000, 
                "cores_used": self.cores
            }
        
        f2 = self.N // f1
        return {
            "status": "FACTORIZED",
            "N": self.N,
            "f1": f1,
            "f2": f2,
            "time_ms": t_elapsed * 1000,
            "parabolic_jumps": jumps,
            "jerk_aborts": aborts,
            "cores_used": self.cores
        }

class MDC_ParallelEngine:
    def __init__(self, N, prime_pool=[2, 3, 5], cores=None):
        self.N = N
        self.prime_pool = prime_pool
        self.cores = cores or mp.cpu_count()
        self.primorial = math.prod(prime_pool)
        
        self.valid_offsets = [
            i for i in range(1, self.primorial + 1)
            if all(i % p != 0 for p in prime_pool)
        ]
        self.cycle_length = len(self.valid_offsets)

    def factorize(self):
        t0 = time.perf_counter()
        m_limit = (math.isqrt(self.N) - 3) // 2
        
        # Cálculo exacto del índice máximo (max_idx) topológico
        cycles = m_limit // self.primorial
        rem = m_limit % self.primorial
        rem_offsets = sum(1 for x in self.valid_offsets if x <= rem)
        max_idx = cycles * self.cycle_length + rem_offsets - 1

        if max_idx < 0:
            return self._finalize(1, time.perf_counter() - t0, 0, 0, is_prime=True)

        # Sobresuscripción de chunks
        chunk_size = max(1, (max_idx + 1) // (self.cores * 4))
        
        manager = mp.Manager()
        stop_event = manager.Event()
        
        tasks = []
        for start_idx in range(0, max_idx + 1, chunk_size):
            end_idx = min(start_idx + chunk_size - 1, max_idx)
            tasks.append((self.N, self.prime_pool, start_idx, end_idx, stop_event))

        factor = None
        total_jumps = 0
        total_aborts = 0

        # Ejecución asíncrona desordenada para máxima eficiencia
        with mp.Pool(processes=self.cores) as pool:
            for result in pool.imap_unordered(_worker_task, tasks):
                if result is not None:
                    f1, jumps, aborts = result
                    factor = f1
                    total_jumps = jumps
                    total_aborts = aborts
                    break

        t_elapsed = time.perf_counter() - t0

        if factor:
            return self._finalize(factor, t_elapsed, total_jumps, total_aborts)
        else:
            return self._finalize(1, t_elapsed, 0, 0, is_prime=True)

    def _finalize(self, f1, t_elapsed, jumps, aborts, is_prime=False):
        if is_prime:
            return {
                "status": "PRIME", 
                "N": self.N, 
                "time_ms": t_elapsed * 1000, 
                "cores_used": self.cores
            }
        
        f2 = self.N // f1
        return {
            "status": "FACTORIZED",
            "N": self.N,
            "f1": f1,
            "f2": f2,
            "time_ms": t_elapsed * 1000,
            "parabolic_jumps": jumps,
            "jerk_aborts": aborts,
            "cores_used": self.cores
        }

def algoritmo_resonancia_espejo(N, primos_filtro=[2, 3, 5]):
    print(f"--- INICIANDO RUTA DE RESONANCIA PARA N = {N} ---")
    
    # 1. MATEMÁTICA DE LA RUEDA (Topología)
    # Definimos el paso según el primorial para saltar espacios vacíos
    p_hash = math.prod(primos_filtro)
    islas = [i for i in range(1, p_hash + 1) if all(i % p != 0 for p in primos_filtro)]
    
    # 2. ACOTACIÓN DE LA TALLA (Evitamos extremos inútiles)
    # El factor no suele estar pegado a 1 ni a la raíz si es un semiprimo grande
    m_min = int(N**(1/4)) 
    m_max = int(math.isqrt(N))
    
    print(f"[*] Espacio de búsqueda acotado: [{m_min}, {m_max}]")
    print(f"[*] Densidad de señal: {len(islas)/p_hash*100:.2f}% de la recta.\n")

    # 3. DIVIDE Y VENCERÁS: Segmentación por Pinzas Espejo
    # Dividimos la talla en bloques del 5% para comparar
    num_bloques = 20
    paso_bloque = (m_max - m_min) // num_bloques

    for b in range(num_bloques):
        m_eval = m_min + (b * paso_bloque)
        
        # --- TEORÍA DE LA PINZA DE 4 PUNTOS ---
        # Medimos la fase en el Segmento A
        def medir_fase(m):
            divisor = 2 * (2 * m + 3)
            resto = N % divisor
            if resto == 0: return 0, True # IMPACTO DIRECTO
            # La fase es la distancia relativa al centro del diente de sierra
            return Fraction(resto, divisor) - Fraction(1, 2), False

        # Tomamos 4 diferenciales para estudiar la parábola de la señal
        fases = []
        for i in range(4):
            f, hit = medir_fase(m_eval + i)
            if hit: 
                print(f"🎯 FACTOR ENCONTRADO POR IMPACTO: {2*(2*(m_eval+i)+3)}")
                return
            fases.append(f)

        # 4. MATEMÁTICA DIFERENCIAL (Velocidad y Aceleración)
        v1 = fases[1] - fases[0]
        v2 = fases[2] - fases[1]
        a1 = v2 - v1 # Aceleración local (Curvatura de la señal)
        
        # 5. COMPARATIVA DE SEGMENTO ESPEJO (B = N/A)
        # Aquí aplicamos la teoría de que A*B = N
        divisor_A = 2 * (2 * m_eval + 3)
        divisor_B_estimado = N // divisor_A
        
        # Estudiamos si la señal en A tiene "coherencia" con su espejo
        print(f"Bloque {b:02}: Analizando m={m_eval} | Vel: {float(v1):.6f} | Acel: {float(a1):.6f}")
        
        # SI LA SEÑAL SE ALARGA (v1 pequeña) Y LA ACELERACIÓN ES ESTABLE:
        if abs(a1) > 0 and v1 < 0:
            # Aplicamos el SALTO BALÍSTICO (Salto de Talla)
            # Predecimos dónde la parábola de la señal tocará el cero
            salto_m = int(-v1 / a1)
            
            if 0 < salto_m < paso_bloque:
                print(f"   --> 🔍 POSIBLE NODO DETECTADO. Desfase convergiendo en +{salto_m} pasos.")
                # Aquí el algoritmo haría "Búsqueda en Profundidad"
                m_target = m_eval + salto_m
                if N % (2*(2*m_target+3)) == 0:
                    print(f"   🎯 ¡FACTORIZACIÓN EXITOSA EN NODO! Factor: {2*(2*m_target+3)}")
                    return

from fractions import Fraction

class MDC_RecursiveSegmentation:
    def __init__(self, N, tolerancia=0.01):
        self.N = N
        self.tolerancia = tolerancia
        self.factor_encontrado = None

    def medir_cinematica(self, m):
        """Aplica la pinza de 4 puntos para extraer vector y aceleración."""
        y = []
        for i in range(4):
            divisor = 2 * (2 * (m + i) + 3)
            # Señal de fase normalizada respecto al divisor
            fase = Fraction(self.N % divisor, divisor) - Fraction(1, 2)
            y.append(fase)
        
        v = y[1] - y[0]  # Vector director
        a = (y[2] - y[1]) - v  # Aceleración (curvatura)
        jerk = ((y[3] - y[2]) - (y[2] - y[1])) - a # Invariante de seguridad
        return v, a, jerk

    def buscar_en_profundidad(self, m_min, m_max, nivel=0):
        if self.factor_encontrado or m_min >= m_max:
            return

        # 1. Punto de corte (Bisección)
        m_mid = (m_min + m_max) // 2
        
        # 2. Medida de control en el centro del segmento
        v, a, jerk = self.medir_cinematica(m_mid)
        
        # Verificación de impacto directo
        divisor_mid = 2 * (2 * m_mid + 3)
        if self.N % divisor_mid == 0:
            self.factor_encontrado = divisor_mid
            return

        # 3. Lógica de Poda y Salto (Divide y Vencerás)
        # Si el segmento es pequeño, intentamos el disparo parabólico
        if (m_max - m_min) < 100:
            if a != 0:
                salto = int(-v / a)
                m_final = m_mid + salto
                div_final = 2 * (2 * m_final + 3)
                if self.N % div_final == 0:
                    self.factor_encontrado = div_final
            return

        # 4. Decisión de Rama por Choque de Vectores
        # Si v es negativo, la caída al cero está a la derecha
        if v < 0:
            # Explorar mitad derecha
            self.buscar_en_profundidad(m_mid + 1, m_max, nivel + 1)
            # Si no se encontró, la señal podría estar desfasada (Efecto Acordeón)
            if not self.factor_encontrado:
                self.buscar_en_profundidad(m_min, m_mid - 1, nivel + 1)

            # Explorar mitad izquierda
            self.buscar_en_profundidad(m_min, m_mid - 1, nivel + 1)
            if not self.factor_encontrado:
                self.buscar_en_profundidad(m_mid + 1, m_max, nivel + 1)

        print(f"Segmentando Talla: [{m_start} ... {m_end}]")

        return self.factor_encontrado

import math
from fractions import Fraction

class MDC_DeepSearch:
    def __init__(self, N):
        self.N = N
        self.encontrado = None
        self.iteraciones = 0

    def obtener_fase(self, m):
        """ Pinza de 4 puntos para extraer la cinemática de la señal """
        y = []
        for i in range(4):
            divisor = 2 * (2 * (m + i) + 3)
            # Fase normalizada: (N % D) / D - 0.5
            fase = Fraction(self.N % divisor, divisor) - Fraction(1, 2)
            y.append(fase)
        
        v = y[1] - y[0]  # Vector Director
        a = (y[2] - y[1]) - v # Aceleración (Curvatura)
        j = ((y[3] - y[2]) - (y[2] - y[1])) - a # Jerk (Invariante)
        return v, a, j, y[0]

    def segmentar(self, m_min, m_max, profundidad=0):
        if self.encontrado or m_min >= m_max:
            return

        self.iteraciones += 1
        m_mid = (m_min + m_max) // 2
        
        # --- MEDICIÓN DE RESONANCIA ESPEJO ---
        v_a, a_a, j_a, f_a = self.obtener_fase(m_mid)
        
        # Comprobación de impacto
        if self.N % (2 * (2 * m_mid + 3)) == 0:
            self.encontrado = 2 * (2 * m_mid + 3)
            return

        # --- LÓGICA DE DIVIDE Y VENCERÁS (CHOQUE DE VECTORES) ---
        # Si la velocidad es negativa, el cero de la señal está adelante.
        # Si es positiva, el cero quedó atrás.
        
        # Umbral de seguridad: Si el segmento es muy pequeño, disparamos parábola
        if (m_max - m_min) < 50:
            if a_a != 0:
                salto = int(-v_a / a_a)
                m_target = m_mid + salto
                if self.N % (2 * (2 * m_target + 3)) == 0:
                    self.encontrado = 2 * (2 * m_target + 3)
            return

        # DECISIÓN DE RAMA:
        # Aquí es donde 'sabemos por donde segmentar' observando el vector director
        if v_a < 0:
            # El vector apunta a la derecha: El factor tiende a estar en la mitad superior
            print(f"[Prof {profundidad}] Vector {float(v_a):.5f} -> Segmentando Derecha")
            self.segmentar(m_mid + 1, m_max, profundidad + 1)
            # Si no se encuentra en la derecha, buscamos en la izquierda (Fase de respaldo)
            if not self.encontrado:
                self.segmentar(m_min, m_mid - 1, profundidad + 1)
        else:
            # El vector apunta a la izquierda: El factor tiende a estar en la mitad inferior
            print(f"[Prof {profundidad}] Vector {float(v_a):.5f} -> Segmentando Izquierda")
            self.segmentar(m_min, m_mid - 1, profundidad + 1)
            if not self.encontrado:
                self.segmentar(m_mid + 1, m_max, profundidad + 1)

    def ejecutar(self):
        # Acotación de la talla según tu teoría de n a n-raiz/4
        inicio = int(self.N**(1/4))
        fin = int(math.isqrt(self.N))
        self.segmentar(inicio, fin)
        return self.encontrado

        """
        def obtener_v_a(pos):
            y = []
            for i in range(4):
                D = 2 * (2 * (pos + i) + 3)
                y.append(Fraction(self.N % D, D) - Fraction(1, 2))
            v = y[1] - y[0]
            a = (y[2] - y[1]) - v
            return v, a

        v1, a1 = obtener_v_a(m)

        v2, a2 = obtener_v_a(m + 10) 
        
        # La frecuencia se detecta por la variación del vector director
        frecuencia = v2 - v1 
        return v1, a1, frecuencia

import math
import time
from fractions import Fraction

class MDC_Resonancia_Fractal:
    def __init__(self, N, p_pool=[2, 3, 5]):
        self.N = N
        self.p_hash = math.prod(p_pool)
        self.islas = [i for i in range(1, self.p_hash + 1) if all(i % p != 0 for p in p_pool)]
        self.factor = None

    def analizar_parabola(self, m):
        """ Extrae vector (v), aceleración (a) y fase (f) de una pinza de 4 puntos """
        y = []
        for i in range(4):
            D = 2 * (2 * (m + i) + 3)
            if self.N % D == 0: return None, None, True # Impacto
            y.append(Fraction(self.N % D, D) - Fraction(1, 2))
        
        v = y[1] - y[0]
        a = (y[2] - y[1]) - v
        return v, a, False

    def medir_frecuencia_segmento(self, m):
        """ Compara dos parábolas consecutivas para validar la frecuencia del segmento """
        v1, a1, hit1 = self.analizar_parabola(m)
        if hit1: return 0, 0, 0, True
        
        # Salto de 10 unidades para medir la siguiente parábola (Consecutividad)
        v2, a2, hit2 = self.analizar_parabola(m + 10)
        if hit2: return 0, 0, 0, True
        
        # Frecuencia: Variación del vector director en la distancia
        freq = (v2 - v1) / 10
        return v1, a1, freq, False

    def biseccion_recursiva(self, m_min, m_max, prof=0):
        if self.factor or m_min >= m_max: return

        m_mid = (m_min + m_max) // 2
        v, a, freq, hit = self.medir_frecuencia_segmento(m_mid)
        
        if hit:
            # El factor se detectó durante la medición de frecuencia
            # Se busca el m exacto que dio resto 0
            for i in range(14): 
                if self.N % (2*(2*(m_mid+i)+3)) == 0:
                    self.factor = 2*(2*(m_mid+i)+3)
            return

        # --- CRITERIO DE ELECCIÓN DE SEGMENTO ---
        # 1. Si el segmento es muy pequeño, ejecutamos el disparo balístico
        if (m_max - m_min) < 100:
            if a != 0:
                m_target = m_mid + int(-v / a)
                D = 2 * (2 * m_target + 3)
                if self.N % D == 0: self.factor = D
            return

        # 2. Divide y Vencerás por Choque de Vectores
        # Si v < 0, la fase cae hacia la derecha (hacia factores mayores)
        if v < 0:
            # Explorar rama derecha (profundidad)
            self.biseccion_recursiva(m_mid + 1, m_max, prof + 1)
            # Si no hay resonancia, probar rama izquierda
            if not self.factor:
                self.biseccion_recursiva(m_min, m_mid - 1, prof + 1)
        else:
            # Explorar rama izquierda
            self.biseccion_recursiva(m_min, m_mid - 1, prof + 1)
            if not self.factor:
                self.biseccion_recursiva(m_mid + 1, m_max, prof + 1)

    def ejecutar(self):
        t0 = time.perf_counter()

        inicio = int(self.N**(1/4))
        fin = int(math.isqrt(self.N))

        self.biseccion_recursiva(inicio, fin)
        
        tf = time.perf_counter()
        if self.factor:
            print(f"🎯 FACTOR LOCALIZADO: {self.factor}")
            print(f"[*] Tiempo: {(tf-t0)*1000:.4f} ms")

import matplotlib.pyplot as plt
from fractions import Fraction
import math

class AnalizadorDienteSierra:
    def __init__(self, N):
        self.N = N
        self.m_raiz = int(math.isqrt(N))
        
    def obtener_fase(self, m):
        """ Calcula el valor de fase (distancia al centro del diente) """
        divisor = 2 * (2 * m + 3)
        return float(Fraction(self.N % divisor, divisor) - Fraction(1, 2))

    def analizar_segmento(self, m_inicio, ancho):
        """ 
        Escanea un segmento para graficar y seguir la señal.
        Aplica la bisección cinematica.
        """
        rango_m = np.arange(m_inicio, m_inicio + ancho)
        restos = []
        fases = []
        
        for m in rango_m:
            divisor = 2 * (2 * m + 3)
            restos.append(self.N % divisor)
            fases.append(self.obtener_fase(m))
            
        # Cálculo de cinemática en el punto inicial del segmento (4 puntos)
        y = fases[:4]
        v = y[1] - y[0]
        a = (y[2] - y[1]) - v
        
        return rango_m, restos, fases, v, a

    def graficar_seguimiento(self, m_centro, ventana=100):
        """ Grafica la señal observada y marca el vector director """
        rango, restos, fases, v, a = self.analizar_segmento(m_centro, ventana)
        
        fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 8), sharex=True)
        
        # Gráfica 1: Onda de Restos (Diente de Sierra Clásico)
        ax1.plot(rango, restos, color='blue', label='Onda de Restos (N % D)')
        ax1.set_title(f"Análisis de Señal para N = {self.N}")
        ax1.set_ylabel("Resto")
        ax1.grid(True, alpha=0.3)
        ax1.legend()

        # Gráfica 2: Onda de Fase (Cinemática)
        ax2.plot(rango, fases, color='red', label='Señal de Fase (Delta)')
        # Dibujar vector director (pendiente)
        ax2.arrow(rango[0], fases[0], 10, v*10, head_width=0.02, color='green', label='Vector Director (v)')
        
        # Predicción de la parábola
        if abs(a) > 1e-9:
            salto_predicho = -v / a
            ax2.scatter([m_centro + salto_predicho], [0], color='black', zorder=5, label='Mínimo Predicho')
        
        ax2.axhline(0, color='black', linestyle='--', alpha=0.5)
        ax2.set_ylabel("Fase (Distancia al 0.5)")
        ax2.set_xlabel("Coordenada m (Talla)")
        ax2.grid(True, alpha=0.3)
        ax2.legend()
        
        plt.tight_layout()
        plt.show()

Explicación de lo que estás viendo en la gráfica:
El Diente de Sierra (Arriba): Verás cómo la línea sube linealmente y "cae" bruscamente. Esa caída es el final de un ciclo del divisor.

La Señal de Fase (Abajo): Aquí es donde aplicamos tu teoría. La línea cruza el eje cero (punteado).

Si el Vector Director (flecha verde) apunta hacia abajo y la curva es suave, el algoritmo sabe que el factor está cerca.

El Mínimo Predicho: El punto negro es el resultado de tu fórmula Δm=−v/a. Es donde la matemática predice que el diente de sierra tocará el fondo absoluto.

Cómo usarlo para segmentar recursivamente:
Si ejecutas este código en diferentes partes de la "talla" (cambiando m_centro), observarás el Efecto Acordeón:

En segmentos alejados del factor, la flecha verde será casi horizontal (velocidad baja).

Al acercarte al factor, la pendiente se vuelve agresiva y la parábola se "marca" con nitidez.

Para implementar el Divide y Vencerás, el algoritmo simplemente mira el valor de v:

Si v < 0, el siguiente segmento a graficar y medir debe ser a la derecha.

Si v > 0, el factor quedó atrás, y segmentamos hacia la izquierda.

import math
import time
import matplotlib.pyplot as plt
import numpy as np
from fractions import Fraction

class MotorResonanciaMDC:
    def __init__(self, N):
        self.N = N
        self.encontrado = None
        self.historial_m = []
        self.historial_v = []

    def obtener_datos_fase(self, m):
        """
        Analiza 4 puntos para obtener el vector director (v)
        y la aceleración (a) del diente de sierra.
        """
        y = []
        for i in range(4):
            D = 2 * (2 * (m + i) + 3)
            if self.N % D == 0: return None, None, True
            # Señal de fase: Distancia relativa al centro del diente
            f = Fraction(self.N % D, D) - Fraction(1, 2)
            y.append(f)
        
        v = y[1] - y[0]  # Velocidad (Vector Director)
        a = (y[2] - y[1]) - v  # Aceleración (Curvatura de la parábola)
        return v, a, False

    def segmentar_recursivo(self, m_min, m_max, profundidad=0):
        if self.encontrado or m_min >= m_max:
            return

        m_mid = (m_min + m_max) // 2
        v, a, hit = self.obtener_datos_fase(m_mid)

        if hit:
            # Re-verificación para marcar el m exacto
            for i in range(4):
                D = 2 * (2 * (m_mid + i) + 3)
                if self.N % D == 0: self.encontrado = D
            return

        # Registro para graficación posterior
        self.historial_m.append(m_mid)
        self.historial_v.append(float(v))

        # --- LÓGICA DE DIVIDE Y VENCERÁS ---
        # Si el segmento es pequeño (< 100), aplicamos el disparo parabólico
        if (m_max - m_min) < 100:
            if a != 0:
                m_target = m_mid + int(-v / a)
                D_target = 2 * (2 * m_target + 3)
                if self.N % D_target == 0:
                    self.encontrado = D_target
            return

        # Decisión de segmentación por dirección del vector
        # v < 0 indica que el mínimo (factor) está hacia la derecha
        if v < 0:
            self.segmentar_recursivo(m_mid + 1, m_max, profundidad + 1)
            if not self.encontrado:
                self.segmentar_recursivo(m_min, m_mid - 1, profundidad + 1)
        else:
            self.segmentar_recursivo(m_min, m_mid - 1, profundidad + 1)
            if not self.encontrado:
                self.segmentar_recursivo(m_mid + 1, m_max, profundidad + 1)

    def ejecutar(self):
        inicio = int(self.N**(1/4))
        fin = int(math.isqrt(self.N))
        t0 = time.perf_counter()
        self.segmentar_recursivo(inicio, fin)
        t_final = (time.perf_counter() - t0) * 1000
        return self.encontrado, t_final

import math
import time
import numpy as np
import matplotlib.pyplot as plt
from fractions import Fraction

class MotorMDC_Resonancia:
    """
    Implementación del Divide y Vencerás por Seguimiento de Diente de Sierra.
    No es lineal: Segmenta la talla (m) mediante el Choque de Vectores.
    """
    def __init__(self, N):
        self.N = N
        self.encontrado = None
        self.pasos = 0

    def analizar_fase(self, m):
        """ Pinza de 4 puntos: Mide velocidad (v) y aceleración (a) """
        y = []
        for i in range(4):
            D = 2 * (2 * (m + i) + 3)
            if self.N % D == 0: return None, None, True
            # f = (N % D) / D - 0.5 (Señal de fase normalizada)
            y.append(Fraction(self.N % D, D) - Fraction(1, 2))
        
        v = y[1] - y[0] # Vector Director
        a = (y[2] - y[1]) - v # Curvatura (Aceleración)
        return v, a, False

    def segmentar_recursivo(self, m_min, m_max, profundidad=0):
        if self.encontrado or m_min >= m_max:
            return

        self.pasos += 1
        m_mid = (m_min + m_max) // 2
        v, a, hit = self.analizar_fase(m_mid)

        if hit:
            for i in range(4):
                D = 2 * (2 * (m_mid + i) + 3)
                if self.N % D == 0: self.encontrado = D
            return

        # UMBRAL DE DISPARO (Cuando el segmento es pequeño, usamos la parábola)
        if (m_max - m_min) < 50:
            if a != 0:
                m_target = m_mid + int(-v / a)
                D_t = 2 * (2 * m_target + 3)
                if self.N % D_t == 0: self.encontrado = D_t
            return

        # CHOQUE DE VECTORES: Decisión de Rama
        # Si v < 0, el factor está a la derecha (caída de fase).
        if v < 0:
            self.segmentar_recursivo(m_mid + 1, m_max, profundidad + 1)
            if not self.encontrado: # Si falló la derecha, probar izquierda
                self.segmentar_recursivo(m_min, m_mid - 1, profundidad + 1)
        else:
            self.segmentar_recursivo(m_min, m_mid - 1, profundidad + 1)
            if not self.encontrado:
                self.segmentar_recursivo(m_mid + 1, m_max, profundidad + 1)

1. El Tensor Cinemático de Fase (δ)
En lugar de buscar si N/D es entero, medimos el desfase respecto al punto crítico. Definimos la función de fase δ(m) como:

δ(m)= 
2(2m+3)

Interpretación: Si δ=0, el resto es exactamente la mitad del divisor. Si δ cruza el cero de forma limpia, hemos localizado el factor.

Dinámica: Evaluamos esta función en 4 puntos discretos para obtener sus derivadas:

Velocidad (v): Δδ/Δm (Pendiente local).

Aceleración (a): Δv/Δm (Curvatura de la parábola).
