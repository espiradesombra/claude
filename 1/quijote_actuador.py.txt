Afegeix al gemell_3vs7_rigoros.py el càlcul explícit de:
  - F_centrífuga per aspa (m_q · ω² · r_k)
  - F_gravitatoria radial (m_q · g · cos(θ_k))
  - P_actuador_k = F_net · dr_dt (potència instantània del bombeig)
  - Balanç net: P_hurto - P_actuadors = guany real verificat

Resposta a la crítica clàssica del camp conservatiu.
"""

import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import matplotlib.gridspec as gridspec

SEED = 42
np.random.seed(SEED)

# ── Paràmetres NREL 5MW ───────────────────────────────────────────────────────
R        = 63.0
P_NOM    = 5.0e6
RHO_A    = 1.225
CP_MAX   = 0.482
LAM_OPT  = 7.55
A_ROT    = np.pi*R**2
V_RATED  = (P_NOM/(0.5*RHO_A*A_ROT*CP_MAX))**(1/3)
OM_RATED = LAM_OPT*V_RATED/R
G        = 9.81

# ── Paràmetres Quijote ────────────────────────────────────────────────────────
RHO_FL   = 3386.0
D_CANAL  = 0.05
A_CANAL  = np.pi*(D_CANAL/2)**2
R_HUB    = 5.0
R_TIP    = 55.0
DR_MAX   = 0.5
K_Q      = 5.0e4
C_FRIC   = 80.0
K_COMP   = 0.10

# ── Simulació ─────────────────────────────────────────────────────────────────
V_BASE   = 11.4
T_TOT    = 90.0
DT       = 0.05
STEPS    = int(T_TOT/DT)
T_VEC    = np.arange(STEPS)*DT
K_KUR    = 0.10

np.random.seed(SEED)
ou = np.zeros(STEPS)
for i in range(1, STEPS):
    ou[i] = ou[i-1] - 0.5*ou[i-1]*DT + 0.8*np.sqrt(DT)*np.random.randn()

def v_vent(s):
    t = s*DT
    v = V_BASE + 2*np.sin(2*np.pi*t/20) + ou[s]*0.6
    if 40 < t < 60:
        v -= 5*np.sin(np.pi*(t-40)/20)
    return max(3.0, float(v))

V_ARR = np.array([v_vent(s) for s in range(STEPS)])

def m_q_pala():
    return RHO_FL * A_CANAL * (R_TIP - R_HUB)

def eta_cp(omega, v):
    if v <= 0: return 0.0
    lam = omega*R/v
    return float(max(0, 1 - ((lam - LAM_OPT)/LAM_OPT)**2))

# ── Simulació principal amb auditoria d'actuador ──────────────────────────────
def simula(N_pales):
    m_q = m_q_pala()

    thetas = np.array([2*np.pi*k/N_pales for k in range(N_pales)])
    omega  = OM_RATED * 0.95
    r_q    = np.full(N_pales, (R_HUB+R_TIP)/2)
    dr_q   = np.zeros(N_pales)

    # Buffer hidràulic
    V_BUF_MAX = 0.10
    Q_BOMBA   = 0.010
    P_ACC_MAX = 20e6
    LLINDAR   = 20.0
    ETA_GEN   = 0.85
    V_buf = 0.0
    P_acc = 0.0

    # Historials
    hPa   = np.zeros(STEPS)
    hPg   = np.zeros(STEPS)
    hPbuf = np.zeros(STEPS)
    hPgen = np.zeros(STEPS)
    hom   = np.zeros(STEPS)

    # ── NOU: historials d'actuador ────────────────────────────────────────────
    hP_act_total  = np.zeros(STEPS)   # Potència total actuadors (W)
    hF_cent_total = np.zeros(STEPS)   # Força centrífuga total (N)
    hF_grav_total = np.zeros(STEPS)   # Component gravitatoria total (N)
    hP_hurto      = np.zeros(STEPS)   # Potència hurto (T_hurto · ω)
    hP_net        = np.zeros(STEPS)   # P_hurto - P_actuadors = guany real

    # Acumuladors energètics
    E_vent    = 0.0
    E_xarxa   = 0.0
    E_frec    = 0.0
    E_gen     = 0.0
    E_act_pos = 0.0   # Energia consumida per actuadors (moviment actiu)
    E_act_neg = 0.0   # Energia retornada per actuadors (moviment passiu)
    E_hurto   = 0.0   # Energia del par hurto

    for s in range(STEPS):
        v = V_ARR[s]

        om_target = min(OM_RATED, LAM_OPT*max(v, 0.1)/R)
        omega += (om_target - omega)*(DT/3.0)
        omega = float(np.clip(omega, OM_RATED*0.3, OM_RATED*1.4))

        etL = eta_cp(omega, v)
        Pa  = 0.5*RHO_A*A_ROT*CP_MAX*etL*v**3

        dJ_total   = 0.0
        T_hurto_s  = 0.0   # Par hurto gravitatori total en aquest pas
        P_act_s    = 0.0   # Potència total actuadors en aquest pas
        F_cent_s   = 0.0
        F_grav_s   = 0.0

        for i in range(N_pales):
            th = thetas[i] % (2*np.pi)

            r_opt0 = R_TIP - (R_TIP-R_HUB)*(1+np.cos(th))/2
            F_c_i  = m_q * omega**2 * r_q[i]
            r_opt  = r_opt0 + K_COMP*(F_c_i/(K_Q*(R_TIP-R_HUB)+1e-9))
            r_opt  = float(np.clip(r_opt, R_HUB, R_TIP))

            F_ctrl = -K_Q*(r_q[i] - r_opt)
            F_fric_r = -C_FRIC*dr_q[i]
            acc    = (F_c_i + F_ctrl + F_fric_r)/m_q

            dr_q[i] += acc*DT
            dr_q[i] = float(np.clip(dr_q[i], -DR_MAX, DR_MAX))
            lim_pos = omega * r_q[i]
            dr_q[i] = float(np.clip(dr_q[i], -lim_pos, lim_pos))
            r_q[i]  = float(np.clip(r_q[i] + dr_q[i]*DT, R_HUB, R_TIP))

            # ── NOVES LÍNIES GEMINI: Auditoria actuador ───────────────────────
            # 1. Força centrífuga sobre la massa
            F_cent_i = m_q * omega**2 * r_q[i]

            # 2. Component gravitatoria radial (en la direcció radial de l'aspa)
            # cos(th) > 0 → aspa baixant (gravetat assisteix extensió)
            # cos(th) < 0 → aspa pujant (gravetat s'oposa a extensió)
            F_grav_i = m_q * G * np.cos(th)

            # 3. Força neta que l'actuador ha de vèncer
            # El control Quijote mou el pes CONTRA la posició d'equilibri natural
            # dr_q[i] > 0 → pes cap a fora (assistit per F_cent, afectat per F_grav)
            # dr_q[i] < 0 → pes cap a dins (combatent F_cent)
            F_act_i = F_cent_i + F_grav_i

            # 4. Potència instantània de l'actuador per l'aspa i
            # P = F · v (positiu = consumeix energia, negatiu = regenera)
            P_act_i = F_act_i * dr_q[i]

            # 5. Par hurto gravitatori d'aquesta aspa (contribució al par net)
            # T_hurto,k = m_q · g · r_k · cos(θ_k) (Ec. 1 del paper)
            T_hurto_i = m_q * G * r_q[i] * np.cos(th)

            # Acumulació
            P_act_s   += P_act_i
            F_cent_s  += F_cent_i
            F_grav_s  += abs(F_grav_i)
            T_hurto_s += T_hurto_i

            # Balanç energètic actuador
            if P_act_i > 0:
                E_act_pos += P_act_i * DT  # J consumits per actuadors
            else:
                E_act_neg += abs(P_act_i) * DT  # J recuperats per actuadors

            dJdt_i = 2*m_q*r_q[i]*dr_q[i]
            dJ_total += dJdt_i
            E_frec += (-C_FRIC*dr_q[i]**2)*DT

        # Potència hurto total: T_hurto · ω
        P_hurto_s = T_hurto_s * omega
        E_hurto  += P_hurto_s * DT

        # Guany net real: P_hurto - P_actuadors (ha de ser > 0 per al guany)
        P_net_s = P_hurto_s - P_act_s

        # Emmagatzemar historials actuador
        hP_act_total[s]  = P_act_s
        hF_cent_total[s] = F_cent_s
        hF_grav_total[s] = F_grav_s
        hP_hurto[s]      = P_hurto_s
        hP_net[s]        = P_net_s

        # Buffer i xarxa (idèntic al codi original)
        P_buf = -0.5*dJ_total*omega**2

        K_kur = K_KUR
        for i in range(N_pales):
            kc  = (K_kur/N_pales)*float(np.sum(np.sin(thetas - thetas[i])))
            dJi = 2*m_q*r_q[i]*dr_q[i]
            pm  = float(np.clip(2.0*np.cos(thetas[i]) + 0.5*omega*np.sin(thetas[i])
                                - 0.6*dJi, -8, 8))
            thetas[i] += (omega + kc + pm*0.01)*DT

        Pgen = 0.0
        if P_buf > LLINDAR and V_buf < V_BUF_MAX:
            P_acc = min(P_acc + P_buf*DT/V_BUF_MAX, P_ACC_MAX)
            Q     = min(Q_BOMBA, P_buf/max(P_acc, 1e3))
            V_buf = min(V_BUF_MAX, V_buf + Q*DT)
        elif P_buf < -LLINDAR and V_buf > 0:
            P_acc = max(P_acc + P_buf*DT/V_BUF_MAX, 0)
            Q     = min(Q_BOMBA, -P_buf/max(P_acc, 1e3))
            Pgen  = P_acc*Q*ETA_GEN
            V_buf = max(0, V_buf - Q*DT)

        Pg = max(0, Pa*(1+0.04) + max(0, P_buf) + Pgen)

        E_vent  += Pa*DT
        E_xarxa += Pg*DT
        E_gen   += Pgen*DT

        hPa[s]   = Pa
        hPg[s]   = Pg
        hPbuf[s] = P_buf
        hPgen[s] = Pgen
        hom[s]   = omega

    millora = (np.mean(hPg) - np.mean(hPa))/np.mean(hPa)*100

    return {
        'Pa': hPa, 'Pg': hPg, 'Pbuf': hPbuf, 'Pgen': hPgen, 'om': hom,
        # Nous camps auditoria
        'P_act': hP_act_total,
        'P_hurto': hP_hurto,
        'P_net': hP_net,
        'F_cent': hF_cent_total,
        'F_grav': hF_grav_total,
        # Acumuladors energètics
        'E_vent':    E_vent/3.6e6,
        'E_xarxa':   E_xarxa/3.6e6,
        'E_frec':    E_frec/3.6e6,
        'E_gen':     E_gen/3.6e6,
        'E_act_pos': E_act_pos/3.6e6,
        'E_act_neg': E_act_neg/3.6e6,
        'E_hurto':   E_hurto/3.6e6,
        'millora': millora,
        'N': N_pales,
        'm_q': m_q_pala()
    }

# ── Execució ──────────────────────────────────────────────────────────────────
print("Simulant 3 pales (amb auditoria actuador)...")

print("Simulant 7 pales (amb auditoria actuador)...")
