PythonCopyS = isqrt(N-9) - 3
while abs(Δ) > 0:
    Δ = S*S + 6*S - (N-9)
    if is_square(Δ):  # bingo
        k = sqrt(Δ)
        v, b = (S+k)//2, (S-k)//2
        return 2*v+3, 2*b+3
    curv = (Δ(S+2) - 2*Δ(S) + Δ(S-2)) / 4  # approx segunda derivada
    if curv > 0:  # cóncava arriba, cerca del mínimo
        S += round( -Δ / (curv*2) )  # Newton geométrico
    else:
        S += 2 if Δ > 0 else 1

vbaCopySub GenerarGraficasVariandoSlash()
    Dim i As Integer
    Dim slash As Double
    Dim rngDatos As Range
    Dim ch As Chart
    
    Set rngDatos = Sheets("Datos").Range("A2:B50")  ' tus datos base
    
    For i = 1 To 10  ' varia 10 veces, o cambia a más
        slash = 1 + i * 0.5  ' ejemplo: 1, 1.5, 2... hasta 6
        
        ' Actualiza la columna de valores según slash
        Sheets("Datos").Range("C2:C50").Formula = "=1 / (" & slash & " * A2)"  ' tu fórmula aquí
        
        ' Crea nueva gráfica
        Set ch = Charts.Add
        With ch
            .SetSourceData Source:=Sheets("Datos").Range("A2:C50")
            .ChartType = xlXYScatterSmooth
            .HasTitle = True
            .ChartTitle.Text = "Curva con slash = " & Format(slash, "0.0")
            .Axes(xlCategory).HasTitle = True
            .Axes(xlCategory).AxisTitle.Text = "X"
            .Axes(xlValue).HasTitle = True
            .Axes(xlValue).AxisTitle.Text = "Y (depende de slash)"
        End With
        
        ' Guarda como imagen o deja en hoja nueva
        ch.Location xlLocationAsNewSheet, "Grafica_" & i
    Next i
    
    MsgBox "¡Listo! 10 gráficas generadas."
End Sub

# Factorización basada en diente de sierra + pinza simple
# No usa floats: todo entero

def d_frac(N, m):
    denom = 2 * (2*m + 3)
    return (N % denom) / denom  # float pequeño (seguro aquí)

def pinza_ok(fracs, tol=0.05):
    # 2 puntos izquierda, 2 derecha
    left = fracs[:2]
    right = fracs[-2:][::-1]

    # velocidad media
    v_l = left[1] - left[0]
    v_r = right[1] - right[0]

    return abs(v_l + v_r) < tol  # simetría

def buscar_factor(N, m_ini=1, pasos=10000):
    m = m_ini

    for _ in range(pasos):
        # 4 puntos
        ms = [m+i for i in range(4)]
        fracs = [d_frac(N, mi) for mi in ms]

        # detectar descenso
        if fracs[1] < fracs[0] and fracs[2] < fracs[1]:
            if pinza_ok(fracs):
                # candidato
                denom = 2*(2*m+3)
                if N % denom == 0:
                    return denom

        m += 1

    return None

def parabola_vertex(x, y):
    # usa 3 puntos
    x1, x2, x3 = x
    y1, y2, y3 = y

    denom = (x1-x2)*(x1-x3)*(x2-x3)
    if denom == 0:
        return None

    a = (y1*(x2-x3) + y2*(x3-x1) + y3*(x1-x2)) / denom
    b = (y1*(x3**2-x2**2) + y2*(x1**2-x3**2) + y3*(x2**2-x1**2)) / denom

    if a == 0:
        return None

    return -b/(2*a)


def buscar_factor_salto(N, m_ini=1):
    m = m_ini

    while True:
        ms = [m, m+1, m+2]
        fracs = [d_frac(N, mi) for mi in ms]

        # descenso
        if fracs[1] < fracs[0] and fracs[2] < fracs[1]:
            mv = parabola_vertex(ms, fracs)
            if mv is not None:
                mv = int(round(mv))
                denom = 2*(2*mv+3)
                if denom > 0 and N % denom == 0:
                    return denom
                m = mv  # salto
                continue

        m += 1

# ================================
# CONFIG
# ================================
N = 10403  # cambia por tu número
MAX_M = 10000
LOG_FILE = "dientes_log.txt"

# ================================
# CORE: decimal exacto
# ================================
def frac_decimal_str(N, m):
    denom = 2*(2*m+3)
    num = N % denom

    # devolvemos decimal como string (sin float)
    return f"{num}/{denom}", num, denom

# ================================
# BUFFER A TXT
# ================================
def log_line(m, frac_str):
    with open(LOG_FILE, "a") as f:
        f.write(f"{m},{frac_str}\n")

# ================================
# MEDICIÓN PINZA
# ================================
def medir_pinza(N, m):
    desc = []
    asc = []

    # descenso (hacia adelante)
    for i in range(4):
        frac_str, num, den = frac_decimal_str(N, m+i)
        desc.append((m+i, num, den))
        log_line(m+i, frac_str)

    # ascenso (hacia atrás)
    for i in range(4):
        frac_str, num, den = frac_decimal_str(N, m-i)
        asc.append((m-i, num, den))
        log_line(m-i, frac_str)

    asc.reverse()  # alinear con desc

    return desc, asc

# ================================
# COMPARAR PARÁBOLA (SIMETRÍA)
# ================================
def misma_parabola(desc, asc, tol=0.05):
    # convertimos a float solo para comparación local
    def val(x): return x[1] / x[2]

    v_desc = [val(desc[i+1]) - val(desc[i]) for i in range(3)]
    v_asc  = [val(asc[i+1]) - val(asc[i]) for i in range(3)]

    # simetría global
    diffs = [abs(v_desc[i] + v_asc[i]) for i in range(3)]
    return sum(diffs)/len(diffs) < tol

# ================================
# DETECCIÓN DE CERO
# ================================
def toca_cero(puntos):
    for (_, num, _) in puntos:
        if num == 0:
            return True
    return False

# ================================
# ALGORITMO PRINCIPAL
# ================================
def factorizar(N):
    open(LOG_FILE, "w").close()  # limpiar log

    m = 1

    while m < MAX_M:

        desc, asc = medir_pinza(N, m)

        # 1. comprobar si misma parábola
        if misma_parabola(desc, asc):

            # 2. comprobar cero en descenso
            if toca_cero(desc):
                denom = desc[0][2]
                return denom

            # 3. comprobar cero en ascenso
            if toca_cero(asc):
                denom = asc[0][2]
                return denom

            # 4. salto (parábola validada pero no cero)
            m += 4
            continue

        else:
            # no misma parábola → saltar a siguiente diente
            m += 1

    return None


# ================================
# TEST
# ================================
if __name__ == "__main__":
    f = factorizar(N)
    print("Factor encontrado:", f)

# ==========================================
# CONFIGURACIÓN
# ==========================================
N = 10403
MAX_ITER = 100000
LOG_FILE = "mdc_log.txt"

# ==========================================
# CORE: FRACCIÓN EXACTA
# ==========================================
def frac(N, m):
    denom = 2*(2*m+3)
    num = N % denom
    return num, denom

# ==========================================
# LOG
# ==========================================
def log_line(m, num, denom):
    with open(LOG_FILE, "a") as f:
        f.write(f"{m},{num},{denom}\n")

# ==========================================
# SEGUNDA DIFERENCIA (PARÁBOLA)
# ==========================================
def segunda_diferencia(p):
    # p = [(m,num,denom), ...]
    # trabajamos en numerador normalizado
    vals = [p[i][1] / p[i][2] for i in range(len(p))]
    return vals[2] - 2*vals[1] + vals[0]

# ==========================================
# PINZA
# ==========================================
def medir_pinza(N, m):
    desc = []
    asc = []

    for i in range(3):
        num, den = frac(N, m+i)
        desc.append((m+i, num, den))
        log_line(m+i, num, den)

    for i in range(3):
        num, den = frac(N, m-i)
        asc.append((m-i, num, den))
        log_line(m-i, num, den)

    asc.reverse()

    return desc, asc

# ==========================================
# MISMA PARÁBOLA (ROBUSTO)
# ==========================================
def misma_parabola(desc, asc, tol=1e-12):
    d2_desc = segunda_diferencia(desc)
    d2_asc  = segunda_diferencia(asc)

    return abs(d2_desc - d2_asc) < tol

# ==========================================
# DETECCIÓN EXACTA DE FACTOR
# ==========================================
def es_factor(num):
    return num == 0

# ==========================================
# VELOCIDAD DISCRETA
# ==========================================
def velocidad(p):
    # p = [(m,num,den)...]
    return (p[1][1]/p[1][2]) - (p[0][1]/p[0][2])

# ==========================================
# SALTO DIFERENCIAL
# ==========================================
def salto(desc):
    y = desc[0][1] / desc[0][2]
    v = velocidad(desc)

    if v == 0:
        return 1

    salto = int(abs(y / v))
    return max(1, salto)

# ==========================================
# ALGORITMO PRINCIPAL
# ==========================================
def factorizar(N):
    open(LOG_FILE, "w").close()

    m = 1
    iteraciones = 0

    while m < MAX_ITER:

        desc, asc = medir_pinza(N, m)

        # 1. ¿factor exacto?
        for p in desc + asc:
            if es_factor(p[1]):
                return 2*(2*p[0]+3)

        # 2. ¿misma parábola?
        if misma_parabola(desc, asc):

            # salto inteligente
            dm = salto(desc)
            m += dm

        else:
            # avanzar diente
            m += 1

        iteraciones += 1

    return None


# ==========================================
# TEST
# ==========================================
if __name__ == "__main__":
    f = factorizar(N)
    print("Factor encontrado:", f)
