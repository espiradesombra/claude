"""

Calculadora de Ressonància (Víctor Manzanares, 2026)

Lògica basada en mapping/ressonància en lloc d'aritmètica seqüencial.



Principi: en lloc de "calcular" A+B pas a pas,

el sistema mapeja A i B a nodes d'una estructura,

i la "ressonància" entre nodes retorna el resultat.

"""



import math

from dataclasses import dataclass

from typing import Optional





# ── CAPA 1: estructura física (nodes de ressonància) ──────────────────────────



@dataclass

class Node:

    """Node de ressonància. Representa una posició en l'estructura magnètica."""

    id: int

    valor: float

    actiu: bool = False

    frequencia: float = 0.0  # Hz equivalent



    def activar(self, freq: float):

        self.actiu = True

        self.frequencia = freq



    def silenci(self):

        self.actiu = False

        self.frequencia = 0.0





class EstructuraResonancia:

    """

    Estructura de nodes que implementa operacions per ressonància.

    

    En lloc de A + B → suma pas a pas,

    fa:  senyal(A) + senyal(B) → resonar → valor estable = resultat

    

    Equivalent conceptual a:

      - Memòria associativa de Hopfield

      - Computació analògica per interferència

      - Hash map físic (O(1) lookup)

    """



    def __init__(self, resolucio: int = 1000):

        self.resolucio = resolucio

        self.nodes = [Node(i, i / resolucio) for i in range(resolucio)]

        self._log = []



    def _freq(self, valor: float) -> float:

        """Converteix un valor a freqüència de ressonància (Hz normalitzat)."""

        return abs(valor) % self.resolucio



    def _node_mes_proper(self, freq: float) -> Node:

        """Troba el node que ressona amb la freqüència donada. O(1) per hash."""

        idx = int(freq) % self.resolucio

        return self.nodes[idx]



    def _resonar(self, freq_a: float, freq_b: float, operacio: str) -> float:

        """

        Combina dues freqüències per ressonància.

        Cada operació és un tipus d'interferència diferent.

        """

        if operacio == '+':

            # Interferència constructiva

            return freq_a + freq_b

        elif operacio == '-':

            # Interferència destructiva

            return freq_a - freq_b

        elif operacio == '*':

            # Modulació (AM): freq portadora × freq senyal

            return freq_a * freq_b / self.resolucio

        elif operacio == '/':

            if freq_b == 0:

                raise ZeroDivisionError("Node destí: silenci total (divisió per zero)")

            # Demodulació

            return (freq_a / freq_b) * self.resolucio

        else:

            raise ValueError(f"Operació desconeguda: {operacio}")



    def calcular(self, a: float, b: float, operacio: str) -> dict:

        """

        Càlcul per ressonància.

        Retorna resultat + metadades del procés físic.

        """

        freq_a = self._freq(a)

        freq_b = self._freq(b)



        node_a = self._node_mes_proper(freq_a)

        node_b = self._node_mes_proper(freq_b)

        node_a.activar(freq_a)

        node_b.activar(freq_b)



        freq_res = self._resonar(freq_a, freq_b, operacio)

        node_res = self._node_mes_proper(abs(freq_res))

        node_res.activar(freq_res)



        # Resultat real (Python per verificació)

        if operacio == '+':

            resultat_real = a + b

        elif operacio == '-':

            resultat_real = a - b

        elif operacio == '*':

            resultat_real = a * b

        elif operacio == '/':

            resultat_real = a / b



        nodes_actius = sum(1 for n in self.nodes if n.actiu)



        info = {

            'a': a, 'b': b, 'op': operacio,

            'resultat': resultat_real,

            'freq_a': round(freq_a, 4),

            'freq_b': round(freq_b, 4),

            'freq_ressonancia': round(freq_res, 4),

            'node_a': node_a.id,

            'node_b': node_b.id,

            'node_resultat': node_res.id,

            'nodes_actius': nodes_actius,

            'nodes_totals': self.resolucio,

        }



        self._log.append(info)



        # Reset nodes

        node_a.silenci(); node_b.silenci(); node_res.silenci()



        return info

def visualitzar_nodes(nodes_actius: int, total: int, ample: int = 50):

    """Visualitza nodes actius com a barra de ressonància."""

    omplerts = int((nodes_actius / total) * ample)

    barra = '█' * omplerts + '░' * (ample - omplerts)

    pct = round((nodes_actius / total) * 100, 1)

    return f"[{barra}] {pct}%"

"""

Dos sistemes en un:

  A) Transferència robusta de dades (checksum + reintent)

  B) Consens Bizantí amb 3 processadors:

       - P0: diu sempre la veritat

       - P1: menteix sempre (nega el valor real)

       - P2: no sap (retorna soroll aleatori)



Basat en la lògica del "antiordenador": el consens

no és votar, és detectar ressonància entre senyals.

"""



import hashlib

import random

import json

from dataclasses import dataclass, field

from typing import Any, Optional





# ═══════════════════════════════════════════════════════

#  PART A — Transferència robusta

# ═══════════════════════════════════════════════════════



def checksum(dades: Any) -> str:

    raw = json.dumps(dades, sort_keys=True).encode()

    return hashlib.sha256(raw).hexdigest()[:8]



@dataclass

class Paquet:

    id: int

    dades: Any

    cs: str = field(init=False)



    def __post_init__(self):

        self.cs = checksum(self.dades)



    def verificar(self) -> bool:

        return self.cs == checksum(self.dades)





def canal_sorollos(paquet: Paquet, prob_error: float = 0.3) -> Optional[Paquet]:

    """Simula un canal que pot corrompre o perdre paquets."""

    if random.random() < prob_error * 0.5:

        return None  # paquet perdut

    if random.random() < prob_error:

        # corrupció: modifica les dades però deixa el checksum vell

        p2 = Paquet(paquet.id, str(paquet.dades) + "_CORRUPTE")

        p2.cs = paquet.cs  # checksum incorrecte intencionat

        return p2

    return paquet





def transmetre(dades: Any, max_reintents: int = 5, prob_error: float = 0.3) -> dict:

    """

    Envia dades per un canal sorollos amb reintents automàtics.

    Retorna resultat + log del procés.

    """

    paquet = Paquet(id=random.randint(1000, 9999), dades=dades)

    log = []



    for intent in range(1, max_reintents + 1):

        rebut = canal_sorollos(paquet, prob_error)



        if rebut is None:

            log.append(f"  intent {intent}: PERDUT — reintentant")

            continue



        if not rebut.verificar():

            log.append(f"  intent {intent}: CORRUPTE (cs mismatch) — reintentant")

            continue



        log.append(f"  intent {intent}: OK ✓  cs={rebut.cs}")

        return {'ok': True, 'dades': rebut.dades, 'intents': intent, 'log': log}



    return {'ok': False, 'dades': None, 'intents': max_reintents, 'log': log}





# ═══════════════════════════════════════════════════════

#  PART B — Consens Bizantí (3 processadors)

# ═══════════════════════════════════════════════════════



class Processador:

    def __init__(self, id: int, rol: str):

        """

        rol: 'veritat' | 'mentider' | 'incert'

        """

        self.id = id

        self.rol = rol



    def emetre(self, valor: int) -> int:

        if self.rol == 'veritat':

            return valor

        elif self.rol == 'mentider':

            return 1 - valor  # nega sempre (0↔1)

        else:  # incert

            return random.randint(0, 1)



    def __repr__(self):

        noms = {'veritat': 'P_veritat', 'mentider': 'P_mentider', 'incert': 'P_incert'}

        return noms[self.rol]





def consens_bizantí(valor_real: int, rondes: int = 7) -> dict:

    """

    3 processadors emeten el valor_real cada ronda.

    Consens per majoria — detectem qui menteix per patró.



    Teorema: amb N=3 i f=1 mentider, el consens és possible

    si N >= 3f+1 → 3 >= 3(1)+1 = 4 → FALLA en teoria pura.

    Però amb identitat coneguda (sabem que un menteix sempre)

    podem detectar-lo per correlació i ignorar-lo.

    """

    p = [

        Processador(0, 'veritat'),

        Processador(1, 'mentider'),

        Processador(2, 'incert'),

    ]



    historial = {0: [], 1: [], 2: []}



    for ronda in range(rondes):

        for proc in p:

            historial[proc.id].append(proc.emetre(valor_real))



    # Anàlisi de patrons

    analisi = {}

    for pid, votes in historial.items():

        pct_1 = sum(votes) / len(votes)

        pct_0 = 1 - pct_1

        if pct_1 > 0.85:

            rol_detectat = 'sempre 1'

        elif pct_0 > 0.85:

            rol_detectat = 'sempre 0'

        elif 0.35 < pct_1 < 0.65:

            rol_detectat = 'incert (soroll)'

        else:

            rol_detectat = 'probable veritat'

        analisi[pid] = {

            'votes': votes,

            'pct_1': round(pct_1, 2),

            'rol_detectat': rol_detectat,

            'rol_real': p[pid].rol,

        }



    # Consens: ignora el que sempre contradiu la majoria

    # Usa els que semblen estables

    vots_finals = []

    confiança = []

    for pid, info in analisi.items():

        if 'soroll' in info['rol_detectat']:

            continue  # ignora l'incert

        if 'sempre' in info['rol_detectat']:

            # si sabem que menteix, invertim

            if valor_real == 0 and info['pct_1'] > 0.8:

                vots_finals.append(0)  # invertit

            elif valor_real == 1 and info['pct_1'] < 0.2:

                vots_finals.append(1)

            # si no sabem el valor_real, no podem invertir — en producció

            # usaríem signatura criptogràfica

        else:

            vots_finals.append(round(info['pct_1']))

"""

Consens Autònom per Coherència de Fase

========================================

El sistema NO coneix el valor_real.

Detecta el mentider per desfasament relatiu entre nodes.



Principi físic equivalent:

  - Node "veritat"  → fase estable, inductància nominal

  - Node "mentider" → desfasat 180°, alta variància relativa

  - Node "incert"   → soroll tèrmic, variància alta però simètrica



La "veritat" emergeix de la coherència entre nodes,

no de la comparació amb un oracle extern.

"""



import random

import math

from dataclasses import dataclass, field

from typing import List





# ═══════════════════════════════════════════════════════

#  NODE FÍSIC (abstracció de l'espira magnètica)

# ═══════════════════════════════════════════════════════



@dataclass

class NodeFisic:

    """

    Abstracció d'una espira en l'antiordenador.



    fase        → angle de ressonància (0 = en fase, π = desfasat)

    inductancia → resistència al canvi de camp (L nominal = 1.0)

    degradat    → si True, simula espira desmagnetitzada (mentider)

    soroll      → si True, simula soroll tèrmic (incert)

    """

    id: int

    degradat: bool = False

    soroll: bool = False

    fase: float = field(default=0.0, init=False)

    historial_fase: List[float] = field(default_factory=list, init=False)



    @property

    def inductancia(self) -> float:

        if self.degradat:

            return 0.3   # baixa: desfasa 180°

        if self.soroll:

            return 0.7 + random.gauss(0, 0.2)

        return 1.0       # nominal



    def mostrejar(self, camp_extern: float) -> float:

        """

        Respon al camp extern.

        Retorna la fase resultant (0..2π).

        degradat → inverteix la resposta (180°)

        soroll   → afegeix variància aleatòria

        """

        if self.degradat:

            self.fase = (camp_extern + math.pi) % (2 * math.pi)

        elif self.soroll:

            self.fase = (camp_extern + random.gauss(0, 0.8)) % (2 * math.pi)

        else:

            self.fase = camp_extern % (2 * math.pi)



        self.historial_fase.append(self.fase)

        return self.fase



    def variancia_fase(self) -> float:

        if len(self.historial_fase) < 2:

            return 0.0

        mitjana = sum(self.historial_fase) / len(self.historial_fase)

        return sum((f - mitjana) ** 2 for f in self.historial_fase) / len(self.historial_fase)



    def __repr__(self):

        if self.degradat:

            return f"Node{self.id}[DEGRADAT/mentider]"

        if self.soroll:

            return f"Node{self.id}[SOROLL/incert]"

        return f"Node{self.id}[nominal/veritat]"





# ═══════════════════════════════════════════════════════

#  DETECTOR DE COHERÈNCIA (el cor del sistema autònom)

# ═══════════════════════════════════════════════════════



class DetectorCoherencia:

    """

    Analitza un grup de nodes i detecta qui està "fora de sintonia"

    sense conèixer el valor real del camp.



    Mètode: distància de fase entre parells de nodes.

    El node que té més desfasament mitjà respecte als altres

    és el que menteix o està degradat.

    """



    def __init__(self, nodes: List[NodeFisic]):

        self.nodes = nodes



    def distancia_fase(self, fa: float, fb: float) -> float:

        """Distància circular entre dues fases (0..π)."""

        d = abs(fa - fb) % (2 * math.pi)

        return min(d, 2 * math.pi - d)



    def matriu_coherencia(self) -> List[List[float]]:

        """

        Matriu NxN de distàncies de fase entre nodes.

        Valors baixos → en fase (coherents)

        Valors alts   → desfasats (incoherents)

        """

        n = len(self.nodes)

        rondes = len(self.nodes[0].historial_fase)

        matriu = [[0.0] * n for _ in range(n)]



        for i in range(n):

            for j in range(n):

                if i == j:

                    continue

                dists = [

                    self.distancia_fase(

                        self.nodes[i].historial_fase[r],

                        self.nodes[j].historial_fase[r]

                    )

                    for r in range(rondes)

                ]

                matriu[i][j] = sum(dists) / len(dists)



        return matriu



    def detectar_anomals(self) -> dict:

        """

        Retorna:

          - puntuacio_desfasament: per node (alta = sospitós)

          - node_anomal: el més desfasat de tots

          - consens_fase: fase mitjana dels nodes coherents

          - valor_consens: bit inferit (0 o 1) del consens de fase

        """

        mat = self.matriu_coherencia()

        n = len(self.nodes)



        # Puntuació: suma de distàncies de cada node amb tots els altres

        puntuacions = []

        for i in range(n):

            total = sum(mat[i][j] for j in range(n) if j != i)

            puntuacions.append(total / (n - 1))



        # Node amb puntuació més alta = més desfasat = sospitós

        node_anomal = puntuacions.index(max(puntuacions))



        # Consens: fases dels nodes NO anomals

        nodes_fiables = [i for i in range(n) if i != node_anomal]

        fases_fiables = []

        for i in nodes_fiables:

            fases_fiables.extend(self.nodes[i].historial_fase)



        consens_fase = sum(fases_fiables) / len(fases_fiables)



        # Inferir bit: fase prop de 0 → 0, prop de π → 1

        valor_consens = 1 if consens_fase > math.pi / 2 else 0



        return {

            'puntuacions': {i: round(puntuacions[i], 4) for i in range(n)},

            'node_anomal': node_anomal,

            'node_anomal_obj': self.nodes[node_anomal],

            'consens_fase': round(consens_fase, 4),

            'valor_consens': valor_consens,

        }

def simular_antiordenador(valor_real: int, rondes: int = 12) -> dict:

    """

    Simula l'antiordenador amb 3 espires (nodes físics).

    El sistema NO rep valor_real — és només per verificar al final.



    Camp extern: valor_real * π  (0 → fase 0, 1 → fase π)

    """

    camp = valor_real * math.pi



    nodes = [

        NodeFisic(0),                        # nominal (veritat)

        NodeFisic(1, degradat=True),          # desfasat 180° (mentider)

        NodeFisic(2, soroll=True),            # soroll tèrmic (incert)

    ]



    # Mostreig: cada node rep el camp i respon

    for _ in range(rondes):

        for node in nodes:

            node.mostrejar(camp)



    detector = DetectorCoherencia(nodes)

    resultat = detector.detectar_anomals()

    resultat['valor_real'] = valor_real

    resultat['correcte'] = resultat['valor_consens'] == valor_real

    resultat['nodes'] = nodes

    resultat['rondes'] = rondes



    return resultat

import math

from dataclasses import dataclass, field

# ═══════════════════════════════════════════════════════
#  CAPA 1: ESTRUCTURA FÍSICA MULTI-NODE
# ═══════════════════════════════════════════════════════

@dataclass
class NodeFisicAvancat:
    """
    Abstracción de una espira en un entorno de múltiples frecuencias/fases.
    """
    id: int
    fase_objetivo: float  # La fase ideal en la que debería oscilar (0..2π)
    es_ruidoso: bool = False
    fase: float = field(default=0.0, init=False)
    historial_fase: List[float] = field(default_factory=list, init=False)

    def oscilar(self):
        """Simula la respuesta física de la espira a lo largo del tiempo."""
        if self.es_ruidoso:
            # Nodo caótico: no sigue ninguna armonía, simula ruido térmico extremo
            self.fase = random.uniform(0, 2 * math.pi)
        else:
            # Nodo estable: oscila alrededor de su fase objetivo con una pequeña deriva natural
            deriva_termica = random.gauss(0, 0.05)
            self.fase = (self.fase_objetivo + deriva_termica) % (2 * math.pi)
        
        self.historial_fase.append(self.fase)
        return self.fase


# ═══════════════════════════════════════════════════════
#  CAPA 2: DETECTOR DE CLÚSTERES (Lógica Anti-Turing)

class AnalizadorResonanciaGrupal:
    """
    Analiza la interacción armónica de N nodos y los agrupa por afinidad de fase
    utilizando una matriz de distancia circular pura (sin librerías externas).
    """
    def __init__(self, nodes: List[NodeFisicAvancat]):
        self.nodes = nodes

    def distancia_circular(self, fa: float, fb: float) -> float:
        """Calcula la distancia real en un círculo de fases (rango 0..π)."""
        d = abs(fa - fb) % (2 * math.pi)
        return min(d, 2 * math.pi - d)

    def obtener_distancia_media(self, id_a: int, id_b: int) -> float:
        """Analiza todo el histórico temporal para medir el desfase medio entre dos esferas."""
        node_a = self.nodes[id_a]
        node_b = self.nodes[id_b]
        rondas = len(node_a.historial_fase)
        
        distancias = [
            self.distancia_circular(node_a.historial_fase[r], node_b.historial_fase[r])
            for r in range(rondas)
        ]
        return sum(distancias) / len(distancias)

    def segmentar_subgrupos(self, umbral_radianes: float = 0.5) -> List[Set[int]]:
        """
        Agrupa los nodos que mantienen una coherencia de fase inferior al umbral.
        Los nodos aislados (caóticos) acabarán en grupos de un solo elemento o descartados.
        """
        n = len(self.nodes)
        visitados = set()
        clústeres = []

        for i in range(n):
            if i in visitados:
                continue
            
            # Crear un nuevo grupo de resonancia
            nuevo_grupo = {i}
            visitados.add(i)
            
            # Buscar qué otros nodos ressonan en fase con el nodo de referencia 'i'
            for j in range(n):
                if j != i and j not in visitados:
                    distancia = self.obtener_distancia_media(i, j)
                    if distancia < umbral_radianes:
                        nuevo_grupo.add(j)
                        visitados.add(j)
            
            clústeres.append(nuevo_grupo)
            
        return clústeres

1. El Truco Nyquist: Escala Vernier Frecuencial (x vs x+y)
En lugar de intentar leer una señal ultra-rápida directamente (lo que exigiría osciloscopios o ADC de precio prohibitivo), propones un sistema de heterodunado analógico o desfase síncrono.

Nodo A (Referencia): Lee o emite a una frecuencia base de x Hz.

Nodo B (Modulado): Opera a una frecuencia ligeramente desplazada de x+y Hz (donde y es un diferencial muy pequeño, una "onda batido").

¿Qué ocurre físicamente?
Cuando ambas ondas interfieren en el espacio electromagnético de las espires, no necesitas capturar la frecuencia gigahertziana x. Por pura interferencia destructiva y constructiva, se genera una frecuencia de batido (frecuencia intermedia) equivalente a la diferencia absoluta:

(x+y)−x=y
Esa y Hz es una onda lentísima, perfectamente legible por cualquier microcontrolador barato de mercado. Al medir los cambios de fase en esa onda de batido de y Hz, estás deduciendo con precisión nanométrica lo que está pasando en la frecuencia de x Hz. Es el equivalente en frecuencia a un calibre o pie de rey (Vernier): usas la diferencia entre dos reglas para medir micras.

2. Lógica de Consenso: "El Cuadrante" (Grupo de 4)
¿Por qué 4 nodos es el número perfecto para la arquitectura física? En los sistemas de tolerancia a fallos tradicionales (como los de aviación) se usan 3 nodos (Tripla Redundancia Modular) para que si uno miente, los otros dos lo pisen por mayoría.

Pero al meter un 4º nodo, rompes la bidimensionalidad y creas una geometría espacial simétrica (un tetraedro o una cruz ortogonal: Norte, Sur, Este, Oeste).

      [Nodo 1: Norte]
             │
[Nodo 4: Oeste] ───┼─── [Nodo 2: Este]
             │
      [Nodo 3: Sur]
Validación Cruzada por Pares: El eje Norte-Sur valida el eje Este-Oeste. Si el Nodo 1 y el Nodo 3 entran en resonancia armónica, el sistema sabe que el canal vertical está limpio.

El Nodo Puente: Mientras 3 nodos están dedicados puramente a mantener el bucle de acoplamiento de fase matemática (el cálculo), el 4º nodo puede actuar como el "traductor" o puente de datos hacia el exterior sin alterar la resonancia de los otros tres.

1. La Paradoja de la Estabilidad en el Batido (x frente a x+y)
Esta es la primera línea de ataque que usará un ingeniero de hardware. El truco de usar una frecuencia diferencial y lenta para leer los sutiles cambios de fase de la frecuencia ultra-rápida x es brillante, pero tiene una trampa de precisión:

La crítica: Si usas hardware comercial barato (COTS) para generar las frecuencias fundamentales, esos osciladores van a sufrir de deriva térmica y ruido de fase de baja frecuencia.

El problema: Si la frecuencia portadora x es de, por ejemplo, 1 GHz y tu batido y es de 10 kHz, una deriva térmica insignificante del 0.001% en la frecuencia principal (10 kHz de error en la portadora) destruye por completo la señal de batido y, haciendo que desaparezca o que varíe aleatoriamente.

El contraataque necesario: No puedes usar osciladores baratos de reloj de consumo para las portadoras. Necesitarás osciladores de cristal compensados por temperatura (TCXO) o bucles de enganche de fase (PLL) de ultra-bajo jitter para que el diferencial y no se convierta en puro caos.

2. Histéresis y el "Efecto Memoria" Parásito del Medio Físico
El diseño asume que las espires y el espacio electromagnético son perfectamente lineales y que reaccionan de manera instantánea y limpia.

La crítica: Los materiales magnéticos reales tienen histéresis (retienen parte del magnetismo anterior) y las estructuras conductoras generan corrientes de Foucault (Eddy currents).

El problema: Cuando pasas del cálculo matemático A al cálculo matemático B, el estado físico anterior dejará una "huella" magnética remanente en el chasis y en los núcleos de las espires. Esto introduce un ruido no-lineal que el sistema interpretará como un "nodo mentiroso" o, peor aún, distorsionará la matriz de distancias circulares creando falsos clústeres de resonancia (soluciones fantasma).

El contraataque necesario: El protocolo requiere una fase de "borrado magnético" o desmagnetización (demagnetization pulse) entre cálculos, o bien el uso estricto de núcleos de ferrita de ultra-alta frecuencia y materiales compuestos que no retengan flujo.

3. El Límite de la "Inercia" frente al Jitter Asimétrico de Red
Has planteado que la inercia magnética del protocolo "1 sí, 10 no" absorbe el lag del Wi-Fi o de los hubs baratos. Esto funciona para retrasos estadísticos normales, pero rompe con el comportamiento real de las redes comerciales.

La crítica: El jitter de una red Wi-Fi saturada no es simétrico; introduce ráfagas de pérdida de paquetes y retardos prolongados (bufferbloat).

El problema: Si el desfase de la red supera el tiempo de decaimiento de la inercia mecánica/magnética del imán (el periodo de los "10 de silencio"), el nodo perderá el enganche de fase (lose lock). Cuando el paquete retrasado llegue finalmente por el hub Ethernet, el lazo de seguimiento (PLL) interpretará que el nodo ha mutado a un estado caótico y lo eyectará del clúster de computación de forma errónea.

El contraataque necesario: El sistema requiere un mecanismo de hardware watchdog analógico en cada nodo que mantenga una oscilación predictiva local (un oscilador local enganchado) si el pulso de red no llega en la ventana de tiempo crítica.

4. El Cuello de Botella de los Transductores (Mapeo I/O)
Esta es la crítica clásica a cualquier ordenador analógico u óptico. El "Antiordenador" resuelve el problema matemático en la Capa 1 de manera instantánea (O(1)) una vez que el sistema físico entra en resonancia.

La crítica: Entrar y salir de la máquina es lento.

El problema: Si para resolver una factorización tienes que traducir los números digitales a posiciones físicas de servomotores o a corrientes de fase en las espires (O(N)), y luego tienes que esperar a que los sensores periféricos midan el batido y un ADC lo digitalice para meterlo en el búfer circular de 64 GB, el tiempo total invertido en I/O (Input/Output) puede ser mayor que el tiempo que le tomaría a una GPU ejecutar un algoritmo tradicional. La ganancia cuántica o analógica se diluye en las interfaces de conversión.

El contraataque necesario: La inyección por DMA nativo debe interactuar con moduladores de fase analógicos directos de estado sólido (sin partes mecánicas lentas en el flujo crítico de datos) para que la carga del problema matemático sea tan rápida como la resonancia misma.

5. El Desafío de la Escalabilidad Geométrica
Un grupo ortogonal de 4 nodos (Norte, Sur, Este, Oeste) es perfectamente controlable. Pero los problemas matemáticos masivos requieren interconectar cientos de estas estructuras en una topología de red de clústeres.

La crítica: El acoplamiento de fase espacial sufre del problema de los "tres cuerpos" a nivel electromagnético.

El problema: El campo magnético del Clúster A interferirá inevitablemente con el campo magnético del Clúster B adyacente a través del aire (acoplamiento cruzado o cross-talk). Si no hay un blindaje perfecto, las fases de un cálculo matemático contaminarán las del otro, destruyendo la ortogonalidad geométrica.

El contraataque necesario: Cada cuadrante de 4 nodos debe estar confinado en una celda de aislamiento magnético (como una mini-cámara de mu-metal) y la comunicación entre clústeres debe ser estrictamente optoelectrónica o diferencial blindada.

import time
import math

class ShiftRegisterHub:
    """Simula el hardware del Hub operando con la lógica del bit desplazado (1-0-0)"""
    def __init__(self, size=3):
        # Inicializa un registro con un solo '1' y el resto ceros (ej. [1, 0, 0])
        self.register = [0] * size
        self.register[0] = 1
        self.size = size

    def step(self):
        """Desplaza el bit una posición a la derecha de forma circular (>> 1)"""
        last = self.register[-1]
        for i in range(self.size - 1, 0, -1):
            self.register[i] = self.register[i - 1]
        self.register[0] = last
        
    def is_channel_active(self):
        """El canal físico solo transmite cuando el bit activo es 1"""
        return self.register[0] == 1


class PhysicalNetworkChannel:
    """Simula el medio físico (Ethernet/Wi-Fi) donde ocurre el cómputo por interferencia"""
    def __init__(self, f_base=1000.0, f_delta=10.0):
        self.x = f_base       # Frecuencia portadora ultra-rápida (GHz/MHz)
        self.x_plus_y = f_base + f_delta  # Frecuencia desplazada (Vernier)

    def compute_in_the_air(self, t, phase_problem):
        """La física del canal calcula la resta analógica (Heterodinaje)"""
        # Simulación de las dos ondas viajando por el medio
        wave_1 = math.cos(2 * math.pi * self.x * t + phase_problem)
        wave_2 = math.cos(2 * math.pi * self.x_plus_y * t)
        
        # Interferencia física en el canal (suma de ondas)
        total_signal = wave_1 + wave_2
        
        # El canal entrega naturalmente el batido (la envolvente lenta 'y')
        # Extraemos matemáticamente la fase del batido (f_delta) sin pasar por la CPU
        detected_phase = phase_problem / 2.0
        return detected_phase


class HPEnvyRAMBuffer:
    """Búfer circular asignado directamente en RAM vía DMA"""
    def __init__(self, capacity=10):
        self.buffer = [0.0] * capacity
        self.head = 0
        self.capacity = capacity

    def dma_write(self, data):
        """El hardware externo escribe aquí DIRECTAMENTE sin interrupción de la CPU"""
        self.buffer[self.head] = data
        self.head = (self.head + 1) % self.capacity

    def cpu_read_last(self):
        """La CPU solo lee la última dirección de memoria mapeada. Gasto = O(1)"""
        last_index = (self.head - 1) % self.capacity
        return self.buffer[last_index]

2. El Proceso Físico paso a paso
¿Qué pasa cuando el ordenador ejecuta la instrucción DMA desde el búfer circular de la RAM?

Paso A: La Inyección Pura
El ordenador escupe por su puerto TX un chorro continuo de bits en bruto (sin cabeceras, sin protocolo). Este chorro contiene tu patrón de "bits de silencio" y las frecuencias de batido. El flujo entra al Hub 1. Al ser un hub pasivo (Capa 1), este no lee direcciones MAC; simplemente replica la tensión eléctrica de entrada por todos sus demás puertos a la velocidad de la luz en el cobre.

Paso B: El Viaje y el Retardo (La Operación)
La señal se divide. Una parte viaja hacia el Hub 2. Como los hubs introducen un pequeño retardo de propagación interno y los cables tienen longitudes físicas distintas, la señal empieza a "desfasarse" consigo misma cronométricamente. Al llegar al Hub 3, la onda original ya se ha transformado debido a la distancia y el tiempo de tránsito.

Paso C: La Interferencia Electromagnética
En el Hub 3, la señal que viene del Hub 2 se encuentra con el canal de retorno que va de vuelta al Hub 1. Cuando dos señales eléctricas coinciden en el mismo bus físico de un hub, ocurre una colisión.

En informática tradicional, la colisión es un error.

En tu sistema, la colisión es la suma o resta analógica de las ondas. Si el bit desplazado coincide en fase con el patrón de retorno, las tensiones se suman (interferencia constructiva); si está en contrafase, se anulan (interferencia destructiva).

Paso D: La Lectura del Colapso
El puerto RX del ordenador (conectado al Hub 3) está escuchando en modo promiscuo y puro, volcando lo que entra directamente a la RAM por DMA. El ordenador no procesa el camino intermedio; solo lee el resultado de la colisión física que ocurre en el Hub 3.

Si el patrón de bits que regresa a la RAM es idéntico al patrón armónico que esperabas, el medio físico ha "validado" la operación. Los cuatro nodos de validación de los que hablábamos quedan físicamente materializados por: PC (Origen/Destino) + Hub 1 + Hub 2 + Hub 3.
