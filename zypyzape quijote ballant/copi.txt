Experiment runner with structured JSON logging, checkpoints and parallel runs.

Usage
-----

Features

- Per-run JSON logging with metadata and metrics.
- Periodic checkpointing to resume experiments.
- Multiprocessing Pool for parallel independent runs.
- Safe overrides handling and per-run exception capture.

def timestamp():
    return datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")

def safe_write_json(path, obj):
    tmp = path + ".tmp"
    with open(tmp, "w") as f:
        json.dump(obj, f, indent=2, default=_json_default)
    os.replace(tmp, path)

def _json_default(o):
    if isinstance(o, np.ndarray):
        return o.tolist()
    if isinstance(o, (np.integer, np.floating)):
        return float(o)
    if isinstance(o, (datetime,)):
        return o.isoformat()
    return str(o)

def load_checkpoint():
    if os.path.exists(CHECKPOINT_FILE):
        try:
            with open(CHECKPOINT_FILE, "r") as f:
                return json.load(f)
        except Exception:
            return None
    return None

def save_checkpoint(state):
    safe_write_json(CHECKPOINT_FILE, state)

def run_and_log_single(run_args):
    """
    Worker function to run a single experiment run and write per-run JSON.
    run_args is a dict with keys:
      - phase: 'baseline'|'policy'|'mc'|'ablation'|'surrogate'
      - seed, overrides (dict), T, extra kwargs
    """
    start_time = time.time()
    seed = int(run_args.get("seed", 0))
    phase = run_args.get("phase", "baseline")
    overrides = run_args.get("overrides", {}) or {}
    T = float(run_args.get("T", T_SHORT))
    extra = run_args.get("extra", {}) or {}

    run_id = f"{phase}_{timestamp()}_s{seed}"
    out_file = os.path.join(OUT_DIR, f"run_{run_id}.json")

    result = {
        "run_id": run_id,
        "phase": phase,
        "seed": seed,
        "overrides": overrides,
        "extra": extra,
        "start_time": timestamp(),
        "status": "failed",
        "error": None,
        "metrics": None,
        "runtime_s": None,
    }

    # Backup original module attributes we may override
    orig_vals = {}
    try:
        # Apply overrides safely
        for k, v in overrides.items():
            if hasattr(mod, k):
                orig_vals[k] = getattr(mod, k)
                setattr(mod, k, v)

        # Execute a short run using the model functions
        metrics = _run_short_local(seed=seed, T=T, overrides=overrides, extra=extra)

        result["metrics"] = metrics
        result["status"] = "ok"
    except Exception as e:
        tb = traceback.format_exc()
        result["error"] = {"message": str(e), "traceback": tb}
    finally:
        # restore original module attributes
        for k, v in orig_vals.items():
            setattr(mod, k, v)
        result["runtime_s"] = time.time() - start_time
        result["end_time"] = timestamp()
        # write per-run file
        try:
            safe_write_json(out_file, result)
        except Exception:
            pass
    return result

def _run_short_local(seed=7, T=T_SHORT, overrides=None, extra=None):
    """
    Local copy of the run_short logic adapted to be self-contained for multiprocessing.
    Returns a metrics dict similar to the earlier script.
    """
    overrides = overrides or {}
    extra = extra or {}

    # Save and apply overrides on module-level constants if present
    orig = {}
    for k, v in overrides.items():
        if hasattr(mod, k):
            orig[k] = getattr(mod, k)
            setattr(mod, k, v)

    STEPS = int(T / DT)
    t_arr = np.arange(STEPS) * DT
    v_arr = mod.wind_profile(t_arr)

    stA = mod.init_state(seed)
    stB = mod.init_state(seed)
    stB['theta'] = stA['theta'].copy()
    stB['omega'] = stA['omega'].copy()
    stB['omega_nat'] = stA['omega_nat'].copy()

    PA_l, PB_l, PQ3_l = [], [], []
    r_l, std_l, sync_l = [], [], []
    lam_l, tau_l, J_l  = [], [], []
    K_bus_l, K_ind_l   = [], []
    E_cyc_l            = []

    for k in range(STEPS):
        v_hub = v_arr[k]
        t_now = k * DT
        P_A_now = mod.power_MW(stA, v_hub)
        stA = mod.step_ZZ_only(stA, v_hub)

        # Run B step
        stB = mod.step_ZZ_Q3(stB, v_hub, P_A_now, t_now)

        if k % 4 == 0:
            pA = mod.power_MW(stA, v_hub)
            pB = mod.power_MW(stB, v_hub)
            pQ3 = mod.power_Q3_MW(stB, v_hub)
            PA_l.append(pA); PB_l.append(pB); PQ3_l.append(pQ3)
            r_l.append(mod.order_parameter(stB))
            std_l.append(mod.omega_std(stB))
            sync_l.append(mod.sync_count(stB))
            lam_l.append(mod.lambda_mean(stB, v_hub))
            tau_l.append(stB.get('tau_var', 0))
            J_l.append(stB.get('J_cost', 0))
            K_bus_l.append(stB.get('K_Q3_BUS', getattr(mod, "K_Q3_BUS_0", 0.05)))
            K_ind_l.append(stB.get('K_Q3_IND', getattr(mod, "K_Q3_IND_0", 0.09)))
            E_cyc_l.append(stB.get('E_cicle', 0)/1e6)

    # Restore overrides
    for k2, v2 in orig.items():
        setattr(mod, k2, v2)

    wu = len(PA_l) // 5
    PA = np.array(PA_l[wu:]); PB = np.array(PB_l[wu:])
    PQ3 = np.array(PQ3_l[wu:]); PBt = PB + PQ3
    dP = float(np.mean(PBt - PA)) if len(PA)>0 else 0.0

    J_arr = np.array(J_l[wu:]) if len(J_l)>wu else np.array([])
    if J_arr.size > 0:
        J_arr_w = np.clip(J_arr, np.percentile(J_arr, 5), np.percentile(J_arr, 95))
    else:
        J_arr_w = J_arr

    mask = np.array(E_cyc_l[wu:]) != 0
    E_m = float(np.mean(np.array(E_cyc_l[wu:])[mask])) if mask.sum()>0 else 0.0

    metrics = {
        'seed': seed,
        'dP_mean': dP,
        'dP_per_wt': dP / mod.N_ZZ * 1000 if mod.N_ZZ else dP,
        'P_A_mean': float(np.mean(PA)) if PA.size>0 else 0.0,
        'P_B_mean': float(np.mean(PBt)) if PBt.size>0 else 0.0,
        'P_std': float(np.std(PBt)) if PBt.size>0 else 0.0,
        'r_mean': float(np.mean(r_l[wu:])) if len(r_l)>wu else 0.0,
        'sync_mean': float(np.mean(sync_l[wu:])) if len(sync_l)>wu else 0.0,
        'std_om': float(np.mean(std_l[wu:])) if len(std_l)>wu else 0.0,
        'lambda': float(np.mean(lam_l[wu:])) if len(lam_l)>wu else 0.0,
        'tau_var': float(np.mean(tau_l[wu:])) if len(tau_l)>wu else 0.0,
        'J_mean': float(np.mean(J_arr_w)) if J_arr_w.size>0 else 0.0,
        'J_final': float(J_arr_w[-1]) if J_arr_w.size>0 else 0.0,
        'K_bus_fin': float(K_bus_l[-1]) if K_bus_l else getattr(mod, "K_Q3_BUS_0", 0.05),
        'K_ind_fin': float(K_ind_l[-1]) if K_ind_l else getattr(mod, "K_Q3_IND_0", 0.09),
        'E_cicle': E_m,
        'dP_pos_pct': float(100*np.mean(PBt > PA)) if PA.size>0 else 0.0,
        'policy_x': list(stB.get('_policy_x', [0]*5)),
    }
    return metrics

def run_montecarlo(n_runs=N_SEEDS_MC, workers=N_WORKERS):
    rng = np.random.default_rng(99)
    turbs = rng.uniform(0.06, 0.20, n_runs)
    gusts = rng.uniform(11.0, 17.0, n_runs)
    seeds = rng.integers(0, 10000, n_runs)
    args = []
    for i in range(n_runs):
        overrides = {"WIND_TURB": float(turbs[i]), "WIND_GUST_V": float(gusts[i])}
        args.append({"phase": "montecarlo", "seed": int(seeds[i]), "T": T_SHORT, "overrides": overrides})
    return _run_parallel(args, workers)

def run_ablation(workers=2):
    configs = {
        'full': {},
        'no_fft': {"disable_fft": True},
        'no_tsr': {"disable_tsr": True},
        'no_kind': {"disable_kind": True},
        'no_load_pen': {"disable_load": True},

    all_results = {}
    for name, cfg in configs.items():
        args = []
        for s in range(5):
            args.append({"phase": f"ablation_{name}", "seed": int(s), "T": T_SHORT, "overrides": cfg})
        res = _run_parallel(args, workers)
        all_results[name] = res
    return all_results

def run_surrogate_and_validate(workers=2):
    # This function will run a small set of extra points and perform a simple surrogate search.
    # For brevity we reuse run_short logic via run_and_log_single.
    rng = np.random.default_rng(77)
    extra_points = []
    for _ in range(20):
        x = list(rng.uniform(0.2, 0.9, 5))
        extra_points.append(x)
    surrogate_results = []
    for i, x in enumerate(extra_points):
        # decode policy to physical overrides
        p_w, m_e, K_b, N_m, K_i = mod.decode_policy(np.array(x))
        overrides = {"K_Q3_BUS_0": float(K_b), "K_Q3_IND_0": float(K_i)}
        res = run_and_log_single({"phase": "surrogate_extra", "seed": 1000+i, "T": 120.0, "overrides": overrides})
        surrogate_results.append(res)
    # simple selection of best
    valid = [r for r in surrogate_results if r.get("status")=="ok"]
    if not valid:
        return {"best": None, "all": surrogate_results}
    best = max(valid, key=lambda r: r["metrics"]["dP_mean"])
    return {"best": best, "all": surrogate_results}

# PARALLEL EXECUTION HELPER
# -------------------------
def _run_parallel(arg_list, workers):
    results_local = []
    if workers <= 1 or len(arg_list) == 1:
        for a in arg_list:
            results_local.append(run_and_log_single(a))
            # checkpoint after each run
            _consolidate_and_checkpoint()
    else:
        with Pool(processes=min(workers, len(arg_list))) as pool:
            for res in pool.imap_unordered(run_and_log_single, arg_list):
                results_local.append(res)
                _consolidate_and_checkpoint()
    return results_local

def _consolidate_and_checkpoint():
    # read all per-run files and consolidate into a single results dict
    runs = []
    for fname in sorted(os.listdir(OUT_DIR)):
        if fname.startswith("run_") and fname.endswith(".json"):
            try:
                with open(os.path.join(OUT_DIR, fname), "r") as f:
                    runs.append(json.load(f))
            except Exception:
                continue
    consolidated = {"runs": runs, "generated_at": timestamp()}
    try:
        safe_write_json(RESULTS_FILE, consolidated)
    except Exception:
        pass
    # checkpoint every CHECKPOINT_INTERVAL runs
    if len(runs) % CHECKPOINT_INTERVAL == 0:
        save_checkpoint({"runs_count": len(runs), "last_update": timestamp()})

    # 1 Baseline
    baseline = run_baseline_multi(n_seeds=N_SEEDS_BASE, workers=N_WORKERS)
    print("Baseline completed")
    # 2 Policy stability
    policy = run_policy_stability(n_runs=5, workers=1)
    print("Policy stability completed")

    mc = run_montecarlo(n_runs=N_SEEDS_MC, workers=N_WORKERS)

    print("Ablation completed")
    # 5 Surrogate
    surrogate = run_surrogate_and_validate(workers=2)
    print("Surrogate completed")
    # Final consolidation
    _consolidate_and_checkpoint()

if __name__ == "__main__":

def run_combination(r_in, r_out, p_win, m_exc, seed, T=T_SIM):
    # Guardem originals per restaurar
    orig_vals = {}
    # Alguns models exposen constants; no toquem globals crítics excepte via overrides si cal
    # Inicialitza estat
    stA = model.init_state(seed)
    stB = model.init_state(seed)
    stB['theta'] = stA['theta'].copy()
    stB['omega'] = stA['omega'].copy()
    stB['omega_nat'] = stA['omega_nat'].copy()

    # Forçar paràmetres de ratchet al state (si el model llegeix d'estat)
    stB['m_exc'] = m_exc
    stB['p_win'] = p_win

    # Històrics per càlculs
    omega_Q3_hist = []
    I_Q3_hist = []
    tau_buf = []

    STEPS = int(T / DT)
    t_arr = np.arange(STEPS) * DT
    v_arr = model.wind_profile(t_arr)

    for k in range(STEPS):
        v_hub = v_arr[k]
        t_now = k * DT
        P_A_now = model.power_MW(stA, v_hub)
        # Actualitzem stA
        stA = model.step_ZZ_only(stA, v_hub)

        # Per forçar el perfil r(theta) utilitzem una petita sobreescriptura local:
        # El model té funcions I_Q3_ratchet o similars en v3/v4; si no, fem una aproximació
        # Aquí definim r_exc dins del bucle segons sin(theta)
        # Actualitzem temporalment els valors que usa step_ZZ_Q3 mitjançant arguments si el model ho permet
        # Si el model no accepta overrides, modifiquem stB temporàniament
        # Assignem valors que el model utilitza: R_EXC_OUT, R_EXC_IN, M_EXC, P_WIN
        # Guardem originals
        if hasattr(model, "R_EXC_OUT"):
            orig_vals['R_EXC_OUT'] = model.R_EXC_OUT
            orig_vals['R_EXC_IN']  = model.R_EXC_IN
            orig_vals['M_EXC']     = getattr(model, "M_EXC", None)
            orig_vals['P_WIN']     = getattr(model, "P_WIN", None)
            model.R_EXC_OUT = float(r_out)
            model.R_EXC_IN  = float(r_in)
            # Alguns models usen M_EXC global
            if getattr(model, "M_EXC", None) is not None:
                model.M_EXC = float(m_exc)
            if getattr(model, "P_WIN", None) is not None:
                model.P_WIN = float(p_win)
        else:
            # Si no hi ha constants globals, intentem posar al state
            stB['m_exc'] = float(m_exc)
            stB['r_exc_out'] = float(r_out)
            stB['r_exc_in']  = float(r_in)
            stB['p_win']     = float(p_win)

        stB = model.step_ZZ_Q3(stB, v_hub, P_A_now, t_now)

        if hasattr(model, "I_Q3_ratchet"):

        if v is None:
            try:
                delattr(model, k)
            except Exception:
                pass

            setattr(model, k, v)

    I_Q3_arr = np.array(I_Q3_hist)

# Surrogate RBF Kernel Ridge (manual)
# -------------------------
def rbf_kernel(X, Y, sigma):
    # X: (n,d), Y: (m,d) -> (n,m)
    X = np.atleast_2d(X)
    Y = np.atleast_2d(Y)
    d2 = np.sum((X[:,None,:] - Y[None,:,:])**2, axis=-1)
    return np.exp(-d2 / (2.0 * sigma**2))

class KernelRidgeSurrogate:
    def __init__(self, sigma=None, alpha=1e-3):
        self.sigma = sigma
        self.alpha = alpha
        self.X = None
        self.y = None
        self.alpha_coef = None
        self.Kinv = None

    def fit(self, X, y):
        X = np.asarray(X, dtype=float)
        y = np.asarray(y, dtype=float)
        self.X = X.copy()
        self.y = y.copy()
        # sigma adaptativo: mediana de distancias si no se especifica
        if self.sigma is None:
            if len(X) > 1:
                dists = []
                for i in range(min(200, len(X))):
                    j = (i+1) % len(X)
                    dists.append(np.linalg.norm(X[i]-X[j]))
                med = np.median(dists) if dists else 1.0
                self.sigma = max(1e-3, med)
            else:
                self.sigma = 1.0
        K = rbf_kernel(self.X, self.X, self.sigma)
        # Regularización
        K_reg = K + self.alpha * np.eye(len(K))
        # Resolver coeficientes
        try:
            self.alpha_coef = np.linalg.solve(K_reg, self.y)
            # Precomputar inversa aproximada para varianza
            self.Kinv = np.linalg.inv(K_reg)
        except np.linalg.LinAlgError:
            # fallback a pseudo-inversa
            self.alpha_coef = np.linalg.lstsq(K_reg, self.y, rcond=None)[0]
            self.Kinv = np.linalg.pinv(K_reg)

    def predict(self, X_new):
        X_new = np.atleast_2d(X_new)
        K_new = rbf_kernel(X_new, self.X, self.sigma)
        mu = K_new.dot(self.alpha_coef)
        # var pred approx: 1 - k Kinv k^T (diagonal)
        # normalizamos para que var en [0,1] aproximadamente
        v = np.maximum(0.0, 1.0 - np.sum(K_new.dot(self.Kinv) * K_new, axis=1))
        return mu, v

def evaluate_point(model_path, params, T_eval=180.0, seed=None, timeout=None, verbose=False):
    """
    Ejecuta una simulación corta con el modelo y devuelve métricas.
    Devuelve dict con keys: success, dP_mean, r_mean, tau_var, E_cicle, E_grav, safe_flag, unsafe_reasons, raw_state (opcional)
    """
    try:
        mod = load_model_module(model_path)
        # Aplicar parámetros al estado inicial o a constantes del modelo según convenga
        # El modelo v3/v4 acepta overrides en step_ZZ_Q3 si se pasan argumentos; si no, intentamos setear atributos globales temporalmente.
        # Guardar originales
        orig = {}
        for attr in ['K_Q3_BUS_0','K_Q3_IND_0','P_WIN','M_EXC','R_EXC_OUT','R_EXC_IN']:
            if hasattr(mod, attr):
                orig[attr] = getattr(mod, attr)
        # Set globals si existen
        if hasattr(mod, 'P_WIN'):
            mod.P_WIN = float(params['p_win'])
        if hasattr(mod, 'M_EXC'):
            mod.M_EXC = float(params['m_exc'])
        if hasattr(mod, 'R_EXC_OUT'):
            mod.R_EXC_OUT = float(params['r_out'])
        if hasattr(mod, 'R_EXC_IN'):
            mod.R_EXC_IN = float(params['r_in'])
        # K defaults
        if hasattr(mod, 'K_Q3_BUS_0'):
            mod.K_Q3_BUS_0 = float(params['K_bus'])
        if hasattr(mod, 'K_Q3_IND_0'):
            mod.K_Q3_IND_0 = float(params['K_ind'])

        # Inicializar estados
        stA = mod.init_state(seed if seed is not None else 7)
        stB = mod.init_state(seed if seed is not None else 7)
        stB['theta'] = stA['theta'].copy()
        stB['omega'] = stA['omega'].copy()
        stB['omega_nat'] = stA['omega_nat'].copy()
        # Si el modelo lee m_exc/p_win desde state, setear
        stB['m_exc'] = float(params['m_exc'])
        stB['p_win'] = float(params['p_win'])
        # Ejecutar bucle
        STEPS = int(T_eval / mod.DT)
        t_arr = np.arange(STEPS) * mod.DT
        v_arr = mod.wind_profile(t_arr)
        rec_P_A = []
        rec_P_B = []
        rec_P_Q3 = []
        rec_r = []
        tau_buf = []
        for k in range(STEPS):
            v_hub = v_arr[k]
            t_now = k * mod.DT
            P_A_now = mod.power_MW(stA, v_hub)
            stA = mod.step_ZZ_only(stA, v_hub)
            stB = mod.step_ZZ_Q3(stB, v_hub, P_A_now, t_now)
            if k % 4 == 0:
                pA = mod.power_MW(stA, v_hub)
                pB = mod.power_MW(stB, v_hub)
                pQ3 = mod.power_Q3_MW(stB, v_hub)
                rec_P_A.append(pA)
                rec_P_B.append(pB)
                rec_P_Q3.append(pQ3)
                rec_r.append(mod.order_parameter(stB))
                tau_buf.append(stB.get('tau_var', 0.0))
        # Restaurar globals
        for k,v in orig.items():
            setattr(mod, k, v)
        # Postprocess
        rec_P_A = np.array(rec_P_A)
        rec_P_B = np.array(rec_P_B)
        rec_P_Q3 = np.array(rec_P_Q3)
        rec_r = np.array(rec_r)
        tau_arr = np.array(tau_buf)
        # Warmup: descartar primer 20%
        wu = max(1, len(rec_P_A)//5)
        PA = rec_P_A[wu:]; PB = rec_P_B[wu:]; PQ3 = rec_P_Q3[wu:]
        PB_tot = PB + PQ3
        dP_mean = float(np.mean(PB_tot - PA)) if len(PA)>0 else 0.0
        r_mean = float(np.mean(rec_r[wu:])) if len(rec_r)>wu else 0.0
        tau_var = float(np.mean(tau_arr)) if len(tau_arr)>0 else 0.0
        E_cicle = float(np.mean([v for v in (stB.get('E_cicle',0.0)/1e6, )])) if 'E_cicle' in stB else float(stB.get('E_cicle',0.0)/1e6 if stB.get('E_cicle',None) is not None else 0.0)
        E_grav = float(stB.get('E_grav_cicle',0.0)/1e6) if stB.get('E_grav_cicle',None) is not None else 0.0
        # Seguridad
        TAU_VAR_MAX = getattr(mod, 'TAU_VAR_MAX', None)
        I_MAX_STRUCT = getattr(mod, 'I_MAX_STRUCT', None)
        safe_flag = True
        reasons = []
        # Comprobar tau_var
        if TAU_VAR_MAX is not None and tau_var > TAU_VAR_MAX:
            safe_flag = False
            reasons.append('tau_var_exceeded')
        # Comprobar I_Q3 max si disponible
        # Intentamos estimar I_Q3 máxima si el modelo guarda I_Q3 en estado (no siempre)
        I_Q3_max = None
        if 'I' in stB:
            I_Q3_max = None
        if I_MAX_STRUCT is not None and I_Q3_max is not None:
            if I_Q3_max > I_MAX_STRUCT:
                safe_flag = False
                reasons.append('I_Q3_exceeded')
        result = {
            'success': True,
            'dP_mean': dP_mean,
            'r_mean': r_mean,
            'tau_var': tau_var,
            'E_cicle_MJ': E_cicle,
            'E_grav_MJ': E_grav,
            'safe_flag': bool(safe_flag),
            'unsafe_reasons': reasons,
            'seed': int(seed) if seed is not None else None,
            'params': params,
        }
        return result
    except Exception as e:
        tb = traceback.format_exc()
        return {'success': False, 'error': str(e), 'traceback': tb, 'params': params}

def J_discret_paper(m_q, r_hub, r_tip, N, steps=1000):
    r0 = (r_hub + r_tip)/2.0

    thetas_0 = np.array([2*np.pi*k/N for k in range(N)])
