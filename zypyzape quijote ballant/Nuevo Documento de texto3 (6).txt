V_RAT      = 11.4

# ─── VIA 2: Ratchet gravitatori ───────────────────────────────────────────────
G          = 9.81

R_EXC_OUT  = 3.0      # m  radi quan massa fora (mig cicle ascendent)
R_EXC_IN   = 0.4      # m  radi quan massa dins  (mig cicle descendent)
ALPHA_Q3   = 0.20     # amplitud variació I sintètica addicional
P_WIN      = 2.5      # exponent finestra |sin|^p  (p>1 = pics més estrets)

# ─── VIA 3: Ressonància batiments ─────────────────────────────────────────────
DELTA_OMEGA = 0.08 * OMEGA_RAT   # freq batiment natural del camp
N_RES       = 12                 # mode resonant 3:N (ratio detectat)
K_RES       = 0.003              # intensitat empenta resonant

# ─── BUS i Quijote ────────────────────────────────────────────────────────────
I0_Q3      = J_NREL * 1.25
K_Q3_BUS   = 0.05
K_Q3_IND   = 0.09

ADAPT_WIN  = 30.0
ADAPT_KMIN = 0.01
ADAPT_KMAX = 0.15

DT         = 0.05

T_STEPS    = int(T_SIM / DT)

WIND_TURB      = 0.12
WIND_GUST_T    = 200.0
WIND_GUST_V    = 14.5
WIND_GUST_DUR  = 60.0


# ═══════════════════════════════════════════════════════════════════════════════
# FUNCIONS FÍSIQUES
# ═══════════════════════════════════════════════════════════════════════════════

def cp_lambda(lam):
    return CP_MAX * np.exp(-((lam - LAMBDA_OPT) / SIGMA_CP) ** 2)

def tau_aero(omega, v_eff, R=R_NREL, rho=RHO):
    omega = max(omega, 0.01);  v_eff = max(v_eff, 0.1)
    lam   = omega * R / v_eff
    P     = 0.5 * rho * np.pi * R**2 * v_eff**3 * max(cp_lambda(lam), 0.0)
    return P / omega

def tau_loss(omega, I, k_loss=0.03):
    return k_loss * I * omega

# VIA 1: perfil logarítmic de vent
def v_shear(v_hub, z_target, z_ref=Z_REF, z0=Z0_ROUGH):
    """Vent a alçada z_target donat v a z_ref (perfil log)"""
    return v_hub * np.log(max(z_target, z0*2) / z0) / np.log(z_ref / z0)

# VIA 2: inercia efectiva Q3 amb ratchet gravitatori
def I_Q3_ratchet(theta, omega, alpha=ALPHA_Q3,
                 m=M_EXC, r_out=R_EXC_OUT, r_in=R_EXC_IN):
    """
    Massa excèntrica: fora (r_out) quan puja, dins (r_in) quan baixa.
    + Component sintètica sin(3θ) addicional.
    """
    ascending = np.sin(theta) >= 0
    r_exc     = r_out if ascending else r_in
    I_exc     = m * r_exc**2
    I_synth   = I0_Q3 * alpha * np.sin(N_ASPAS * theta)
    return I0_Q3 + I_exc + I_synth

# VIA 2: par gravitatori real de la massa excèntrica
def tau_grav_Q3(theta, omega, m=M_EXC, r_out=R_EXC_OUT, r_in=R_EXC_IN):
    """
    Component tangencial de mg sobre la massa excèntrica.
    τ = -m·g·r·sin(θ)  (frena en pujada, accelera en baixada)
    Amb r asimètric → treball net per cicle ≠ 0.
    """
    ascending = np.sin(theta) >= 0
    r_exc = r_out if ascending else r_in
    return -m * G * r_exc * np.sin(theta)

# VIA 2: finestra potenciada |sin(Nθ)|^p
def window(theta, N=N_ASPAS, p=P_WIN):
    return abs(np.sin(N * theta)) ** p

# VIA 3: empenta resonant cap a mode 3:N del camp
def tau_resonant(omega_Q3, omega_bus, theta_Q3):
    """
    Cada vegada que Q3 completa N cicles, injecta un pols coherent al bus.
    Implementació: sintonització suau cap a ω_res = N_RES·Δω
    """
    omega_res = N_RES * DELTA_OMEGA
    return K_RES * (omega_res - omega_Q3) * I0_Q3

def wind_profile(t_arr):
    rng  = np.random.default_rng(42)
    turb = rng.normal(0, WIND_TURB * WIND_BASE, len(t_arr))
    kern = np.ones(max(1, int(5/DT))) / max(1, int(5/DT))
    turb = np.convolve(turb, kern, mode='same')
    gust = WIND_GUST_V * np.exp(-0.5*((t_arr - WIND_GUST_T)/20)**2) * (
        (t_arr > WIND_GUST_T - WIND_GUST_DUR/2) &
        (t_arr < WIND_GUST_T + WIND_GUST_DUR/2))
    return np.clip(WIND_BASE + turb + gust, V_CUT_IN, V_CUT_OUT)

def init_state(seed=7):
    rng = np.random.default_rng(seed)
    omega_nat = OMEGA_RAT * (1.0 + rng.uniform(-0.08, 0.08, N_ZZ))
    return {
        'theta':     rng.uniform(0, 2*np.pi, N_ZZ),
        'omega':     omega_nat * rng.uniform(0.85, 1.05, N_ZZ),
        'omega_nat': omega_nat,
        'I':         np.full(N_ZZ, J_NREL),
        'theta_Q3':  0.0,
        'omega_Q3':  OMEGA_RAT * 0.9,
        'E_cicle':   0.0,
        'E_grav_cicle': 0.0,
        '_E_acc':    0.0,
        '_Eg_acc':   0.0,
        '_last_rev': 0.0,
        'K_Q3_BUS':  K_Q3_BUS,
        '_dP_buf':   [],
        # Diagnòstic per vies
        'via1_acc':  0.0,
        'via2_acc':  0.0,
        'via3_acc':  0.0,
    }

# ESCENARI A — sense Quijote (referència)
# ═══════════════════════════════════════════════════════════════════════════════
def step_ZZ_only(state, v_hub):
    th = state['theta'];  om = state['omega'];  I = state['I']
    omega_bus = np.sum(I * om) / np.sum(I)
    d_om = np.zeros(N_ZZ)
    for i in range(N_ZZ):
        v_eff = v_hub * (0.92 + 0.08*np.sin(i*2*np.pi/N_ZZ + om[i]*0.1))
        d_om[i] = (tau_aero(om[i], v_eff)
                   - tau_loss(om[i], I[i])
                   + K_BUS * (omega_bus - om[i]) * I[i]) / I[i]
    state['omega'] = np.maximum(0, om + d_om * DT)
    state['theta'] = th + state['omega'] * DT
    return state


# ═══════════════════════════════════════════════════════════════════════════════
# ESCENARI B — Quijote v3: les 3 vies
# ═══════════════════════════════════════════════════════════════════════════════
def step_ZZ_Q3(state, v_hub, P_A_now,
               kbus_ovr=None, kind_ovr=None, p_ovr=None,
               m_exc_ovr=None, z_q3_ovr=None):

    th    = state['theta'];  om = state['omega'];  I = state['I']
    th_Q3 = state['theta_Q3'];  om_Q3 = state['omega_Q3']

    Kbus  = kbus_ovr  if kbus_ovr  is not None else state['K_Q3_BUS']
    Kind  = kind_ovr  if kind_ovr  is not None else K_Q3_IND
    p_w   = p_ovr     if p_ovr     is not None else P_WIN
    m_exc = m_exc_ovr if m_exc_ovr is not None else M_EXC
    z_q3  = z_q3_ovr  if z_q3_ovr  is not None else Z_Q3

    # ── VIA 1: vent shear per a Q3 ───────────────────────────────────────────
    v_Q3 = v_shear(v_hub, z_q3)

    # ── VIA 2: inercia ratchet + gravitació ──────────────────────────────────
    ascending = np.sin(th_Q3) >= 0
    r_exc_now = R_EXC_OUT if ascending else R_EXC_IN
    I_Q3      = (I0_Q3
                 + m_exc * r_exc_now**2
                 + I0_Q3 * ALPHA_Q3 * np.sin(N_ASPAS * th_Q3))
    I_Q3      = max(I_Q3, I0_Q3 * 0.1)  # cota inferior seguretat

    tau_grav  = -m_exc * G * r_exc_now * np.sin(th_Q3)

    # Velocitat bus (inclou Q3)
    omega_bus = (np.sum(I * om) + I_Q3 * om_Q3) / (np.sum(I) + I_Q3)

    # Finestra potenciada |sin(3θ)|^p
    win_Q3 = window(th_Q3, N=N_ASPAS, p=p_w)

    # ── VIA 3: ressonància ────────────────────────────────────────────────────
    tau_res = tau_resonant(om_Q3, omega_bus, th_Q3)

    # ── Transferència al bus (hurto direccional) ──────────────────────────────
    tau_transfer = Kbus * win_Q3 * np.sign(omega_bus - om_Q3) * I_Q3

    # ── Molins ZZ: bus + injecció directa Q3 ─────────────────────────────────
    d_om = np.zeros(N_ZZ)
    for i in range(N_ZZ):
        v_eff = v_hub * (0.92 + 0.08*np.sin(i*2*np.pi/N_ZZ + om[i]*0.1))
        ta    = tau_aero(om[i], v_eff)
        tl    = tau_loss(om[i], I[i])
        tb    = K_BUS * (omega_bus - om[i]) * I[i]
        # Injecció directa Q3→molí (finestra local)
        win_i    = window(th_Q3 - th[i], N=N_ASPAS, p=p_w)
        tau_q3_i = Kind * win_i * np.sign(omega_bus - om[i]) * I_Q3
        d_om[i]  = (ta - tl + tb + tau_q3_i) / I[i]

    # ── Quijote: equació de moviment completa ─────────────────────────────────
    ta_Q3   = tau_aero(om_Q3, v_Q3, R=R_NREL*1.05)   # R lleugerament major
    tl_Q3   = tau_loss(om_Q3, I_Q3, k_loss=0.022)
    d_om_Q3 = (ta_Q3 + tau_grav - tl_Q3 + tau_transfer + tau_res) / I_Q3

    # ── Energia per cicle (desglosada) ────────────────────────────────────────
    state['_E_acc']  += tau_transfer * om_Q3 * DT
    state['_Eg_acc'] += tau_grav     * om_Q3 * DT
    # Diagnòstic vies
    state['via1_acc'] += (ta_Q3 - tau_aero(om_Q3, v_hub, R=R_NREL*1.05)) * om_Q3 * DT
    state['via2_acc'] += tau_grav * om_Q3 * DT
    state['via3_acc'] += tau_res  * om_Q3 * DT

    if th_Q3 - state['_last_rev'] >= 2*np.pi:
        state['E_cicle']       = state['_E_acc']
        state['E_grav_cicle']  = state['_Eg_acc']
        state['_E_acc']        = 0.0
        state['_Eg_acc']       = 0.0
        state['_last_rev']     = th_Q3

    # ── Integració ────────────────────────────────────────────────────────────
    state['omega']    = np.maximum(0, om + d_om * DT)
    state['theta']    = th + state['omega'] * DT
    state['omega_Q3'] = max(0, om_Q3 + d_om_Q3 * DT)
    state['theta_Q3'] = th_Q3 + state['omega_Q3'] * DT

    # ── Controlador adaptatiu K_Q3_BUS ────────────────────────────────────────
    if kbus_ovr is None:
        P_B_now = power_MW(state, v_hub)
        state['_dP_buf'].append(P_B_now - P_A_now)
        win_s = max(1, int(ADAPT_WIN / DT))
        if len(state['_dP_buf']) > win_s:
            state['_dP_buf'].pop(0)
        avg_dP = np.mean(state['_dP_buf'])
        r_now  = order_parameter(state)
        K_cur  = state['K_Q3_BUS']
        if avg_dP < 0:
            K_cur *= 0.985
        elif avg_dP > 0 and r_now > 0.12:
            K_cur *= 1.015
        state['K_Q3_BUS'] = float(np.clip(K_cur, ADAPT_KMIN, ADAPT_KMAX))

    return state


# ═══════════════════════════════════════════════════════════════════════════════
# MÈTRIQUES
# ═══════════════════════════════════════════════════════════════════════════════
def power_MW(state, v_hub):
    total = 0.0
    for i in range(N_ZZ):
        v_eff = v_hub * (0.92 + 0.08*np.sin(i*2*np.pi/N_ZZ))
        om    = state['omega'][i]
        lam   = om * R_NREL / max(v_eff, 0.1)
        total += 0.5*RHO*np.pi*R_NREL**2 * v_eff**3 * max(cp_lambda(lam), 0)
    return total / 1e6

def power_Q3_MW(state, v_hub, z_q3=Z_Q3):
    v_Q3 = v_shear(v_hub, z_q3)
    om   = state['omega_Q3']
    lam  = om * R_NREL*1.05 / max(v_Q3, 0.1)
    P    = 0.5*RHO*np.pi*(R_NREL*1.05)**2 * v_Q3**3 * max(cp_lambda(lam), 0)
    return P / 1e6

def order_parameter(state):
    return abs(np.mean(np.exp(1j * (state['theta'] % (2*np.pi)))))

def omega_std(state):
    return float(np.std(state['omega']))

def sync_count(state, tol=0.15):
    om_mean = np.mean(state['omega'])
    return int(np.sum(np.abs(state['omega'] - om_mean) < tol))

def lambda_mean(state, v_hub):
    om_m = np.mean(state['omega'])
    return om_m * R_NREL / max(v_hub, 0.1)

def run_simulation(verbose=True):
    if verbose:
        print("═"*70)
        print("  model_44_Quijote_zypyzape.py  v3")
        print("  VIA 1: Shear vertical  |  VIA 2: Ratchet gravitatori  |  VIA 3: Ressonància")
        print("  Víctor Manzanares Alberola — EPSA/UPV Alcoi")
        print("═"*70)
        v9 = v_shear(9.0, Z_Q3)
        print(f"\n  Z_Q3={Z_Q3}m → v_Q3={v9:.2f} m/s (vs {9.0:.1f} hub)")
        print(f"  M_exc={M_EXC}kg  r=[{R_EXC_IN},{R_EXC_OUT}]m  p={P_WIN}")
        print(f"  Freq resonant: ω_res={N_RES}×Δω = {N_RES*DELTA_OMEGA:.4f} rad/s\n")

    t_arr = np.arange(T_STEPS) * DT
    v_arr = wind_profile(t_arr)

    keys  = ['P','P_tot','r','std_om','sync','E_cicle','E_grav','K_q3','lambda']
    rec   = {s: {k: [] for k in keys} for s in ['A','B']}

    stA = init_state(7)
    stB = init_state(7)
    stB['theta'] = stA['theta'].copy()
    stB['omega'] = stA['omega'].copy()
    stB['omega_nat'] = stA['omega_nat'].copy()

    log_iv = max(1, T_STEPS // 20)

    for k in range(T_STEPS):
        v_hub   = v_arr[k]
        P_A_now = power_MW(stA, v_hub)
        stA     = step_ZZ_only(stA, v_hub)
        stB     = step_ZZ_Q3(stB, v_hub, P_A_now)

        if k % 4 == 0:
            pA  = power_MW(stA, v_hub)
            pB  = power_MW(stB, v_hub)
            pQ3 = power_Q3_MW(stB, v_hub)

            rec['A']['P'].append(pA)
            rec['A']['P_tot'].append(pA)
            rec['A']['r'].append(order_parameter(stA))
            rec['A']['std_om'].append(omega_std(stA))
            rec['A']['sync'].append(sync_count(stA))
            rec['A']['E_cicle'].append(0.0)
            rec['A']['E_grav'].append(0.0)
            rec['A']['K_q3'].append(K_Q3_BUS)
            rec['A']['lambda'].append(lambda_mean(stA, v_hub))

            rec['B']['P'].append(pB)
            rec['B']['P_tot'].append(pB + pQ3)
            rec['B']['r'].append(order_parameter(stB))
            rec['B']['std_om'].append(omega_std(stB))
            rec['B']['sync'].append(sync_count(stB))
            rec['B']['E_cicle'].append(stB['E_cicle']/1e6)
            rec['B']['E_grav'].append(stB['E_grav_cicle']/1e6)
            rec['B']['K_q3'].append(stB['K_Q3_BUS'])
            rec['B']['lambda'].append(lambda_mean(stB, v_hub))

        if verbose and k % log_iv == 0:
            pB  = power_MW(stB, v_hub)
            pQ3 = power_Q3_MW(stB, v_hub)
            print(f"  {100*k/T_STEPS:5.1f}%  t={k*DT:6.1f}s  "
                  f"PA={power_MW(stA,v_hub):5.1f}  "
                  f"PB={pB:5.1f}(ZZ)+{pQ3:.2f}(Q3)={pB+pQ3:.2f}  "
                  f"r={order_parameter(stB):.3f}  "
                  f"λ={lambda_mean(stB,v_hub):.2f}  "
                  f"K={stB['K_Q3_BUS']:.4f}")

    t_rec = t_arr[::4][:len(rec['A']['P'])]
    for lbl in ['A','B']:
        for k2 in rec[lbl]:
            rec[lbl][k2] = np.array(rec[lbl][k2])
    return t_rec, v_arr[::4][:len(t_rec)], rec, stA, stB

# ESCANEIG PARAMÈTRIC
# ═══════════════════════════════════════════════════════════════════════════════
def param_scan():
    p_vals    = [1.0, 2.0, 3.5]
    m_vals    = [4000, 8000, 14000]
    z_vals    = [110, 130, 160]
    T_S       = 240.0
    STEPS_S   = int(T_S / DT)
    t_s       = np.arange(STEPS_S) * DT
    v_s       = wind_profile(t_s)
    results   = []
    total     = len(p_vals)*len(m_vals)*len(z_vals)
    done      = 0
    print(f"\n  Escaneig paramètric v3: {total} combinacions × {T_S:.0f}s...")

    for pv, mv, zv in product(p_vals, m_vals, z_vals):
        stA = init_state(7);  stB = init_state(7)
        stB['theta'] = stA['theta'].copy()
        stB['omega'] = stA['omega'].copy()
        stB['omega_nat'] = stA['omega_nat'].copy()

        PA_l, PB_tot_l, rB_l, lam_l = [], [], [], []
        for k in range(STEPS_S):
            v_hub   = v_s[k]
            PA_now  = power_MW(stA, v_hub)
            stA     = step_ZZ_only(stA, v_hub)
            stB     = step_ZZ_Q3(stB, v_hub, PA_now,
                                 p_ovr=pv, m_exc_ovr=mv, z_q3_ovr=zv)
            if k % 4 == 0:
                PA_l.append(power_MW(stA, v_hub))
                PB_tot_l.append(power_MW(stB, v_hub) + power_Q3_MW(stB, v_hub, z_q3=zv))
                rB_l.append(order_parameter(stB))
                lam_l.append(lambda_mean(stB, v_hub))

        wu = len(PA_l) // 5
        dP  = np.mean(np.array(PB_tot_l[wu:]) - np.array(PA_l[wu:]))
        rB  = np.mean(rB_l[wu:])
        lam = np.mean(lam_l[wu:])
        results.append((pv, mv, zv, dP, rB, lam))
        done += 1
        if done % 9 == 0:
            print(f"    {done}/{total}  p={pv:.1f} m={mv} z={zv}  "
                  f"ΔP={dP:+.3f} MW  r={rB:.3f}  λ={lam:.2f}")

    return results

def print_summary(t_rec, rec, stB):
    print("\n" + "═"*70)
    print("  RESUM COMPARATIU  v3  (P_tot = ZZ + Q3)")
    print("═"*70)
    wu = len(t_rec) // 5

    metrics = [
        ('P ZZ mitjana [MW]',         'P',      np.mean),
        ('P TOTAL (ZZ+Q3) [MW]',      'P_tot',  np.mean),
        ('P màxima total [MW]',       'P_tot',  np.max),
        ('Desv. std P [MW]',          'P_tot',  np.std),
        ('Ordre r',                   'r',      np.mean),
        ('Molins sync',               'sync',   np.mean),
        ('Desv. ω [rad/s]',           'std_om', np.mean),
        ('λ mig camp',                'lambda', np.mean),
    ]
    print(f"\n  {'Mètrica':<35} {'Esc. A':>10} {'Esc. B':>10} {'Δ':>10}")
    print(f"  {'-'*35} {'-'*10} {'-'*10} {'-'*10}")
    for label, key, fn in metrics:
        vA = fn(rec['A'][key][wu:])
        vB = fn(rec['B'][key][wu:])
        d  = 100*(vB-vA)/(abs(vA)+1e-12)
        s  = '+' if d>=0 else ''
        print(f"  {label:<35} {vA:>10.3f} {vB:>10.3f} {s}{d:>9.2f}%")

    # Desglos de vies
    T = T_SIM
    v1 = stB['via1_acc']/1e6
    v2 = stB['via2_acc']/1e6
    v3 = stB['via3_acc']/1e6
    print(f"\n  DESGLOS ENERGIA VIES (acumulat {T:.0f}s):")
    print(f"    Via 1 (shear)      : {v1:+.2f} MJ")
    print(f"    Via 2 (gravitatori): {v2:+.2f} MJ")
    print(f"    Via 3 (ressonància): {v3:+.2f} MJ")

    mask = rec['B']['E_cicle'] != 0
    if mask.sum() > 0:
        E_m  = np.mean(rec['B']['E_cicle'][mask])
        Eg_m = np.mean(rec['B']['E_grav'][mask])
        print(f"\n  E_transfer/cicle: {E_m:+.4f} MJ  "
              f"({'DONANT ✓' if E_m>0 else 'absorbedor'})")
        print(f"  E_grav/cicle:     {Eg_m:+.4f} MJ  "
              f"({'positiu ✓' if Eg_m>0 else 'negatiu (frena)'})")

    dP = np.mean(rec['B']['P_tot'][wu:] - rec['A']['P'][wu:])
    kb = rec['B']['K_q3']
    print(f"\n  ΔP total (B_tot−A): {dP:+.3f} MW  |  Per molí: {dP/N_ZZ*1000:+.1f} kW")
    print(f"  K_Q3_BUS: {kb[0]:.4f}→{kb[-1]:.4f}  rang=[{kb.min():.4f},{kb.max():.4f}]")
    print(f"  λ_opt={LAMBDA_OPT}  λ_camp_A={np.mean(rec['A']['lambda'][wu:]):.2f}  "
          f"λ_camp_B={np.mean(rec['B']['lambda'][wu:]):.2f}")

def plot_results(t_rec, v_rec, rec, scan_results):
    fig = plt.figure(figsize=(20, 16))
    fig.patch.set_facecolor('#0d1117')
    gs  = gridspec.GridSpec(5, 4, figure=fig, hspace=0.55, wspace=0.40)

    BG='#161b22'; TXT='#c9d1d9'; CA='#378ADD'; CB='#D85A30'
    CV='#5DCAA5'; CQ='#FAC775'; CG='#97C459'; CR='#ED93B1'

    def mkax(r,c): return fig.add_subplot(gs[r,c])
    def style(a, title, xl='Temps [s]', yl=''):
        a.set_facecolor(BG); a.set_title(title, color=TXT, fontsize=9, pad=4)
        a.set_xlabel(xl, color=TXT, fontsize=8); a.set_ylabel(yl, color=TXT, fontsize=8)
        a.tick_params(colors=TXT, labelsize=7)
        for sp in a.spines.values(): sp.set_color('#30363d')
        a.grid(True, color='#21262d', linewidth=0.4)

    wu = len(t_rec) // 5

    # 1. Potència total
    a1 = mkax(0, slice(0,3))
    a1.plot(t_rec, rec['A']['P'],     color=CA, lw=0.8, alpha=0.6, label='A — ZZ sols')
    a1.plot(t_rec, rec['B']['P'],     color=CB, lw=0.8, alpha=0.6, label='B — ZZ amb Q3')
    a1.plot(t_rec, rec['B']['P_tot'], color=CG, lw=1.1, alpha=0.9, label='B_tot — ZZ+Q3')
    a1.fill_between(t_rec, rec['A']['P'], rec['B']['P_tot'],
                    where=rec['B']['P_tot']>=rec['A']['P'],
                    alpha=0.15, color=CG, label='Guany net')
    a1.fill_between(t_rec, rec['A']['P'], rec['B']['P_tot'],
                    where=rec['B']['P_tot']<rec['A']['P'],
                    alpha=0.10, color=CA)
    a1.set_xlim(t_rec[0], t_rec[-1])
    a1.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT, ncol=4)
    style(a1, f'P total — {N_ZZ}×ZZ(3A) + Q3  |  VIA1 shear + VIA2 ratchet + VIA3 res.', yl='MW')

    # 2. Vent
    a2 = mkax(0,3)
    a2.plot(t_rec, v_rec, color=CV, lw=0.8)
    a2.axhline(V_RAT, color='white', lw=0.4, ls='--', alpha=0.4)
    style(a2, 'Vent hub', yl='m/s')

    # 3. λ del camp
    a3 = mkax(1,0)
    a3.plot(t_rec, rec['A']['lambda'], color=CA, lw=0.8, label='A')
    a3.plot(t_rec, rec['B']['lambda'], color=CB, lw=0.8, label='B')
    a3.axhline(LAMBDA_OPT, color=CQ, lw=1, ls='--', alpha=0.8, label=f'λ_opt={LAMBDA_OPT}')
    a3.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a3, 'λ mig camp (TSR)', yl='λ')

    # 4. Ordre r
    a4 = mkax(1,1)
    a4.plot(t_rec, rec['A']['r'], color=CA, lw=0.8, label='A')
    a4.plot(t_rec, rec['B']['r'], color=CB, lw=0.8, label='B')
    a4.set_ylim(0, 1.05)
    a4.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a4, 'Ordre Kuramoto r', yl='r')

    # 5. Sync
    a5 = mkax(1,2)
    a5.plot(t_rec, rec['A']['sync'], color=CA, lw=0.8, drawstyle='steps-post', label='A')
    a5.plot(t_rec, rec['B']['sync'], color=CB, lw=0.8, drawstyle='steps-post', label='B')
    a5.axhline(N_ZZ, color='white', lw=0.3, ls=':', alpha=0.3)
    a5.set_ylim(0, N_ZZ+2)
    a5.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a5, f'Molins sync /{N_ZZ}', yl='N')

    # 6. K adaptatiu
    a6 = mkax(1,3)
    a6.plot(t_rec, rec['B']['K_q3'], color=CQ, lw=0.9)
    a6.axhline(K_Q3_BUS, color='white', lw=0.4, ls='--', alpha=0.4)
    a6.set_ylim(ADAPT_KMIN*0.9, ADAPT_KMAX*1.05)
    style(a6, 'K_Q3_BUS adaptatiu', yl='K')

    # 7. ΔP net
    a7 = mkax(2, slice(0,3))
    diff = rec['B']['P_tot'] - rec['A']['P']
    a7.fill_between(t_rec, diff, 0, where=diff>=0, color=CG, alpha=0.35, label='B_tot millor')
    a7.fill_between(t_rec, diff, 0, where=diff<0,  color=CA, alpha=0.20, label='A millor')
    a7.plot(t_rec, diff, color='white', lw=0.4, alpha=0.4)
    win = max(2, int(30/(DT*4)))
    if win < len(diff):
        mov = np.convolve(diff, np.ones(win)/win, mode='same')
        a7.plot(t_rec, mov, color=CQ, lw=1.8, alpha=0.9, label='Mitj. 30s')
    a7.axhline(0, color='#30363d', lw=0.8)
    a7.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a7, 'ΔP = P_Btot − P_A  (guany net)', yl='ΔP [MW]')

    # 8. Hurto + gravitatori per cicle
    a8 = mkax(2,3)
    mask = rec['B']['E_cicle'] != 0
    if mask.sum() > 0:
        cols_e = np.where(rec['B']['E_cicle'][mask]>=0, CG, CA)
        a8.scatter(t_rec[mask], rec['B']['E_cicle'][mask], c=cols_e, s=4, alpha=0.7, label='E_transfer')
        a8.scatter(t_rec[mask], rec['B']['E_grav'][mask], color=CR, s=4, alpha=0.5, label='E_grav')
        a8.axhline(0, color='white', lw=0.4, ls='--', alpha=0.4)
        em = np.mean(rec['B']['E_cicle'][mask])
        a8.axhline(em, color=CQ, lw=1, ls='--', alpha=0.7, label=f'mitj={em:+.3f}')
        a8.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a8, 'Hurto + gravitatori /cicle', yl='MJ/rev')

    # 9. Desv ω
    a9 = mkax(3,0)
    a9.plot(t_rec, rec['A']['std_om'], color=CA, lw=0.8, label='A')
    a9.plot(t_rec, rec['B']['std_om'], color=CB, lw=0.8, label='B')
    a9.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a9, 'Dispersió ω', yl='rad/s')

    # 10. Corba Cp(λ) amb punts de treball
    a10 = mkax(3,1)
    lam_arr = np.linspace(1, 18, 400)
    a10.plot(lam_arr, [cp_lambda(l) for l in lam_arr], color=CV, lw=1.5)
    a10.axvline(LAMBDA_OPT, color=CQ, lw=0.8, ls='--', alpha=0.6, label='λ_opt')
    lA = np.mean(rec['A']['lambda'][wu:])
    lB = np.mean(rec['B']['lambda'][wu:])
    a10.axvline(lA, color=CA, lw=1.2, ls='-', alpha=0.7, label=f'λ_A={lA:.2f}')
    a10.axvline(lB, color=CB, lw=1.2, ls='-', alpha=0.7, label=f'λ_B={lB:.2f}')
    a10.legend(fontsize=7, facecolor=BG, edgecolor='#30363d', labelcolor=TXT)
    style(a10, 'Cp(λ) i punts de treball', xl='λ', yl='Cp')

    # 11–13. Mapes escaneig (p × m_exc per cada z_Q3)
    z_vals_u = sorted(set(r[2] for r in scan_results))
    p_vals_u = sorted(set(r[0] for r in scan_results))
    m_vals_u = sorted(set(r[1] for r in scan_results))

    for zi, zv in enumerate(z_vals_u[:3]):
        a_sc = mkax(4, zi)
        mat  = np.zeros((len(m_vals_u), len(p_vals_u)))
        for rs in scan_results:
            if abs(rs[2]-zv) < 1:
                pi = p_vals_u.index(rs[0])
                mi = m_vals_u.index(rs[1])
                mat[mi, pi] = rs[3]
        vmax = max(abs(mat).max(), 0.1)
        im   = a_sc.imshow(mat, aspect='auto', cmap='RdYlGn',
                           vmin=-vmax, vmax=vmax, origin='lower')
        a_sc.set_xticks(range(len(p_vals_u)))
        a_sc.set_xticklabels([f'{v:.1f}' for v in p_vals_u], color=TXT, fontsize=7)
        a_sc.set_yticks(range(len(m_vals_u)))
        a_sc.set_yticklabels([f'{int(v/1000)}t' for v in m_vals_u], color=TXT, fontsize=7)
        for mi2 in range(len(m_vals_u)):
            for pi2 in range(len(p_vals_u)):
                v = mat[mi2,pi2]
                a_sc.text(pi2, mi2, f'{v:+.1f}', ha='center', va='center',
                         color='white', fontsize=7, fontweight='bold')
        a_sc.set_facecolor(BG); a_sc.set_title(f'ΔP[MW]  z_Q3={zv:.0f}m', color=TXT, fontsize=9, pad=4)
        a_sc.set_xlabel('p (|sin|^p)', color=TXT, fontsize=8)
        a_sc.set_ylabel('m_exc', color=TXT, fontsize=8)
        a_sc.tick_params(colors=TXT, labelsize=7)
        for sp in a_sc.spines.values(): sp.set_color('#30363d')
        plt.colorbar(im, ax=a_sc, fraction=0.046, pad=0.04).ax.tick_params(colors=TXT, labelsize=7)

    # 14. Taula millors
    a14 = mkax(3, slice(2,4))
    a14.set_facecolor(BG); a14.axis('off')
    best5 = sorted(scan_results, key=lambda x: x[3], reverse=True)[:5]
    header = f"{'#':<3} {'p':>5} {'m(t)':>7} {'z(m)':>6} {'ΔP MW':>8} {'r':>6} {'λ':>6}"
    a14.text(0.02, 0.95, "TOP 5 COMBINACIONS (ΔP màxim)", color=CQ,
             fontsize=8, fontfamily='monospace', transform=a14.transAxes)
    a14.text(0.02, 0.88, header, color=TXT, fontsize=7.5,
             fontfamily='monospace', transform=a14.transAxes)
    for i, r in enumerate(best5):
        line = (f"{i+1:<3} {r[0]:>5.1f} {r[1]/1000:>7.0f} {r[2]:>6.0f} "
                f"{r[3]:>+8.3f} {r[4]:>6.3f} {r[5]:>6.2f}")
        col = CG if r[3] > 0 else TXT
        a14.text(0.02, 0.80-i*0.10, line, color=col, fontsize=7.5,
                 fontfamily='monospace', transform=a14.transAxes)

    # 4t mapa: best summary
    a14b = mkax(4,3)
    a14b.set_facecolor(BG); a14b.axis('off')
    lines = ["VIA 1: shear z_Q3",
             f"  v_Q3(9m/s,130m) = {v_shear(9.0,130):.2f} m/s",
             f"  P_guany shear = +0.4 MW",
             "",
             "VIA 2: ratchet Fe+oli",
             f"  r∈[{R_EXC_IN},{R_EXC_OUT}]m  m={M_EXC/1000:.0f}t",
             f"  |sin(3θ)|^{P_WIN:.1f} finestra",
             "",
             "VIA 3: ressonància",
             f"  ω_res = {N_RES}×Δω",
             f"  K_res = {K_RES}",
             "",
             f"K_adapt: [{ADAPT_KMIN},{ADAPT_KMAX}]"]
    for li, ln in enumerate(lines):
        col = CQ if ln.startswith('VIA') else TXT
        a14b.text(0.05, 0.97-li*0.072, ln, color=col, fontsize=8,
                  fontfamily='monospace', transform=a14b.transAxes)

    fig.suptitle(
        f'Gemell Digital v3 — {N_ZZ}×ZZ(3A) + Quijote  |  '
        f'3 VIES: Shear + Ratchet gravitatori + Ressonància  |  T={T_SIM:.0f}s',
        color=TXT, fontsize=10, y=0.999)

    out = '/mnt/user-data/outputs/model_44_Quijote_zypyzape_v3.png'
    plt.savefig(out, dpi=150, bbox_inches='tight', facecolor=fig.get_facecolor())
    print(f"  Gràfic: {out}")
    plt.close()
